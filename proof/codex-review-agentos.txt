#              tmr_val tmr_re. GRUB prints values as %x (lowercase, no 0x, no padding); bit tests are regexps.
# Optional caller override: lock_mode (default refuse, set below). Callers source this file FIRST, then set it.
#   Policy is consumed per call: tco_arm resets lock_mode to refuse first thing, so set it before EACH call.
#   refuse : TCO_LOCK set -> do not arm (the original M1 design: it needs SMI_EN.TCO_EN cleared).
#   accept : TCO_LOCK set -> arm anyway, preserving TCO_LOCK and never writing SMI_EN: TCO_EN is not checked, so the
#            first timeout raises an SMI to the firmware and the reset relies on the second timeout. Only valid
#            once a hardware test showed this firmware's SMI handler does NOT swallow the timeout (NC110 step 1).
#            Linux iTCO_wdt does the same on v2 by default (turn_SMI_watchdog_clear_off=1 < version 2).
set lock_mode=refuse
# tco_arm sets wdt_ok=yes ONLY if every live predicate holds. The caller must boot only inside that branch.

# Bit predicates on %x strings:
#   GCS bit 5 (NO_REBOOT) clear      : nibble 1 in {0,1,4,5,8,9,c,d}
#   TCO1_CNT bit 11 (TMR_HLT) clear  : nibble 2 < 8
#   TCO1_CNT bit 12 (TCO_LOCK) clear : nibble 3 even
#   SMI_EN bit 13 (TCO_EN) clear     : nibble 3 in {0,1,4,5,8,9,c,d}
set re_noreboot_clear='^([0-9a-f]|[0-9a-f]*[014589cd][0-9a-f])$'

set re_lock_clear='^([0-9a-f]{1,3}|[0-9a-f]*[02468ace][0-9a-f]{3})$'

function tco_re_check {
  # $1 = regexp, $2 = value, $3 = expected (yes/no). Sets re_bad=yes on a wrong answer.
  if regexp "$1" "$2"; then set got=yes; else set got=no; fi
  if [ "$got" != "$3" ]; then echo "wdt: selftest FAILED: $2 vs $1 gave $got"; set re_bad=yes; fi
}

function tco_selftest {
  # Proves THIS GRUB's regexp/test behave as the predicates assume, before they guard anything.
  set re_bad=no
  tco_re_check "$re_noreboot_clear" "0" yes
  tco_re_check "$re_noreboot_clear" "fed1f401" yes
  tco_re_check "$re_noreboot_clear" "20" no
  tco_re_check "$re_noreboot_clear" "fed1f421" no
  tco_re_check "$re_hlt_clear" "0" yes
  tco_re_check "$re_hlt_clear" "1200" yes
  tco_re_check "$re_hlt_clear" "800" no
  tco_re_check "$re_hlt_clear" "1a00" no
  tco_re_check "$re_lock_clear" "800" yes
  tco_re_check "$re_lock_clear" "2000" yes
  tco_re_check "$re_lock_clear" "1000" no
  tco_re_check "$re_lock_clear" "3800" no
  tco_re_check "$re_tcoen_clear" "21" yes
  tco_re_check "$re_tcoen_clear" "1021" yes
  tco_re_check "$re_tcoen_clear" "2021" no
  tco_re_check "$re_tcoen_clear" "f021" no
  tco_re_check "$re_hex16" "1000" yes

  tco_re_check "$re_hex16" "11000" no

  if [ "$got" != "yes" ]; then echo "wdt: selftest FAILED: test -o"; set re_bad=yes; fi
  if [ "$re_bad" = "no" ]; then echo "wdt: selftest OK (21 regexp cases + test -o)"; fi
}

function tco_arm {
  set wdt_ok=no
  set tco_mode="$lock_mode"
  set lock_mode=refuse
  tco_selftest
  if [ "$re_bad" != "no" ]; then echo "wdt: GRUB predicates untrustworthy, NOT arming"; return 1; fi
  set lpcid=unset; set pmbase=unset; set rcba=unset
  setpci -s 00:1f.0 -v lpcid 0.l
  setpci -s 00:1f.0 -v pmbase 40.l
  setpci -s 00:1f.0 -v rcba f0.l
  echo "wdt: lpcid=$lpcid pmbase=$pmbase rcba=$rcba"
  if [ "$lpcid" != "$lpc_expect" -o "$pmbase" != "$pmbase_expect" -o "$rcba" != "$rcba_expect" ]; then
    echo "wdt: chipset mismatch, NOT arming"; return 1
  fi
  set cntpre=unset; set sts2pre=unset; set gcspre=unset
  inw -v cntpre $tco1_cnt
  inw -v sts2pre $tco2_sts
  read_dword -v gcspre $gcs_addr
  echo "wdt: before: GCS=$gcspre TCO1_CNT=$cntpre TCO2_STS=$sts2pre (6 = last reset was the watchdog)"
  if regexp "$re_hex16" "$cntpre"; then true; else echo "wdt: TCO1_CNT read invalid ($cntpre), NOT arming"; return 1; fi
  if regexp "$re_lock_clear" "$cntpre"; then set tco_locked=no; else set tco_locked=yes; fi
  if [ "$tco_locked" = "yes" ]; then
    if [ "$tco_mode" != "accept" ]; then echo "wdt: TCO_LOCK set, NOT arming (lock_mode=$tco_mode)"; return 1; fi
    echo "wdt: TCO_LOCK set, lock_mode=accept: TCO_EN left as read, 1st timeout goes to firmware SMI, reset on 2nd"
  fi
  write_dword $gcs_addr 0 0x20       # GCS: clear NO_REBOOT (bit 5), masked RMW
  if [ "$tco_locked" = "no" ]; then
    outl $smi_en 0 0x2000            # SMI_EN: clear TCO_EN (bit 13), masked RMW; locked: never written
  fi
  outw $tco1_sts 0x0008              # TCO1_STS: clear TIMEOUT (RW1C)
  outw $tco2_sts 0x0002              # TCO2_STS: clear SECOND_TO_STS first ...
  outw $tco2_sts 0x0004              # ... then BOOT_STS (Intel order)
  outw $tco_tmr $tmr_val 0x03ff      # TCO_TMR: ticks of 0.6 s in bits 9:0 only
  outw $tco_rld 1                    # TCO_RLD: reload
  outw $tco1_cnt 0 0x0800            # TCO1_CNT: clear TCO_TMR_HLT (bit 11), masked RMW; bit 12 TCO_LOCK written back as read
  set gcs=unset; set cnt=unset; set tmr=unset; set smien=unset; set c0=unset; set c1=unset
  read_dword -v gcs $gcs_addr
  inw -v cnt $tco1_cnt
  inw -v tmr $tco_tmr
  inl -v smien $smi_en
  inw -v c0 $tco_rld
  sleep 1
  inw -v c1 $tco_rld
  echo "wdt: after: GCS=$gcs TCO1_CNT=$cnt TCO_TMR=$tmr SMI_EN=$smien count $c0 -> $c1 locked=$tco_locked mode=$tco_mode"
  if regexp "$re_noreboot_clear" "$gcs"; then true; else echo "wdt: NO_REBOOT still set"; return 1; fi
  if regexp "$re_hlt_clear" "$cnt"; then true; else echo "wdt: timer still halted"; return 1; fi
  if [ "$tco_locked" = "no" ]; then
    if regexp "$re_tcoen_clear" "$smien"; then true; else echo "wdt: SMI_EN.TCO_EN still set"; return 1; fi
  fi
  if regexp "$tmr_re" "$tmr"; then true; else echo "wdt: TCO_TMR not $tmr_val"; return 1; fi
  if [ "$c0" = "$c1" -o "$c0" = "unset" -o "$c1" = "unset" ]; then echo "wdt: counter not moving"; return 1; fi
  set wdt_ok=yes
}

//! In-RAM chain loader, driven over bmctl (the DS OTA equivalent: nothing is written to /boot).
//! `lbeg <size> <sha256>` opens a staging buffer at STAGE; binary `ldat` ++ u32 offset ++ bytes appends in order;
//! `lcom <cmdline>` verifies length + SHA-256 + the multiboot header (a.out kludge, flag bit 16, as GRUB applies it),
//! checks every destination range, builds a multiboot info block (cmdline, the ORIGINAL GRUB memory map, and a
//! rebuilt 72-byte key module with this boot's seed), quiesces the NIC and jumps through the trampoline in boot.s,
//! which turns paging off, copies the image to load_addr, zeroes its bss and enters it like GRUB would.
//! The TCO stays armed across the hop (the attended successor re-arms it in kmain). Any refusal leaves us running.
use crate::kprintln;
use crate::mb;
use crate::nic::Rtl;
use sha2::{Digest, Sha256};

pub const STAGE: u64 = 0x0400_0000; // 64 MiB
pub const CAP: u64 = 0x0080_0000; // 8 MiB (STAGE + CAP < 128 MiB, so q35 -m 128 works)
const TRAMP: u64 = 0x8000; // trampoline code + its stack (< 0xC00)
const PARAMS: u64 = 0x8C00;
const INFO: u64 = 0x8D00;
const MODENT: u64 = 0x8E80;
const CMDLINE: u64 = 0x8F00; // <= 255 bytes + NUL
const MODULE: u64 = 0x9000; // page-aligned (our header asks for it: flag bit 0)
const MMAP: u64 = 0x9100;
const LOW_END: u64 = 0xA000;
const MMAP_MAX: usize = (LOW_END - MMAP) as usize;

/// GRUB's boot metadata, copied out at boot (before any staging could overwrite it). (flags, mem_lower, mem_upper)
static mut ORIG: (u32, u32, u32) = (0, 0, 0);
static mut ORIG_MMAP: [u8; MMAP_MAX] = [0; MMAP_MAX];
static mut ORIG_MMAP_LEN: usize = 0;
static mut ORIG_MMAP_TOO_BIG: bool = false;

/// Call once, early, with the multiboot info pointer GRUB (or a previous loader) gave us.
pub fn snapshot_boot_info(info: u64) {
    if info == 0 {
        return;
    }
    unsafe {
        let r32 = |a: u64| core::ptr::read_volatile(a as *const u32);
        let f = r32(info);
        ORIG = (f, if f & 1 != 0 { r32(info + 4) } else { 0 }, if f & 1 != 0 { r32(info + 8) } else { 0 });
        if f & (1 << 6) != 0 {
            let (l, a) = (r32(info + 44) as usize, r32(info + 48) as u64);
            if l <= MMAP_MAX {
                core::ptr::copy(a as *const u8, core::ptr::addr_of_mut!(ORIG_MMAP) as *mut u8, l);
                ORIG_MMAP_LEN = l;
            } else {
                ORIG_MMAP_TOO_BIG = true;
            }
        }
    }
}

extern "C" {
    static tramp_start: u8;
    static tramp_end: u8;
    static __load_start: u8;
    static __bss_end: u8;
}

pub struct Stage {
    size: u64,
    sha: [u8; 32],
    got: u64,
}

fn overlaps(a: u64, al: u64, b: u64, bl: u64) -> bool {
    a < b.saturating_add(bl) && b < a.saturating_add(al)
}

fn kernel_image() -> (u64, u64) {
    let (s, e) = (core::ptr::addr_of!(__load_start) as u64, core::ptr::addr_of!(__bss_end) as u64);
    (s, e - s)
}

pub fn begin(size: u64, sha_hex: &str) -> Result<Stage, &'static str> {
    if size == 0 || size > CAP {
        return Err("size 1..8 MiB");
    }
    let (ks, kl) = kernel_image();
    if !mb::ram_contains(STAGE, size) || overlaps(STAGE, size, ks, kl) {
        return Err("staging area not free RAM");
    }
    if sha_hex.len() != 64 || !sha_hex.bytes().all(|c| c.is_ascii_hexdigit()) {
        return Err("sha256: 64 hex chars");
    }
    let mut sha = [0u8; 32];
    for (i, b) in sha.iter_mut().enumerate() {
        *b = u8::from_str_radix(&sha_hex[2 * i..2 * i + 2], 16).map_err(|_| "sha256: hex")?;
    }
    Ok(Stage { size, sha, got: 0 })
}

pub fn data(st: &mut Stage, off: u64, bytes: &[u8]) -> Result<u64, &'static str> {
    if off != st.got {
        return Err("out of order (send chunks in sequence)");
    }
    let end = off.checked_add(bytes.len() as u64).ok_or("overflow")?;
    if end > st.size {
        return Err("past announced size");
    }
    unsafe { core::ptr::copy_nonoverlapping(bytes.as_ptr(), (STAGE + off) as *mut u8, bytes.len()) };
    st.got = end;
    Ok(end)
}

/// Everything checked, nothing moved yet. Returns (src, dst, len, bss_len, entry).
fn plan(st: &Stage) -> Result<(u64, u64, u64, u64, u64), &'static str> {
    if st.got != st.size {
        return Err("incomplete image");
    }
    let img = unsafe { core::slice::from_raw_parts(STAGE as *const u8, st.size as usize) };
    let d: [u8; 32] = Sha256::digest(img).into();
    if d != st.sha {
        return Err("sha256 mismatch");
    }
    let rd = |o: usize| u32::from_le_bytes(img[o..o + 4].try_into().unwrap());
    let lim = img.len().min(8192);
    let hoff = (0..lim.saturating_sub(31)).step_by(4).find(|&o| {
        rd(o) == 0x1BAD_B002 && rd(o).wrapping_add(rd(o + 4)).wrapping_add(rd(o + 8)) == 0
    }).ok_or("no multiboot header in the first 8 KiB")?;
    let flags = rd(hoff + 4);
    if flags & (1 << 16) == 0 {
        return Err("multiboot header lacks the address kludge (flag 16)");
    }
    if flags & 0xFFFC != 0 {
        return Err("multiboot header requires features this loader does not provide (bits 2..15)");
    }
    let (haddr, laddr, lend, bend, entry) =
        (rd(hoff + 12) as u64, rd(hoff + 16) as u64, rd(hoff + 20) as u64, rd(hoff + 24) as u64, rd(hoff + 28) as u64);
    if laddr > haddr || lend <= laddr || bend < lend || entry < laddr || entry >= lend {
        return Err("inconsistent header addresses");
    }
    let foff = (hoff as u64).checked_sub(haddr - laddr).ok_or("header offset")?;
    let len = lend - laddr;
    if foff.checked_add(len).is_none_or(|e| e > st.size) {
        return Err("image shorter than load_end - load_addr");
    }
    if laddr < 0x10_0000 || bend > STAGE || !mb::ram_contains(laddr, bend - laddr) {
        return Err("target range not RAM in [1 MiB, staging)");
    }
    if overlaps(laddr, bend - laddr, TRAMP, LOW_END - TRAMP) {
        return Err("target overlaps the trampoline/info pages");
    }
    Ok((STAGE + foff, laddr, len, bend - lend, entry))
}

pub struct Plan {
    params: [u64; 6],
}

/// All checks plus writing the low pages (trampoline, info, module, cmdline, mmap). Nothing of ours is touched:
/// the low pages are outside the kernel image; GRUB's info/mmap come from the boot-time snapshot.
pub fn prepare(st: &Stage, cmdline: &str, seed: &[u8; 32]) -> Result<Plan, &'static str> {
    let (src, dst, len, bss, entry) = plan(st)?;
    let tl = (core::ptr::addr_of!(tramp_end) as u64) - (core::ptr::addr_of!(tramp_start) as u64);
    if tl > PARAMS - TRAMP || !mb::ram_contains(TRAMP, LOW_END - TRAMP) {
        return Err("low pages 0x8000-0xA000 not usable");
    }
    if cmdline.len() > 255 {
        return Err("cmdline > 255");
    }
    unsafe {
        if ORIG_MMAP_TOO_BIG {
            return Err("GRUB memory map larger than the loader's buffer: refusing rather than dropping it");
        }
        let (oflags, mlo, mhi) = ORIG;
        let ml = ORIG_MMAP_LEN;
        let mmap = &*core::ptr::addr_of!(ORIG_MMAP);
        let w32 = |a: u64, v: u32| core::ptr::write_volatile(a as *mut u32, v);
        core::ptr::write_bytes(TRAMP as *mut u8, 0, (LOW_END - TRAMP) as usize);
        core::ptr::copy_nonoverlapping(core::ptr::addr_of!(tramp_start), TRAMP as *mut u8, tl as usize);
        // key module: MAGIC + seed + public key, exactly the 72 bytes GRUB's `module` line gave us
        core::ptr::copy_nonoverlapping(mb::MAGIC.as_ptr(), MODULE as *mut u8, 8);
        core::ptr::copy_nonoverlapping(seed.as_ptr(), (MODULE + 8) as *mut u8, 32);
        let publ = sunset::ed25519_dalek::SigningKey::from_bytes(seed).verifying_key().to_bytes();
        core::ptr::copy_nonoverlapping(publ.as_ptr(), (MODULE + 40) as *mut u8, 32);
        w32(MODENT, MODULE as u32);
        w32(MODENT + 4, (MODULE + 72) as u32);
        core::ptr::copy_nonoverlapping(cmdline.as_ptr(), CMDLINE as *mut u8, cmdline.len());
        // multiboot info: flags 0 (mem), 2 (cmdline), 3 (mods), 6 (GRUB's memory map, verbatim)
        let mut flags = (1 << 2) | (1 << 3);
        if oflags & 1 != 0 {
            flags |= 1;
            w32(INFO + 4, mlo);
            w32(INFO + 8, mhi);
        }
        if ml > 0 {
            core::ptr::copy_nonoverlapping(mmap.as_ptr(), MMAP as *mut u8, ml);
            flags |= 1 << 6;
            w32(INFO + 44, ml as u32);
            w32(INFO + 48, MMAP as u32);
        }
        w32(INFO, flags);
        w32(INFO + 16, CMDLINE as u32);
        w32(INFO + 20, 1);
        w32(INFO + 24, MODENT as u32);
    }
    kprintln!("loader: ready: {} B image -> {:#x}..{:#x} (+{} B bss), entry {:#x}, cmdline '{}'", st.size, dst, dst + len, bss, entry, cmdline);
    Ok(Plan { params: [src, dst, len, bss, entry, INFO] })
}

/// Point of no return: NIC quiesced (reset completed), trampoline entered. The TCO stays armed across the hop.
/// If the NIC reset does not complete, its DMA may still target memory the new image would overwrite: hard reset
/// instead of jumping (back to Slot A via GRUB's default).
pub fn jump(p: &Plan, nic: &Rtl) -> ! {
    if let Err(e) = nic.quiesce() {
        kprintln!("loader: NIC quiesce failed ({}): NOT jumping, hard reset", e);
        crate::wdt::hard_reset();
    }
    unsafe {
        for (i, v) in p.params.iter().enumerate() {
            core::ptr::write_volatile((PARAMS + 8 * i as u64) as *mut u64, *v);
        }
        crate::wdt::pet();
        let f: extern "sysv64" fn(u64) -> ! = core::mem::transmute(TRAMP as usize);
        f(PARAMS)
    }
}

/// Negative control (`--features neg-control`, QEMU q35 only; boot.s never self-arms on NM10): boot.s SETS GCS.NO_REBOOT instead of clearing it
/// (QEMU's GCS powers up 0 with noreboot=false, so merely skipping the clear would prove nothing).

extern "C" {
    static tco_result: u32;
    static tco_gcs_pre: u32;
    static tco_gcs: u32;
    static tco_base: u32;
    static tco_sts1_pre: u16;
    static tco_sts2_pre: u16;
    static tco_cnt: u16;
}

fn rd<T: Copy>(p: *const T) -> T {
    unsafe { read_volatile(p) }
}

pub fn armed() -> bool {
    matches!(rd(addr_of!(tco_result)), 0 | 4 | 5)
}

pub fn pmbase() -> u16 {
    (pci::read32(pci::b0(31, 0), 0x40) & 0xFF80) as u16
}

pub fn base() -> u16 {
    pmbase() + 0x60
}

pub fn report() {
    let r = rd(addr_of!(tco_result));
    let what = match r {
        0 => "armed by boot.s",
        1 => "skipped (LPC 00:1f.0 is not ICH9 8086:2918 / NM10 8086:27bc)",
        2 => "NOT armed: PMBASE unset",
        3 => "NOT armed: RCBA disabled",
        4 => "left to GRUB (grub-armed build); live registers follow",
        5 => "NM10: boot.s does not arm (attended Rust arms it; else GRUB must); live registers follow",
        _ => "NOT armed: unknown",
    };
    kprintln!("wdt: {} (result={})", what, r);
    if r == 4 || r == 5 {
        let b = base();
        let gcs = unsafe { read_volatile(((pci::read32(pci::b0(31, 0), 0xF0) & 0xFFFF_C000) + 0x3410) as *const u32) };
        let (s1, s2, cnt, tmr) = unsafe { (inw(b + 4), inw(b + 6), inw(b + 8), inw(b + 0x12)) };
        kprintln!(
            "wdt: live TCOBASE={:#x} GCS={:#010x} NO_REBOOT={} TCO1_CNT={:#06x} HLT={} TCO1_STS={:#06x} TCO2_STS={:#06x} TCO_TMR={}",
            b, gcs, (gcs >> 5) & 1, cnt, (cnt >> 11) & 1, s1, s2, tmr & 0x3FF
        );
        return;
    }
    if r != 0 {
        return;
    }
    let (b, gcs, s1, s2, cnt) = {
        (rd(addr_of!(tco_base)) as u16, rd(addr_of!(tco_gcs)), rd(addr_of!(tco_sts1_pre)), rd(addr_of!(tco_sts2_pre)), rd(addr_of!(tco_cnt)))
    };
    kprintln!(
        "wdt: TCOBASE={:#x} GCS {:#010x} -> {:#010x} NO_REBOOT={} TCO1_CNT={:#06x} HLT={} LOCK={}",
        b, rd(addr_of!(tco_gcs_pre)), gcs, (gcs >> 5) & 1, cnt, (cnt >> 11) & 1, (cnt >> 12) & 1
    );
    kprintln!(
        "wdt: before-clear TCO1_STS={:#06x} TCO2_STS={:#06x} SECOND_TO_STS={} BOOT_STS={}",
        s1, s2, (s2 >> 1) & 1, (s2 >> 2) & 1
    );
    let tmr = unsafe { inw(b + 0x12) } & 0x3FF;
    kprintln!("wdt: TCO_TMR={} ticks x 0.6 s, reset expected after 2nd expiry (~{} ms)", tmr, tmr as u32 * 1200);
}

/// Live check of the reset path, whoever armed it: NO_REBOOT=0, HLT=0, TMR>=4, and the counter
/// (TCO_RLD reads back the current value on v2) decrements across ~1.3 ticks.
pub fn verify_live() -> bool {
    if !armed() {
        return false;
    }
    let b = base();
    let rcba = pci::read32(pci::b0(31, 0), 0xF0);
    if rcba & 1 == 0 {
        return false;
    }
    let gcs = unsafe { read_volatile(((rcba & 0xFFFF_C000) + 0x3410) as *const u32) };
    let (cnt, tmr) = unsafe { (inw(b + 8), inw(b + 0x12) & 0x3FF) };
    let c0 = unsafe { inw(b) } & 0x3FF;
    delay_ms(800);
    let c1 = unsafe { inw(b) } & 0x3FF;
    let ok = gcs & 0x20 == 0 && cnt & 0x0800 == 0 && tmr >= 4 && c1 != c0;
    kprintln!(
        "wdt: verify NO_REBOOT={} HLT={} TMR={} count {}->{} => {}",
        (gcs >> 5) & 1, (cnt >> 11) & 1, tmr, c0, c1, if ok { "VERIFIED" } else { "NOT VERIFIED" }
    );
    ok
}

/// Attended `wdt=probe`: arm the NM10 TCO ourselves without touching locked bits (SMI_EN.TCO_EN stays as the
/// BIOS left it): clear GCS.NO_REBOOT (the BIOS leaves it SET on the NC110), clear stale status, TMR=50 ticks
/// (30 s per stage, reset ~60 s after the last pet), reload, un-halt. Returns true if the readback says armed.
pub fn arm_probe() -> bool {
    let lpc = pci::read32(pci::b0(31, 0), 0);
    let rcba_raw = pci::read32(pci::b0(31, 0), 0xF0);
    if !(lpc == 0x27BC_8086 || lpc == 0x2918_8086) || pmbase() == 0 || rcba_raw & 1 == 0 {
        kprintln!("wdt: probe arm: prerequisites fail (LPC {:#010x}, PMBASE {:#x}, RCBA {:#x})", lpc, pmbase(), rcba_raw);
        return false;
    }
    let b = base();
    let rcba = rcba_raw & 0xFFFF_C000;
    let gcs_p = (rcba + 0x3410) as *mut u32;
    unsafe {
        let g0 = read_volatile(gcs_p);
        let (cnt0, tmr0, rld0) = (inw(b + 8), inw(b + 0x12) & 0x3FF, inw(b) & 0x3FF);
        kprintln!("wdt: inherited: GCS {:#x} NO_REBOOT={} TCO1_CNT={:#06x} HLT={} TMR={} RLD={} ({})", g0, (g0 >> 5) & 1, cnt0,
            (cnt0 >> 11) & 1, tmr0, rld0, if g0 & 0x20 == 0 && cnt0 & 0x800 == 0 { "inherited enabled: NO_REBOOT clear, timer running" } else { "inherited not enabled" });
        core::ptr::write_volatile(gcs_p, g0 & !0x20);
        outw(b + 4, 0x0008); // TCO1_STS.TIMEOUT (RW1C)
        outw(b + 6, 0x0002); // TCO2_STS.SECOND_TO_STS
        outw(b + 6, 0x0004); // TCO2_STS.BOOT_STS
        outw(b + 0x12, (inw(b + 0x12) & !0x3FF) | 50);
        outw(b, 1);
        outw(b + 8, inw(b + 8) & !0x0800); // un-halt; bit 12 (lock) written back as read
        let (g1, cnt, tmr, c0) = (read_volatile(gcs_p), inw(b + 8), inw(b + 0x12) & 0x3FF, inw(b) & 0x3FF);
        delay_ms(1300);
        let c1 = inw(b) & 0x3FF;
        let ok = g1 & 0x20 == 0 && cnt & 0x0800 == 0 && tmr == 50 && c1 != c0;
        kprintln!("wdt: probe arm: GCS {:#x} -> {:#x} TCO1_CNT={:#06x} TMR={} RLD {}->{} SMI_EN={:#x} => {}",
            g0, g1, cnt, tmr, c0, c1, inl(pmbase() + 0x30), if ok { "ARMED" } else { "NOT ARMED" });
        ok
    }
}

pub fn pet() {
    unsafe { outw(base(), 1) };
}

/// ACPI PM timer (PMBASE+0x08, 3.579545 MHz, 24-bit) based delay; same block on ICH9 and NM10.
fn pm_timer() -> u32 {
    unsafe { inl(pmbase() + 8) & 0x00FF_FFFF }
}

pub fn delay_ms(ms: u32) {
    let need = ms as u64 * 3_579_545 / 1000;
    let mut last = pm_timer();
    let mut acc = 0u64;
    while acc < need {
        let now = pm_timer();
        acc += (now.wrapping_sub(last) & 0x00FF_FFFF) as u64;
        last = now;
    }
}

/// `wdt=pet:N` pets every 500 ms for N s, then stops and hangs. `wdt=hang` hangs at once.
pub fn rehearse(cmdline: &str) -> bool {
    let Some(i) = cmdline.find("wdt=") else { return false };
    let arg = cmdline[i + 4..].split(' ').next().unwrap_or("");
    if arg == "probe" {
        return false; // attended net, handled in main (k5 hung here before the NIC: result 5 counts as armed)
    }
    if !armed() {
        kprintln!("wdt: mode {} requested but watchdog not armed", arg);
        return false;
    }
    let secs: u32 = arg.strip_prefix("pet:").and_then(|n| n.parse().ok()).unwrap_or(0);
    for s in 0..secs {
        for _ in 0..2 {
            pet();
            delay_ms(500);
        }
        kprintln!("wdt: alive t={}s (petting every 500 ms)", s + 1);
    }
    kprintln!("wdt: HANG now: cli; hlt with the timer running, nobody pets");
    loop {
        unsafe { core::arch::asm!("cli; hlt") };
    }
}

/// ACPI PM timer (PMBASE+0x08, 3.579545 MHz, 24-bit) based delay; same block on ICH9 and NM10.
fn pm_timer() -> u32 {
    unsafe { inl(pmbase() + 8) & 0x00FF_FFFF }
}

pub fn delay_ms(ms: u32) {
    let need = ms as u64 * 3_579_545 / 1000;
    let mut last = pm_timer();
    let mut acc = 0u64;
    while acc < need {
        let now = pm_timer();
        acc += (now.wrapping_sub(last) & 0x00FF_FFFF) as u64;
        last = now;
    }
}

//! Multiboot v1 info: cmdline, memory map, and the client-key module (M2.5).
//!
//! The SSH client key is NOT compiled into the ELF. GRUB loads it with `module /baremetal/<file>` (QEMU:
//! `-initrd <file>`), so the key lives on /boot and can be rotated without a rebuild.
//! Module format (72 bytes): b"NC110K1\n" | ed25519 seed (32) | ed25519 public key (32).
//! The kernel derives the public key from the seed and refuses on mismatch (truncation / corruption / wrong
//! file), copies the seed into a static, then zeroes the module bytes in RAM.
use crate::kprintln;
use core::ptr::{read_volatile, write_volatile};

pub const MAGIC: &[u8; 8] = b"NC110K1\n";
const KEYMOD_LEN: u64 = 72;

static mut RAM: [(u64, u64); 32] = [(0, 0); 32];
static mut NRAM: usize = 0;
static mut HAVE_MAP: bool = false;

unsafe fn rd32(a: u64) -> u32 {
    read_volatile(a as *const u32)
}

/// Record available-RAM ranges (mmap type 1, or mem_lower/mem_upper as a fallback).
pub fn parse_memory(info: u64) {
    unsafe {
        if info == 0 {
            return;
        }
        let flags = rd32(info);
        if flags & (1 << 6) != 0 {
            let len = rd32(info + 44) as u64;
            let mut p = rd32(info + 48) as u64;
            let end = p + len;
            while p + 24 <= end && NRAM < 32 {
                let size = rd32(p) as u64;
                let base = read_volatile((p + 4) as *const u64);
                let l = read_volatile((p + 12) as *const u64);
                let ty = rd32(p + 20);
                if ty == 1 && l > 0 {
                    RAM[NRAM] = (base, l);
                    NRAM += 1;
                }
                p += size + 4;
            }
            HAVE_MAP = true;
        } else if flags & 1 != 0 {
            RAM[0] = (0, rd32(info + 4) as u64 * 1024);
            RAM[1] = (0x10_0000, rd32(info + 8) as u64 * 1024);
            NRAM = 2;
            HAVE_MAP = true;
        }
    }
}

/// True if [base, base+len) touches available RAM, or if no memory map was given (fail closed).
pub fn ram_overlaps(base: u64, len: u64) -> bool {
    unsafe {
        if !HAVE_MAP {
            return true;
        }
        let ram = &*core::ptr::addr_of!(RAM);
        ram[..NRAM].iter().any(|&(b, l)| base < b + l && b < base + len)
    }
}

/// True only if [base, base+len) lies entirely inside ONE available-RAM range of the multiboot map.
pub fn ram_contains(base: u64, len: u64) -> bool {
    let Some(end) = base.checked_add(len) else { return false };
    unsafe {
        let ram = &*core::ptr::addr_of!(RAM);
        HAVE_MAP && ram[..NRAM].iter().any(|&(b, l)| base >= b && end <= b.saturating_add(l))
    }
}

pub fn cmdline(info: u64) -> &'static str {
    unsafe {
        if info == 0 || rd32(info) & 4 == 0 {
            return "";
        }
        let p = rd32(info + 16) as *const u8;
        let mut n = 0;
        while n < 256 && *p.add(n) != 0 {
            n += 1;
        }
        core::str::from_utf8(core::slice::from_raw_parts(p, n)).unwrap_or("")
    }
}

//! Monotonic milliseconds from the TSC, calibrated once against PIT channel 2 (gate via port 0x61).
use crate::port::{inb, outb, rdtsc};

static mut TSC_PER_MS: u64 = 1_000_000; // fallback guess (1 GHz) if calibration fails
static mut T0: u64 = 0;

pub fn init() -> u64 {
    unsafe {
        // PIT ch2, mode 0, lobyte/hibyte, count for 10 ms at 1.193182 MHz.
        const COUNT: u16 = 11932;
        let g = inb(0x61);
        outb(0x61, (g & 0xFD) | 0x01); // gate on, speaker off
        outb(0x43, 0b1011_0000);
        outb(0x42, COUNT as u8);
        outb(0x42, (COUNT >> 8) as u8);
        // Re-trigger: toggle gate low->high so mode 0 starts counting now.
        let g = inb(0x61);
        outb(0x61, g & 0xFE);
        outb(0x61, g | 0x01);
        let a = rdtsc();
        let mut spins: u64 = 0;
        while inb(0x61) & 0x20 == 0 && spins < 200_000_000 {
            spins += 1;
        }
        let b = rdtsc();
        if spins < 200_000_000 && b > a {
            TSC_PER_MS = ((b - a) / 10).max(1);
        }
        T0 = rdtsc();
        TSC_PER_MS
    }
}

pub fn now_ms() -> i64 {
    unsafe { ((rdtsc() - T0) / TSC_PER_MS) as i64 }
}

/// Busy-wait at least `us` microseconds (TSC, after init()).
pub fn delay_us(us: u64) {
    unsafe {
        let t = rdtsc();
        let n = us * TSC_PER_MS / 1000 + 1;
        while rdtsc().wrapping_sub(t) < n {
            core::hint::spin_loop();
        }
    }
}
