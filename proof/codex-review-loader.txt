+//! In-RAM chain loader, driven over bmctl (the DS OTA equivalent: nothing is written to /boot).
+//! `lbeg <size> <sha256>` opens a staging buffer at STAGE; binary `ldat` ++ u32 offset ++ bytes appends in order;
+//! `lcom <cmdline>` verifies length + SHA-256 + the multiboot header (a.out kludge, flag bit 16, as GRUB applies it),
+//! checks every destination range, builds a multiboot info block (cmdline, the ORIGINAL GRUB memory map, and a
+//! rebuilt 72-byte key module with this boot's seed), quiesces the NIC and jumps through the trampoline in boot.s,
+//! which turns paging off, copies the image to load_addr, zeroes its bss and enters it like GRUB would.
+//! The TCO stays armed across the hop; the new kernel re-arms it as its first act. Any refusal leaves us running.
+use crate::kprintln;
+use crate::mb;
+use crate::nic::Rtl;
+use sha2::{Digest, Sha256};
+
+pub const STAGE: u64 = 0x0400_0000; // 64 MiB
+pub const CAP: u64 = 0x0080_0000; // 8 MiB (STAGE + CAP < 128 MiB, so q35 -m 128 works)
+const TRAMP: u64 = 0x8000; // trampoline code + its stack (< 0xE00)
+const PARAMS: u64 = 0x8E00;
+const INFO: u64 = 0x9000;
+const MODENT: u64 = 0x9080;
+const MODULE: u64 = 0x9100;
+const CMDLINE: u64 = 0x9200;
+const MMAP: u64 = 0x9400;
+const LOW_END: u64 = 0xA000;
+
+extern "C" {
+    static tramp_start: u8;
+    static tramp_end: u8;
+    static __load_start: u8;
+    static __bss_end: u8;
+}
+
+pub struct Stage {
+    size: u64,
+    sha: [u8; 32],
+    got: u64,
+}
+
+fn overlaps(a: u64, al: u64, b: u64, bl: u64) -> bool {
+    a < b.saturating_add(bl) && b < a.saturating_add(al)
+}
+
+fn kernel_image() -> (u64, u64) {
+    let (s, e) = (core::ptr::addr_of!(__load_start) as u64, core::ptr::addr_of!(__bss_end) as u64);
+    (s, e - s)
+}
+
+pub fn begin(size: u64, sha_hex: &str) -> Result<Stage, &'static str> {
+    if size == 0 || size > CAP {
+        return Err("size 1..8 MiB");
+    }
+    let (ks, kl) = kernel_image();
+    if !mb::ram_contains(STAGE, size) || overlaps(STAGE, size, ks, kl) {
+        return Err("staging area not free RAM");
+    }
+    if sha_hex.len() != 64 {
+        return Err("sha256: 64 hex chars");
+    }
+    let mut sha = [0u8; 32];
+    for (i, b) in sha.iter_mut().enumerate() {
+        *b = u8::from_str_radix(&sha_hex[2 * i..2 * i + 2], 16).map_err(|_| "sha256: hex")?;
+    }
+    Ok(Stage { size, sha, got: 0 })
+}
+
+pub fn data(st: &mut Stage, off: u64, bytes: &[u8]) -> Result<u64, &'static str> {
+    if off != st.got {
+        return Err("out of order (send chunks in sequence)");
+    }
+    let end = off.checked_add(bytes.len() as u64).ok_or("overflow")?;
+    if end > st.size {
+        return Err("past announced size");
+    }
+    unsafe { core::ptr::copy_nonoverlapping(bytes.as_ptr(), (STAGE + off) as *mut u8, bytes.len()) };
+    st.got = end;
+    Ok(end)
+}

+/// Everything checked, nothing moved yet. Returns (src, dst, len, bss_len, entry).
+fn plan(st: &Stage) -> Result<(u64, u64, u64, u64, u64), &'static str> {
+    if st.got != st.size {
+        return Err("incomplete image");
+    }
+    let img = unsafe { core::slice::from_raw_parts(STAGE as *const u8, st.size as usize) };
+    let d: [u8; 32] = Sha256::digest(img).into();
+    if d != st.sha {
+        return Err("sha256 mismatch");
+    }
+    let rd = |o: usize| u32::from_le_bytes(img[o..o + 4].try_into().unwrap());
+    let lim = img.len().min(8192);
+    let hoff = (0..lim.saturating_sub(31)).step_by(4).find(|&o| {
+        rd(o) == 0x1BAD_B002 && rd(o).wrapping_add(rd(o + 4)).wrapping_add(rd(o + 8)) == 0
+    }).ok_or("no multiboot header in the first 8 KiB")?;
+    let flags = rd(hoff + 4);
+    if flags & (1 << 16) == 0 {
+        return Err("multiboot header lacks the address kludge (flag 16)");
+    }
+    let (haddr, laddr, lend, bend, entry) =
+        (rd(hoff + 12) as u64, rd(hoff + 16) as u64, rd(hoff + 20) as u64, rd(hoff + 24) as u64, rd(hoff + 28) as u64);
+    if laddr > haddr || lend <= laddr || bend < lend || entry < laddr || entry >= lend {
+        return Err("inconsistent header addresses");
+    }
+    let foff = (hoff as u64).checked_sub(haddr - laddr).ok_or("header offset")?;
+    let len = lend - laddr;
+    if foff.checked_add(len).is_none_or(|e| e > st.size) {
+        return Err("image shorter than load_end - load_addr");
+    }
+    if laddr < 0x10_0000 || bend > STAGE || !mb::ram_contains(laddr, bend - laddr) {
+        return Err("target range not RAM in [1 MiB, staging)");
+    }
+    if overlaps(laddr, bend - laddr, TRAMP, LOW_END - TRAMP) {
+        return Err("target overlaps the trampoline/info pages");
+    }
+    Ok((STAGE + foff, laddr, len, bend - lend, entry))
+}

+pub struct Plan {
+    params: [u64; 6],
+}

+/// the low pages are outside the kernel image, and GRUB's info/mmap are copied out before they are overwritten.

+        core::ptr::copy_nonoverlapping(core::ptr::addr_of!(tramp_start), TRAMP as *mut u8, tl as usize);

+        if oflags & 1 != 0 {

+            w32(INFO + 8, mhi);
+        }

+            flags |= 1 << 6;
+            w32(INFO + 44, ml as u32);
+            w32(INFO + 48, MMAP as u32);
+        }

+        w32(INFO + 20, 1);

+    }

+    Ok(Plan { params: [src, dst, len, bss, entry, INFO] })
+}

+/// Point of no return: NIC quiesced, trampoline entered. The TCO stays armed across the hop.
+pub fn jump(p: &Plan, nic: &Rtl) -> ! {

+        for (i, v) in p.params.iter().enumerate() {

+        crate::wdt::pet();

+        f(PARAMS)
+    }

//! Multiboot v1 info: cmdline, memory map, and the client-key module (M2.5).
//!
//! The SSH client key is NOT compiled into the ELF. GRUB loads it with `module /baremetal/<file>` (QEMU:
//! `-initrd <file>`), so the key lives on /boot and can be rotated without a rebuild.
//! Module format (72 bytes): b"NC110K1\n" | ed25519 seed (32) | ed25519 public key (32).
//! The kernel derives the public key from the seed and refuses on mismatch (truncation / corruption / wrong
//! file), copies the seed into a static, then zeroes the module bytes in RAM.
use crate::kprintln;
use core::ptr::{read_volatile, write_volatile};

pub const MAGIC: &[u8; 8] = b"NC110K1\n";
const KEYMOD_LEN: u64 = 72;

static mut RAM: [(u64, u64); 32] = [(0, 0); 32];
static mut NRAM: usize = 0;
static mut HAVE_MAP: bool = false;

unsafe fn rd32(a: u64) -> u32 {
    read_volatile(a as *const u32)
}

/// Record available-RAM ranges (mmap type 1, or mem_lower/mem_upper as a fallback).
pub fn parse_memory(info: u64) {
    unsafe {
        if info == 0 {
            return;
        }
        let flags = rd32(info);
        if flags & (1 << 6) != 0 {
            let len = rd32(info + 44) as u64;
            let mut p = rd32(info + 48) as u64;
            let end = p + len;
            while p + 24 <= end && NRAM < 32 {
                let size = rd32(p) as u64;
                let base = read_volatile((p + 4) as *const u64);
                let l = read_volatile((p + 12) as *const u64);
                let ty = rd32(p + 20);
                if ty == 1 && l > 0 {
                    RAM[NRAM] = (base, l);
                    NRAM += 1;
                }
                p += size + 4;
            }
            HAVE_MAP = true;
        } else if flags & 1 != 0 {
            RAM[0] = (0, rd32(info + 4) as u64 * 1024);
            RAM[1] = (0x10_0000, rd32(info + 8) as u64 * 1024);
            NRAM = 2;
            HAVE_MAP = true;
        }
    }
}

/// True if [base, base+len) touches available RAM, or if no memory map was given (fail closed).
pub fn ram_overlaps(base: u64, len: u64) -> bool {
    unsafe {
        if !HAVE_MAP {
            return true;
        }
        let ram = &*core::ptr::addr_of!(RAM);
        ram[..NRAM].iter().any(|&(b, l)| base < b + l && b < base + len)
    }
}

/// True only if [base, base+len) lies entirely inside ONE available-RAM range of the multiboot map.
pub fn ram_contains(base: u64, len: u64) -> bool {
    let Some(end) = base.checked_add(len) else { return false };
    unsafe {
        let ram = &*core::ptr::addr_of!(RAM);
        HAVE_MAP && ram[..NRAM].iter().any(|&(b, l)| base >= b && end <= b.saturating_add(l))
    }
}

pub fn cmdline(info: u64) -> &'static str {
    unsafe {
        if info == 0 || rd32(info) & 4 == 0 {
            return "";
        }
        let p = rd32(info + 16) as *const u8;
        let mut n = 0;
        while n < 256 && *p.add(n) != 0 {
            n += 1;
        }
        core::str::from_utf8(core::slice::from_raw_parts(p, n)).unwrap_or("")
    }
}

/// Find the key module, validate it, copy the seed out, wipe the module. None (with a log line) if absent/bad.
pub fn take_client_key(info: u64) -> Option<[u8; 32]> {
    unsafe {
        if info == 0 || rd32(info) & (1 << 3) == 0 || rd32(info + 20) == 0 {
            kprintln!("keymod: no multiboot module");
            return None;
        }
        let count = rd32(info + 20).min(16) as u64;
        let mods = rd32(info + 24) as u64;
        for i in 0..count {
            let e = mods + 16 * i;
            let (start, end) = (rd32(e) as u64, rd32(e + 4) as u64);
            if end <= start || end - start != KEYMOD_LEN {
                continue;
            }
            let m = core::slice::from_raw_parts(start as *const u8, KEYMOD_LEN as usize);
            if &m[..8] != MAGIC {
                continue;
            }
            let mut seed = [0u8; 32];
            let mut publ = [0u8; 32];
            seed.copy_from_slice(&m[8..40]);
            publ.copy_from_slice(&m[40..72]);
            for j in 0..KEYMOD_LEN {
                write_volatile((start + j) as *mut u8, 0);
            }
            let derived = sunset::ed25519_dalek::SigningKey::from_bytes(&seed).verifying_key().to_bytes();
            if derived != publ {
                kprintln!("keymod: module {} public key does not match its seed -> ignored (wiped)", i);
                return None;
            }
            kprintln!("keymod: client key from module {} at {:#x} (72 B), public key verified, module wiped", i, start);
            return Some(seed);
        }
        kprintln!("keymod: {} module(s), none is a 72-byte NC110K1 key", count);
        None
    }
}
//! TCO watchdog (ICH9 in QEMU q35, NM10 on the NC110; both iTCO v2). boot.s arms it before any
//! Rust runs; this module reports what boot.s did, pets it, and runs the M0.5 rehearsal modes.

#[cfg(all(feature = "neg-control", feature = "grub-armed"))]
compile_error!("neg-control is QEMU-only and must never be combined with grub-armed (the hardware artifact)");

/// Negative control (`--features neg-control`, QEMU q35 only; boot.s never self-arms on NM10): boot.s SETS GCS.NO_REBOOT instead of clearing it
/// (QEMU's GCS powers up 0 with noreboot=false, so merely skipping the clear would prove nothing).
#[no_mangle]
static TCO_NEG_CONTROL: u8 = cfg!(feature = "neg-control") as u8;

/// `--features grub-armed`: boot.s does NOT arm; used to test the GRUB iorw/memrw arming sequence.
#[no_mangle]
static TCO_GRUB_ARMED: u8 = cfg!(feature = "grub-armed") as u8;

extern "C" {
    static tco_result: u32;
    static tco_gcs_pre: u32;
    static tco_gcs: u32;
    static tco_base: u32;
    static tco_sts1_pre: u16;
    static tco_sts2_pre: u16;
    static tco_cnt: u16;
}

fn rd<T: Copy>(p: *const T) -> T {
    unsafe { read_volatile(p) }
}

pub fn armed() -> bool {
    matches!(rd(addr_of!(tco_result)), 0 | 4 | 5)
}

pub fn pmbase() -> u16 {
    (pci::read32(pci::b0(31, 0), 0x40) & 0xFF80) as u16
}

pub fn base() -> u16 {
    pmbase() + 0x60
}

pub fn report() {
    let r = rd(addr_of!(tco_result));
    let what = match r {
        0 => "armed by boot.s",
        1 => "skipped (LPC 00:1f.0 is not ICH9 8086:2918 / NM10 8086:27bc)",
        2 => "NOT armed: PMBASE unset",
        3 => "NOT armed: RCBA disabled",
        4 => "left to GRUB (grub-armed build); live registers follow",
        5 => "NM10: boot.s does not arm (attended Rust arms it; else GRUB must); live registers follow",
        _ => "NOT armed: unknown",
    };
    kprintln!("wdt: {} (result={})", what, r);
    if r == 4 || r == 5 {
        let b = base();
        let gcs = unsafe { read_volatile(((pci::read32(pci::b0(31, 0), 0xF0) & 0xFFFF_C000) + 0x3410) as *const u32) };
        let (s1, s2, cnt, tmr) = unsafe { (inw(b + 4), inw(b + 6), inw(b + 8), inw(b + 0x12)) };
        kprintln!(
            "wdt: live TCOBASE={:#x} GCS={:#010x} NO_REBOOT={} TCO1_CNT={:#06x} HLT={} TCO1_STS={:#06x} TCO2_STS={:#06x} TCO_TMR={}",
            b, gcs, (gcs >> 5) & 1, cnt, (cnt >> 11) & 1, s1, s2, tmr & 0x3FF
        );
        return;
    }
    if r != 0 {
        return;
    }
    let (b, gcs, s1, s2, cnt) = {
        (rd(addr_of!(tco_base)) as u16, rd(addr_of!(tco_gcs)), rd(addr_of!(tco_sts1_pre)), rd(addr_of!(tco_sts2_pre)), rd(addr_of!(tco_cnt)))
    };
    kprintln!(
        "wdt: TCOBASE={:#x} GCS {:#010x} -> {:#010x} NO_REBOOT={} TCO1_CNT={:#06x} HLT={} LOCK={}",
        b, rd(addr_of!(tco_gcs_pre)), gcs, (gcs >> 5) & 1, cnt, (cnt >> 11) & 1, (cnt >> 12) & 1
    );
    kprintln!(
        "wdt: before-clear TCO1_STS={:#06x} TCO2_STS={:#06x} SECOND_TO_STS={} BOOT_STS={}",
        s1, s2, (s2 >> 1) & 1, (s2 >> 2) & 1
    );
    let tmr = unsafe { inw(b + 0x12) } & 0x3FF;
    kprintln!("wdt: TCO_TMR={} ticks x 0.6 s, reset expected after 2nd expiry (~{} ms)", tmr, tmr as u32 * 1200);
}

/// Live check of the reset path, whoever armed it: NO_REBOOT=0, HLT=0, TMR>=4, and the counter
/// (TCO_RLD reads back the current value on v2) decrements across ~1.3 ticks.
pub fn verify_live() -> bool {
    if !armed() {
        return false;
    }
    let b = base();
    let rcba = pci::read32(pci::b0(31, 0), 0xF0);
    if rcba & 1 == 0 {
        return false;
    }
    let gcs = unsafe { read_volatile(((rcba & 0xFFFF_C000) + 0x3410) as *const u32) };
    let (cnt, tmr) = unsafe { (inw(b + 8), inw(b + 0x12) & 0x3FF) };
    let c0 = unsafe { inw(b) } & 0x3FF;
    delay_ms(800);
    let c1 = unsafe { inw(b) } & 0x3FF;
    let ok = gcs & 0x20 == 0 && cnt & 0x0800 == 0 && tmr >= 4 && c1 != c0;
    kprintln!(
        "wdt: verify NO_REBOOT={} HLT={} TMR={} count {}->{} => {}",
        (gcs >> 5) & 1, (cnt >> 11) & 1, tmr, c0, c1, if ok { "VERIFIED" } else { "NOT VERIFIED" }
    );
    ok
}

/// QEMU `-machine pc` (PIIX3 8086:7000 at 00:01.0) has no TCO; the M0 NIC demo may run unprotected there.
pub fn is_qemu_pc() -> bool {
    pci::read32(pci::b0(1, 0), 0) == 0x7000_8086 && rd(addr_of!(tco_result)) == 1
}

/// The negative control must reach its hang unverified (that is the point); only on q35, never NM10.
pub fn neg_control_on_q35() -> bool {
    let on = cfg!(feature = "neg-control") && rd(addr_of!(tco_result)) == 0;
    if on {
        kprintln!("wdt: neg-control build on q35: proceeding unverified on purpose");
    }
    on
}

/// Hard reset via the reset control register (spec M2 `reboot`): back to the firmware / Slot A.
pub fn hard_reset() -> ! {
    kprintln!("wdt: hard reset via 0xCF9");
    unsafe {
        outb(0xCF9, 0x02);
        outb(0xCF9, 0x06);
    }
    loop {
        unsafe { core::arch::asm!("cli; hlt") };
    }
}

/// Attended `wdt=probe`: arm the NM10 TCO ourselves without touching locked bits (SMI_EN.TCO_EN stays as the
/// BIOS left it): clear GCS.NO_REBOOT (the BIOS leaves it SET on the NC110), clear stale status, TMR=50 ticks
/// (30 s per stage, reset ~60 s after the last pet), reload, un-halt. Returns true if the readback says armed.
pub fn arm_probe() -> bool {
    let lpc = pci::read32(pci::b0(31, 0), 0);
    let rcba_raw = pci::read32(pci::b0(31, 0), 0xF0);
    if !(lpc == 0x27BC_8086 || lpc == 0x2918_8086) || pmbase() == 0 || rcba_raw & 1 == 0 {
        kprintln!("wdt: probe arm: prerequisites fail (LPC {:#010x}, PMBASE {:#x}, RCBA {:#x})", lpc, pmbase(), rcba_raw);
        return false;
    }
    let b = base();
    let rcba = rcba_raw & 0xFFFF_C000;
    let gcs_p = (rcba + 0x3410) as *mut u32;
    unsafe {
        let g0 = read_volatile(gcs_p);
        let (cnt0, tmr0, rld0) = (inw(b + 8), inw(b + 0x12) & 0x3FF, inw(b) & 0x3FF);
        kprintln!("wdt: inherited: GCS {:#x} NO_REBOOT={} TCO1_CNT={:#06x} HLT={} TMR={} RLD={} ({})", g0, (g0 >> 5) & 1, cnt0,
            (cnt0 >> 11) & 1, tmr0, rld0, if g0 & 0x20 == 0 && cnt0 & 0x800 == 0 { "inherited enabled: NO_REBOOT clear, timer running" } else { "inherited not enabled" });
        core::ptr::write_volatile(gcs_p, g0 & !0x20);
        outw(b + 4, 0x0008); // TCO1_STS.TIMEOUT (RW1C)
        outw(b + 6, 0x0002); // TCO2_STS.SECOND_TO_STS
        outw(b + 6, 0x0004); // TCO2_STS.BOOT_STS
        outw(b + 0x12, (inw(b + 0x12) & !0x3FF) | 50);
        outw(b, 1);
        outw(b + 8, inw(b + 8) & !0x0800); // un-halt; bit 12 (lock) written back as read
        let (g1, cnt, tmr, c0) = (read_volatile(gcs_p), inw(b + 8), inw(b + 0x12) & 0x3FF, inw(b) & 0x3FF);
        delay_ms(1300);
        let c1 = inw(b) & 0x3FF;
        let ok = g1 & 0x20 == 0 && cnt & 0x0800 == 0 && tmr == 50 && c1 != c0;
        kprintln!("wdt: probe arm: GCS {:#x} -> {:#x} TCO1_CNT={:#06x} TMR={} RLD {}->{} SMI_EN={:#x} => {}",
            g0, g1, cnt, tmr, c0, c1, inl(pmbase() + 0x30), if ok { "ARMED" } else { "NOT ARMED" });
        ok
    }
}

pub fn pet() {
    unsafe { outw(base(), 1) };
}

/// ACPI PM timer (PMBASE+0x08, 3.579545 MHz, 24-bit) based delay; same block on ICH9 and NM10.
fn pm_timer() -> u32 {
    unsafe { inl(pmbase() + 8) & 0x00FF_FFFF }
}

pub fn delay_ms(ms: u32) {
    let need = ms as u64 * 3_579_545 / 1000;
    let mut last = pm_timer();
    let mut acc = 0u64;
    while acc < need {
        let now = pm_timer();
        acc += (now.wrapping_sub(last) & 0x00FF_FFFF) as u64;
        last = now;
    }
}

/// `wdt=pet:N` pets every 500 ms for N s, then stops and hangs. `wdt=hang` hangs at once.
pub fn rehearse(cmdline: &str) -> bool {
    let Some(i) = cmdline.find("wdt=") else { return false };
    let arg = cmdline[i + 4..].split(' ').next().unwrap_or("");
    if arg == "probe" {
        return false; // attended net, handled in main (k5 hung here before the NIC: result 5 counts as armed)
    }
    if !armed() {
        kprintln!("wdt: mode {} requested but watchdog not armed", arg);
        return false;
    }
    let secs: u32 = arg.strip_prefix("pet:").and_then(|n| n.parse().ok()).unwrap_or(0);
    for s in 0..secs {
        for _ in 0..2 {
            pet();
            delay_ms(500);
        }
        kprintln!("wdt: alive t={}s (petting every 500 ms)", s + 1);
    }
    kprintln!("wdt: HANG now: cli; hlt with the timer running, nobody pets");
    loop {
        unsafe { core::arch::asm!("cli; hlt") };
    }
}

"""bmctl client for the NC110 kernel's authenticated UDP control port (see src/bmctl.rs).

Usage: tools/.venv/bin/python tools/bmctl.py [--host 192.168.50.120] [--key test-keys/client.key] <command...>
  commands: ping | stats | log [offset] | logall | peek ... | reboot
            load <kernel.elf> [cmdline...]   stage over the network, verify, chain-load in RAM (nothing on /boot),
                                             then wait for the new kernel's bmctl (new boot id) and print its ping
The key file is the same 72-byte multiboot key module the kernel gets (magic 8 + ed25519 seed 32 + pub 32);
master = SHA-256("nc110-bmctl-v1" + seed). Handshake: send "who" + fresh random 16-byte Nc; the reply carries boot id,
session and check = SHA-256("nc110-bmctl-v1 who" + master + Nc + boot + session)[:16], verified here; traffic key =
SHA-256("nc110-bmctl-v1 session" + master + boot + session_u32_le + Nc); seq counts from 1 per session. Uses the `cryptography` package's ChaCha20Poly1305,
an implementation independent of the kernel's, so agreement is a real interop check.
"""
import argparse, hashlib, hmac, os, socket, struct, sys, time
from cryptography.hazmat.primitives.ciphers.aead import ChaCha20Poly1305

ap = argparse.ArgumentParser()
ap.add_argument("--host", default="192.168.50.120")
ap.add_argument("--port", type=int, default=5515)
ap.add_argument("--key", default="test-keys/client.key")
ap.add_argument("cmd", nargs="+")
a = ap.parse_args()
km = open(a.key, "rb").read()
assert len(km) == 72, "key module must be 72 bytes"
master = hashlib.sha256(b"nc110-bmctl-v1" + km[8:40]).digest()
s = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
s.settimeout(2.0)


def rpc(raw):
    for _ in range(3):  # a retry of the same frame gets the kernel's cached reply, nothing runs twice
        s.sendto(raw, (a.host, a.port))
        try:
            while True:
                d, src = s.recvfrom(2048)
                if src[0] == socket.gethostbyname(a.host):
                    return d
        except socket.timeout:
            continue
    sys.exit(f"no reply from {a.host}:{a.port}")


nc = os.urandom(16)
who = rpc(b"who" + nc).decode().split()
if len(who) != 7 or who[:2] != ["nc110", "boot"] or who[3] != "session" or who[5] != "check" or len(who[2]) != 32:
    sys.exit(f"bad who reply: {who}")
boot, sid = bytes.fromhex(who[2]), int(who[4])
want = hashlib.sha256(b"nc110-bmctl-v1 who" + master + nc + boot + struct.pack("<I", sid)).digest()[:16]
if not hmac.compare_digest(want, bytes.fromhex(who[6])):
    sys.exit("who reply failed its check (wrong key, or a forged/replayed reply)")
aead = ChaCha20Poly1305(hashlib.sha256(b"nc110-bmctl-v1 session" + master + boot + struct.pack("<I", sid) + nc).digest())
SEQ = [0]


def call(cmd):
    return call_raw(cmd.encode())


def call_raw(body):
    SEQ[0] += 1  # per-session counter from 1; a fresh session per run, so (key, nonce) never repeats
    seq = SEQ[0]
    ct = aead.encrypt(b"C2K\0" + struct.pack("<Q", seq), body, boot)
    r = rpc(struct.pack("<IQ", sid, seq) + ct)
    rseq = struct.unpack("<Q", r[:8])[0]
    if rseq != seq:
        sys.exit(f"reply seq {rseq} != {seq}")
    return aead.decrypt(b"K2C\0" + struct.pack("<Q", seq), r[8:], boot)


cmd = " ".join(a.cmd)
if a.cmd[0] == "load":
    img = open(a.cmd[1], "rb").read()
    cl = " ".join(a.cmd[2:])
    sha = os.environ.get("BMCTL_TEST_SHA") or hashlib.sha256(img).hexdigest()  # test hooks: wrong digest,
    if os.environ.get("BMCTL_TEST_SEND"):                                      # truncated send,
        img_send = img[: int(os.environ["BMCTL_TEST_SEND"])]
    else:
        img_send = img
    r = call(f"lbeg {len(img)} {sha}").decode()
    print(r)
    if not r.startswith("ok"):
        sys.exit(1)
    t0 = time.time()
    offs = list(range(0, len(img_send), 1000))
    if os.environ.get("BMCTL_TEST_SWAP") and len(offs) > 2:                    # out-of-order chunks
        offs[1], offs[2] = offs[2], offs[1]
    for off in offs:
        r = call_raw(b"ldat" + struct.pack("<I", off) + img_send[off:off + 1000]).decode()
        if not r.startswith("ok"):
            sys.exit(f"ldat at {off}: {r}")
    print(f"staged {len(img)} B in {time.time() - t0:.1f} s")
    r = call(f"lcom {cl}".strip()).decode()
    print(r)
    if not r.startswith("ok"):
        sys.exit(1)
    old = boot
    for _ in range(60):
        time.sleep(1)
        try:
            s.sendto(b"who" + os.urandom(16), (a.host, a.port))
            w = s.recvfrom(256)[0].split()
            if len(w) >= 3 and bytes.fromhex(w[2].decode()) != old:
                print(f"new kernel up: boot {w[2].decode()} after {time.time() - t0:.1f} s")
                sys.exit(0)
        except (socket.timeout, ValueError):
            pass
    sys.exit("no new kernel answered within 60 s")
if cmd == "logall":
    off, out = 0, b""
    while True:
        chunk = call(f"log {off}")
        if not chunk:
            break
        out += chunk
        off += len(chunk)
    sys.stdout.write(out.decode(errors="replace"))
else:
    print(call(cmd).decode(errors="replace"))

//! Realtek C+ descriptor-ring NIC driver, two chips behind a PCI device-id match (M2.5):
//!   10ec:8136 RTL8101E family (the NC110's wired NIC): MMIO only, r8169-style init. NOT RUNNABLE IN QEMU.
//!   10ec:8139 rtl8139 in C+ mode (QEMU stand-in): port I/O BAR0, or MMIO BAR1 with cmdline `nic=mmio`.
//! Both use the same 16-byte descriptors and ring registers; the per-chip deltas live in nic_regs.rs and are
//! checked against pinned Linux v6.12 sources on the host (tools/test-nic-regs.sh).
//!
//! Safety rules this file keeps (spec rule 4: no disk access, ever):
//! - The only PCI config write is to the matched NIC's command register (low 16 bits; status written as 0).
//! - MMIO base must be a 32-bit-reachable memory BAR, 256-byte aligned, outside multiboot RAM, and unable to
//!   overlap any mass-storage (class 01h, e.g. SATA/AHCI) BAR (span bounded by its alignment, read-only).
//!   Otherwise the NIC is refused.
//! - Bus mastering is OFF across the chip reset and ring programming (iPXE leaves the NIC DMAing), and only
//!   turned on after the rings point at our buffers.
use crate::nic_regs::{self as R, Regs};
use crate::pci::{self, Bdf};
use crate::port::{inb, inl, inw, outb, outl, outw, rdtsc};
use crate::{kprintln, time};
use core::ptr::{addr_of, addr_of_mut, read_volatile, write_volatile};
use core::sync::atomic::{fence, Ordering};

const NRX: usize = 16;
const BUFSZ: usize = 1536;

#[repr(C)]
#[derive(Clone, Copy)]
struct Desc {
    opts1: u32,
    opts2: u32,
    addr: u64,
}

#[repr(C, align(256))]
struct Ring<const N: usize>([Desc; N]);

#[repr(C, align(16))]
struct Bufs<const N: usize>([[u8; BUFSZ]; N]);

const Z: Desc = Desc { opts1: 0, opts2: 0, addr: 0 };
static mut TX_DESC: Ring<1> = Ring([Z; 1]);
static mut RX_DESC: Ring<NRX> = Ring([Z; NRX]);
static mut TX_BUF: Bufs<1> = Bufs([[0; BUFSZ]; 1]);
static mut RX_BUF: Bufs<NRX> = Bufs([[0; BUFSZ]; NRX]);

#[derive(Clone, Copy)]
pub enum Bus {
    Pio(u16),
    Mmio(u64),
}

impl Bus {
    unsafe fn r8(self, o: u16) -> u8 {
        match self {
            Bus::Pio(p) => inb(p + o),
            Bus::Mmio(m) => read_volatile((m + o as u64) as *const u8),
        }
    }
    unsafe fn r16(self, o: u16) -> u16 {
        match self {
            Bus::Pio(p) => inw(p + o),
            Bus::Mmio(m) => read_volatile((m + o as u64) as *const u16),
        }
    }
    unsafe fn r32(self, o: u16) -> u32 {
        match self {
            Bus::Pio(p) => inl(p + o),
            Bus::Mmio(m) => read_volatile((m + o as u64) as *const u32),
        }
    }
    unsafe fn w8(self, o: u16, v: u8) {
        match self {
            Bus::Pio(p) => outb(p + o, v),
            Bus::Mmio(m) => write_volatile((m + o as u64) as *mut u8, v),
        }
    }
    unsafe fn w16(self, o: u16, v: u16) {
        match self {
            Bus::Pio(p) => outw(p + o, v),
            Bus::Mmio(m) => write_volatile((m + o as u64) as *mut u16, v),
        }
    }
    unsafe fn w32(self, o: u16, v: u32) {
        match self {
            Bus::Pio(p) => outl(p + o, v),
            Bus::Mmio(m) => write_volatile((m + o as u64) as *mut u32, v),
        }
    }
}

impl core::fmt::Display for Bus {
    fn fmt(&self, f: &mut core::fmt::Formatter) -> core::fmt::Result {
        match self {
            Bus::Pio(p) => write!(f, "io={:#x}", p),
            Bus::Mmio(m) => write!(f, "mmio={:#x} (UC)", m),
        }
    }
}

pub struct Rtl {
    pub bus: Bus,
    pub regs: &'static Regs,
    pub bdf: Bdf,
    pub mac: [u8; 6],
    rx_next: usize,
    rx_cfg: u32,
    ver: u8, // r8169 RTL_GIGA_MAC_VER_xx (0 = rtl8139 C+)
}

const PCI_CMD_IO: u32 = 1 << 0;
const PCI_CMD_MEM: u32 = 1 << 1;
const PCI_CMD_BM: u32 = 1 << 2;

/// The one PCI config write: the NIC's command register. Upper half (status, RW1C) written as 0.
fn set_pci_cmd(b: Bdf, set: u32, clear: u32) {
    let cmd = pci::read32(b, 0x04) & 0xFFFF;
    pci::write32(b, 0x04, (cmd & !clear) | set);
}

extern "C" {
    static mut pd: [u64; 2048]; // boot.s: 4 x 512 2-MiB identity entries covering 0..4 GiB
}

#[repr(C, align(4096))]
struct Pt([u64; 512]);
static mut UC_PT: Pt = Pt([0; 512]);

/// Make ONLY the 4 KiB page holding the NIC registers uncached (PCD|PWT): split the 2 MiB identity PDE that
/// covers it into 512 identity 4 KiB entries with the PDE's own flags, so nothing else in that 2 MiB (on the
/// NM10 possibly the AHCI ABAR) changes attributes. Review fix (GLM-5.2 B1): previously the whole 2 MiB went UC.
fn map_uc_4k(base: u64) {
    unsafe {
        let pde = addr_of_mut!(pd[(base >> 21) as usize]);
        let old = read_volatile(pde);
        let region = base & !0x1F_FFFF;
        let flags = old & 0x1F & !(1 << 7); // P, RW, US, PWT, PCD from the PDE; PS dropped for a PTE
        let pt = &mut *addr_of_mut!(UC_PT);
        for (i, e) in pt.0.iter_mut().enumerate() {
            *e = (region + (i as u64) * 4096) | flags;
        }
        pt.0[((base >> 12) & 0x1FF) as usize] |= (1 << 4) | (1 << 3);
        fence(Ordering::SeqCst);
        write_volatile(pde, (addr_of!(UC_PT) as u64) | 0x3); // present | rw, points at UC_PT
        core::arch::asm!("mov {t}, cr3", "mov cr3, {t}", t = out(reg) _, options(nostack)); // flush incl. the 2 MiB entry
    }
}

/// Validate a memory BAR for use as the NIC register window, then map its 2 MiB page uncached (PCD|PWT).
fn mmio_window(b: Bdf) -> Result<u64, &'static str> {
    let (base, _is64) = pci::first_mem_bar(b).ok_or("no memory BAR")?;
    if base == 0 {
        return Err("memory BAR unassigned (0)");
    }
    if base & 0xFF != 0 || base + 0x100 > 0x1_0000_0000 {
        return Err("memory BAR not 256-byte aligned or not below 4 GiB");
    }
    if crate::ram_overlaps(base, 0x100) {
        return Err("memory BAR overlaps RAM in the multiboot memory map");
    }
    let mut st = [0u64; 16];
    let n = pci::storage_mem_bars(&mut st);
    for &x in &st[..n] {
        kprintln!("nic: mass-storage memory BAR {:#x} (max span {:#x}), NIC window {:#x}+0x100", x, 1u64 << x.trailing_zeros().min(32), base);
    }
    if !R::clear_of_storage(base, &st[..n]) {
        return Err("NIC register window could overlap a storage controller BAR");
    }
    map_uc_4k(base);
    Ok(base)
}

/// Subset of rtl8169_get_mac_version (r8169_main.c:2318-2325, first match wins): the 8101E/8102E/8103E
/// versions whose init this file implements, plus the 8105E (VER_29/30) found in the real NC110 (same generic path;
/// plus rtl_hw_start_8105e_*, see hw_start_8105e). Anything else (8106E, 8402, ...) is refused.
fn xid_version(xid: u32) -> Option<u8> {
    const T: [(u32, u32, u8); 10] = [
        (0x7cf, 0x409, 29), // RTL8105E (r8169_main.c:2316-2317); the real NC110 is XID 0x40a -> VER_30
        (0x7c8, 0x408, 30),
        (0x7cf, 0x349, 8),
        (0x7cf, 0x249, 8),
        (0x7cf, 0x348, 7),
        (0x7cf, 0x248, 7),
        (0x7cf, 0x240, 14), // RTL8401: listed only so it is NOT mistaken for VER_09 below
        (0x7c8, 0x348, 9),
        (0x7c8, 0x248, 9),
        (0x7c8, 0x340, 10),
    ];
    T.iter().find(|&&(m, v, _)| xid & m == v).map(|t| t.2).filter(|&v| v != 14)
}

pub fn probe(prefer_mmio: bool) -> Result<Rtl, &'static str> {
    if let Some(b) = pci::find(0x10EC, 0x8136) {
        return probe_8101e(b);
    }
    if let Some(b) = pci::find(0x10EC, 0x8139) {
        return probe_8139cp(b, prefer_mmio);
    }
    Err("no 10ec:8136 or 10ec:8139 found (bridges walked)")
}

fn probe_8139cp(b: Bdf, mmio: bool) -> Result<Rtl, &'static str> {
    let bus = if mmio {
        set_pci_cmd(b, 0, PCI_CMD_BM);
        let m = mmio_window(b)?;
        set_pci_cmd(b, PCI_CMD_MEM, PCI_CMD_BM);
        Bus::Mmio(m)
    } else {
        set_pci_cmd(b, PCI_CMD_IO, PCI_CMD_BM);
        let bar0 = pci::read32(b, 0x10);
        if bar0 & 1 == 0 {
            return Err("rtl8139 BAR0 is not I/O");
        }
        Bus::Pio((bar0 & !0x3) as u16)
    };
    let mut mac = [0u8; 6];
    for (i, m) in mac.iter_mut().enumerate() {
        *m = unsafe { bus.r8(R::RTL8139CP.mac0 + i as u16) };
    }
    Ok(Rtl { bus, regs: &R::RTL8139CP, bdf: b, mac, rx_next: 0, rx_cfg: 0x0000_E70A, ver: 0 })
}

fn probe_8101e(b: Bdf) -> Result<Rtl, &'static str> {
    set_pci_cmd(b, 0, PCI_CMD_BM | PCI_CMD_IO); // stop any DMA left running by iPXE before anything else
    let m = mmio_window(b)?;
    set_pci_cmd(b, PCI_CMD_MEM, PCI_CMD_BM | PCI_CMD_IO);
    let bus = Bus::Mmio(m);
    let r = &R::RTL8101E;
    let txc = unsafe { bus.r32(r.tx_config) };
    if txc == 0xFFFF_FFFF {
        return Err("TxConfig reads all-ones (PCI read failed, r8169_main.c:5419)");
    }
    let xid = (txc >> 20) & 0xFCF; // r8169_main.c:5422
    let ver = match xid_version(xid) {
        Some(v) => v,
        None => {
            kprintln!("nic: 10ec:8136 XID {:#05x} is not RTL8101E/8102E/8103E/8105E (VER_07..10, 29, 30): refusing", xid);
            return Err("unsupported 10ec:8136 chip version");
        }
    };
    kprintln!("nic: 10ec:8136 XID {:#05x} -> r8169 RTL_GIGA_MAC_VER_{:02}", xid, ver);
    // rtl_init_rxcfg (r8169_main.c:2486-2490 and default 2510): VER_10 FIFO_THRESH|DMA_BURST, VER_07..09 default.
    let rx_cfg = if ver == 10 { R::RX_FIFO_THRESH | R::RX_DMA_BURST } else { R::RX128_INT_EN | R::RX_DMA_BURST };
    let mut mac = [0u8; 6];
    for (i, x) in mac.iter_mut().enumerate() {
        *x = unsafe { bus.r8(r.mac0 + i as u16) };
    }
    Ok(Rtl { bus, regs: r, bdf: b, mac, rx_next: 0, rx_cfg: rx_cfg | R::ACCEPT_BROADCAST | R::ACCEPT_MY_PHYS, ver })
}

impl Rtl {
    pub fn name(&self) -> &'static str {
        self.regs.name
    }

    fn mdio_write(&self, reg: u32, val: u16) -> Result<(), &'static str> {
        // r8169_mdio_write, r8169_main.c:1154-1164: flag set, wait for it to clear (25 x 20 us), then 20 us.
        unsafe { self.bus.w32(self.regs.phyar, R::PHYAR_FLAG | (reg & 0x1F) << 16 | val as u32) };
        for _ in 0..25 {
            time::delay_us(20);
            if unsafe { self.bus.r32(self.regs.phyar) } & R::PHYAR_FLAG == 0 {
                time::delay_us(20);
                return Ok(());
            }
        }
        Err("PHYAR write timeout")
    }

    fn mdio_read(&self, reg: u32) -> Result<u16, &'static str> {
        // r8169_mdio_read, r8169_main.c:1166-1180: flag clear, wait for it to SET, data in 15:0, then 20 us.
        unsafe { self.bus.w32(self.regs.phyar, (reg & 0x1F) << 16) };
        for _ in 0..25 {
            time::delay_us(20);
            let v = unsafe { self.bus.r32(self.regs.phyar) };
            if v & R::PHYAR_FLAG != 0 {
                time::delay_us(20);
                return Ok(v as u16);
            }
        }
        Err("PHYAR read timeout")
    }

    pub fn skip_8105e(&mut self) {
        if self.ver == 29 || self.ver == 30 {
            kprintln!("nic: nic=no8105e: skipping rtl_hw_start_8105e (generic 810x path only)");
            self.ver += 100;
        }
    }

    /// Stop the chip and its DMA before control leaves this kernel (chain load): Rx/Tx off, bus mastering off.
    pub fn quiesce(&self) {
        unsafe { self.bus.w8(self.regs.chip_cmd, 0) };
        set_pci_cmd(self.bdf, 0, PCI_CMD_BM);
    }

    /// rtl_ephy_write / rtl_ephy_read (r8169_main.c:1261-1277): EPHYAR 0x80, flag bit 31, reg 20:16, data 15:0.

            if unsafe { self.bus.r32(0x80) } & 0x8000_0000 == 0 {
                time::delay_us(10);
                return Ok(());
            }
        }
        Err("EPHYAR write timeout")
    }

//! Attended build: broadcast the captured console log as UDP (192.168.50.118:5514 -> 255.255.255.255:5514).

const SRC_IP: [u8; 4] = [192, 168, 50, 118];
const PORT: u16 = 5514;

/// Copy of the NIC handles so the log can still be sent after ssh::run took the Rtl. Single-threaded and
/// tx_with is synchronous (waits for the descriptor), so interleaving with smoltcp between polls is safe.
static mut LOGNIC: Option<(Bus, &'static Regs, [u8; 6])> = None;

pub fn set_nic(n: &Rtl) {

}

fn send(data: &[u8]) {
    let Some((bus, regs, mac)) = (unsafe { *core::ptr::addr_of!(LOGNIC) }) else { return };
    let len = 42 + data.len();
    nic::tx_with(bus, regs, len, |f| {
        f[0..6].copy_from_slice(&[0xff; 6]);
        f[6..12].copy_from_slice(&mac);
        f[12..14].copy_from_slice(&[0x08, 0x00]);
        let ip = &mut f[14..34];
        ip.copy_from_slice(&[0x45, 0, 0, 0, 0, 0, 0x40, 0, 64, 17, 0, 0, 0, 0, 0, 0, 255, 255, 255, 255]);
        ip[2..4].copy_from_slice(&((20 + 8 + data.len()) as u16).to_be_bytes());
        ip[12..16].copy_from_slice(&SRC_IP);
        let c = dhcp::csum(ip);
        ip[10..12].copy_from_slice(&c.to_be_bytes());
        f[34..36].copy_from_slice(&PORT.to_be_bytes());
        f[36..38].copy_from_slice(&PORT.to_be_bytes());
        f[38..40].copy_from_slice(&((8 + data.len()) as u16).to_be_bytes());
        f[40..42].copy_from_slice(&[0, 0]); // UDP checksum 0 = none (IPv4)
        f[42..].copy_from_slice(data);
    });
}

/// Send the whole captured log, 1000-byte chunks, each prefixed "nc110 <round> <i>/<n> ".
pub fn dump(round: u8) {
    let log = console::captured();
    let total = log.len().div_ceil(1000).max(1);
    for (i, c) in log.chunks(1000).enumerate() {
        let mut b = [0u8; 1024];
        let h = [b'n', b'c', b'1', b'1', b'0', b' ', b'0' + round, b' ', b'0' + (i / 10) as u8, b'0' + (i % 10) as u8,
                 b'/', b'0' + (total / 10) as u8, b'0' + (total % 10) as u8, b' '];
        b[..h.len()].copy_from_slice(&h);
        b[h.len()..h.len() + c.len()].copy_from_slice(c);
        for _ in 0..2 { // chunks after the first went missing on the real LAN (k9, k11); the listener dedups
            send(&b[..h.len() + c.len()]);
            wdt::pet();
            wdt::delay_ms(50);
        }
    }
}

pub fn delay_ms(ms: u32) {
    let need = ms as u64 * 3_579_545 / 1000;
    let mut last = pm_timer();
    let mut acc = 0u64;
    while acc < need {
        let now = pm_timer();

        last = now;
    }
}

        let d = addr_of_mut!(TX_DESC.0[0]);
        if TX_WEDGED || read_volatile(addr_of!((*d).opts1)) & r.desc_own != 0 {
            TX_WEDGED = true;
            // Frame is built into scratch and dropped: the NIC may still be DMAing the real buffer.
            let scratch = &mut *core::ptr::addr_of_mut!(TX_SCRATCH);
            return fill(&mut scratch[..len]);
        }
        let buf = core::slice::from_raw_parts_mut(addr_of_mut!(TX_BUF.0[0]) as *mut u8, BUFSZ);
        let res = fill(&mut buf[..len]);
        let wire = len.max(60);
        for b in &mut buf[len..wire] {
            *b = 0;
        }
        write_volatile(
            d,
            Desc {
                opts1: r.desc_own | r.ring_end | r.first_frag | r.last_frag | wire as u32,
                opts2: 0,
                addr: buf.as_ptr() as u64,
            },
        );
        fence(Ordering::SeqCst);
        bus.w8(r.tx_poll, r.npq);
        let t0 = rdtsc();
        while read_volatile(addr_of!((*d).opts1)) & r.desc_own != 0 {
            if rdtsc() - t0 > 2_000_000_000 {
                TX_WEDGED = true;
                kprintln!("nic: TX descriptor not returned by NIC, TX disabled");
                break;
            }
            core::hint::spin_loop();
        }
        fence(Ordering::SeqCst);
        let st = bus.r16(r.intr_status);
        bus.w16(r.intr_status, st);
        res
    }
}

The trampoline’s normal execution path looks correct. I would fix DMA shutdown and commit-state handling before trusting this on the NC110. Reviewed `0addab9`; no edits or test reruns.

**Trampoline audit**

At [boot.s:224](~/nc110-baremetal/src/boot.s:224), I found no definite mode-transition bug:

- RIP-relative references survive relocation; the patched GDTR contains the relocated GDT address.
- `lretq` consumes the two eight-byte pushes and selects `0x18`, whose descriptor has `L=0, D=1`. The subsequent segment loads select flat, writable data.
- Paging is cleared from compatibility mode while code, GDT and stack remain identity-mapped. Clearing `EFER.LME` afterward is correct.
- Keeping `CR4.PAE` set is legal with paging disabled. Loading legacy CR3 and re-enabling paging are unnecessary here: Intel’s remaining steps describe returning to **paged** protected mode. [Intel SDM §9.8.5.4](https://www.intel.com/content/dam/www/public/us/en/documents/manuals/64-ia-32-architectures-software-developer-vol-3a-part-1-manual.pdf)
- `ECX` and `EDX` survive `rdmsr`; `ESI/EDI/EBP/EBX` remain intact. `cld`, flat DS/ES, and bounded, nonoverlapping source/destination ranges make both string operations correct.
- The private stack is outside the copy/zero ranges. Entry receives `EAX=0x2BADB002`, `EBX=INFO`, paging off and interrupts disabled.
- A20 is inherited from the working Multiboot boot; nothing here disables it.

**MUST-FIX**

1. **NIC shutdown does not establish DMA completion.**  
   [nic.rs:288](~/nc110-baremetal/src/nic.rs:288) writes `ChipCmd=0`, clears bus mastering and immediately returns. There is no chip-stop/reset completion handshake or demonstrated drain of outstanding transactions. The old RX descriptors and buffers are precisely the memory the trampoline may overwrite.

   Clearing BME prevents further initiation; it is not proof that previously issued writes have reached RAM. This is a hardware correctness gap, not a reproduced corruption. Linux’s Realtek teardown closes reception, performs version-dependent stop handling, then resets and waits before reclaiming buffers. Follow the relevant RTL8105E sequence, with bounded completion checks and a failure path that does not overwrite DMA-owned memory. [Linux r8169 teardown](https://kernel.googlesource.com/pub/scm/linux/kernel/git/torvalds/linux/+/333a396d71ce4083da31ccbf30772bf2da1509b4/drivers/net/ethernet/realtek/r8169_main.c)

   The q35 script uses **rtl8139**, so its success does not validate this RTL8105E behavior.

2. **A committed image remains mutable during the 300 ms delay.**  
   [bmctl.rs:379](~/nc110-baremetal/src/bmctl.rs:379) accepts another `lbeg` while `jump_at` remains armed. A subsequent `ldat` overwrites staging, but the pending plan still uses the earlier image’s validated addresses and entry.

   Concrete sequence: valid upload → successful `lcom` → new `lbeg` → overwrite the first chunk → old plan jumps into modified bytes without another hash check. Authentication does not prevent accidental overlapping clients. Freeze loader mutations after commit, or explicitly cancel the pending plan before permitting replacement.

**SHOULD-FIX**

3. **The client’s success check is unauthenticated.**  
   [bmctl.py:100](~/nc110-baremetal/tools/bmctl.py:100) accepts any UDP sender and any three-token response whose third token decodes to a different byte string. It checks neither the fresh nonce’s authenticator nor even boot-ID length. A forged response can report success while the target is dead.

   Reuse the authenticated handshake and perform the advertised encrypted `ping`. Also handle a lost `lcom` reply: the kernel can hop before the client’s two-second retry, leaving the client retrying an obsolete session and reporting failure.

4. **Malformed SHA text can panic instead of refusing.**  
   [loader.rs:59](~/nc110-baremetal/src/loader.rs:59) slices UTF-8 at arbitrary byte boundaries. A 64-byte argument beginning with `€` and followed by 61 ASCII characters panics on `[0..2]`. This halts until watchdog recovery. Validate ASCII hex or parse byte pairs.

5. **The accepted Multiboot contract exceeds what is implemented.**  
   [loader.rs:93](~/nc110-baremetal/src/loader.rs:93) ignores unsupported required flags. More concretely, this kernel requests page-aligned modules through header bit 0, but the rebuilt module starts at **`0x9100`**. Your own key parser tolerates that, so this does not explain a failed self-hop, but it violates the requested boot contract. Align the module and reject requirements you cannot fulfill. [Multiboot header requirements](https://www.gnu.org/software/grub/manual/multiboot/html_node/Header-magic-fields.html)

6. **Preserve boot metadata before staging, and refuse oversized maps.**  
   Copying original flags, memory sizes and mmap before zeroing low pages is correct. Original metadata inside the new image’s destination is also safe: it is copied before the trampoline overwrites that destination.

   However, [loader.rs:51](~/nc110-baremetal/src/loader.rs:51) excludes only the running kernel from staging. If GRUB placed info/mmap within the upload range, `ldat` destroys it before `prepare` reads it. Snapshot it into owned storage during initial boot, or reserve its ranges.

   Separately, mmap lengths above 3072 bytes are silently discarded. The successor can then fall back to coarse memory-size fields and lose reserved holes. Refuse rather than silently weakening the memory map.

**Remaining calibration**

`cli` does not block NMI, and there is no established NMI-safe IDT transition. Intel explicitly distinguishes NMI suppression from maskable-interrupt suppression. This is an inherited platform-wide weakness—the initial boot also lacks that protection—not evidence that this trampoline necessarily fails on the N570. Establish the machine’s NMI-source/masking policy before claiming robust mode transitions.

Reply ordering is reasonable on the healthy path: the next loop polls queued output, TX normally waits synchronously, and the final log dump precedes quiesce. It is not a delivery guarantee: enqueue failures are ignored and wedged TX silently drops output.

TCO continuity looks sound: the loader never disarms it, log dumping pets it, and `jump` pets immediately before transfer. **NIT:** the comment claiming the successor “re-arms it as its first act” is inaccurate on NM10; assembly leaves it alone, and the attended Rust path re-arms after entering `kmain`.

**6/10**
