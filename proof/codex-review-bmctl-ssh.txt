/// OpenSSH-style "SHA256:<base64, no padding>" of the ssh-ed25519 public key blob.
fn fingerprint(key: &[u8; 32], out: &mut [u8; 43]) {
    let mut h = Sha256::new();
    h.update(11u32.to_be_bytes());
    h.update(b"ssh-ed25519");
    h.update(32u32.to_be_bytes());
    h.update(key);
    let d = h.finalize();
    const A: &[u8; 64] = b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";
    let mut bits: u32 = 0;
    let mut n = 0;
    let mut o = 0;
    for &b in d.iter() {
        bits = (bits << 8) | b as u32;
        n += 8;
        while n >= 6 {
            n -= 6;
            out[o] = A[((bits >> n) & 63) as usize];
            o += 1;
        }
    }
    if n > 0 {
        out[o] = A[((bits << (6 - n)) & 63) as usize];
    }
}

    // Attended (unattended overnight) runs: an absolute bound, so a remote shell that hangs after its first
    // output cannot keep this loop petting the watchdog forever (Codex night review MUST-FIX 2).
    let hard_end = time::now_ms() + 180_000;

    'outer: while (got_output || time::now_ms() < deadline) && (!cfg!(feature = "attended") || ctl.is_some() || time::now_ms() < hard_end) {
        if wdt::armed() && time::now_ms() - last_pet >= 1000 {
            wdt::pet(); // the main loop is the liveness signal: a hang anywhere in it stops the petting
            last_pet = time::now_ms();
        }
        if ctl.is_some() {
            crate::net::rx_budget(Some(64)); // bmctl's deadlines run every pass even under a flood
        }
        iface.poll(now(), &mut dev, &mut sockets);
        if let Some(c) = ctl.as_mut() {
            if let Some(a) = c.step(sockets.get_mut::<udp::Socket>(udp_h), time::now_ms()) {
                if !c.act(a, &dev) {
                    bm_idle = true;
                    break 'outer;
                }
            }
        }
        let sock = sockets.get_mut::<tcp::Socket>(tcp_h);
        if !connected {
            if sock.may_send() {
                connected = true;
                kprintln!("tcp: connected to {}:{}", SERVER.0, SERVER.1);
            } else if !sock.is_open() {
                kprintln!("tcp: connection failed");
                break;
            }
        }

        // network -> runner
        if sock.can_recv() && runner.is_input_ready() {
            let r = sock.recv(|buf| {
                if !banner_shown {
                    for &b in buf.iter() {
                        if b == b'\n' {
                            banner_shown = true;
                            break;
                        }
                        if b != b'\r' && banner_len < banner.len() {
                            banner[banner_len] = b;
                            banner_len += 1;
                        }
                    }
                }
                match runner.input(buf) {
                    Ok(n) => (n, Ok(())),
                    Err(e) => (0, Err(e)),
                }
            });
            if banner_shown && banner_len > 0 {
                kprintln!("ssh(a): server banner: {}", core::str::from_utf8(&banner[..banner_len]).unwrap_or("?"));
                banner_len = 0;
            }
            if let Ok(Err(e)) = r {
                kprintln!("ssh: input error {:?}", e);
                break;
            }
        }
        // Peer sent FIN (CloseWait) or the socket died (reset/timeout): no more input once the buffer is drained.
        if connected && !sock.may_recv() && !sock.can_recv() {
            runner.close_input();
            if !sock.is_open() {
                kprintln!("\r\nssh: TCP connection gone ({:?}), ending session", sock.state());
                break 'outer;
            }
        }

        // runner events
        loop {
            let mut open_session = false;
            let mut stop_all = false;
            let mut idle = false;
            {
            let ev = match runner.progress() {
                Ok(ev) => ev,
                Err(e) => {
                    kprintln!("ssh: error {:?}", e);
                    break 'outer;
                }
            };
            match ev {
                Event::Cli(CliEvent::Hostkey(h)) => {
                    let pk = match h.hostkey() {
                        Ok(PubKey::Ed25519(k)) => Some(k.key.0),
                        _ => None,
                    };
                    match pk {
                        Some(k) if ct_eq(&k, &HOST_PIN) => {
                            let f = fp_str(&k);
                            kprintln!(
                                "ssh: host key ssh-ed25519 SHA256:{} matches pin",
                                core::str::from_utf8(&f).unwrap_or("?")
                            );
                            if let Err(e) = h.accept() {
                                kprintln!("ssh: host key accept failed (exchange signature?) {:?}", e);
                                stop_all = true;
                            }
                        }
                        Some(k) => {
                            let f = fp_str(&k);
                            let p = fp_str(&HOST_PIN);
                            kprintln!(
                                "ssh: hostkey MISMATCH: got SHA256:{} pinned SHA256:{} -> rejecting",
                                core::str::from_utf8(&f).unwrap_or("?"),
                                core::str::from_utf8(&p).unwrap_or("?")
                            );
                            let _ = h.reject();
                            stop_all = true;
                        }
                        None => {
                            kprintln!("ssh: hostkey MISMATCH: not ssh-ed25519 -> rejecting");
                            let _ = h.reject();
                            stop_all = true;
                        }
                    }

                    let _ = p.skip();

                Event::Cli(CliEvent::Pubkey(p)) => {

                        pubkey_tries += 1;
                        kprintln!("ssh: offering publickey ssh-ed25519 (throwaway test key)");
                        let _ = p.pubkey(signkey.clone());
                    } else {
                        let _ = p.skip();
                    }
                }

                    let _ = s.skip();

                    open_session = true;

                    let pty = Pty { term: tn, cols: 80, rows: 25, width: 0, height: 0, modes: heapless::Vec::new() };
                    let rp = o.pty(pty);
                    let rs = o.shell();
                    kprintln!("ssh: session channel open, pty-req xterm 80x25 ({:?}), shell ({:?})", rp.is_ok(), rs.is_ok());

                        stop_all = true;
                    } else {
                        shell_open = true;
                        winch = true;
                        kbd::init();
                        console::vga_log_off();
                        term::init();
                        kprintln!("term: VGA now belongs to the remote shell; keyboard is live");
                    }
                }
                Event::Cli(CliEvent::SessionExit(x)) => {
                    kprintln!("ssh: remote exit {:?}", x);
                }
                Event::Cli(CliEvent::Defunct) => {
                    kprintln!("ssh: connection closed");
                    finished = true;
                    idle = true;
                }
                Event::Cli(CliEvent::PollAgain) | Event::Progressed => {}
                Event::None => idle = true,
                _ => {}
            }
            }
            if stop_all {
                break 'outer;
            }
            if idle && !open_session {
                break;
            }
            if open_session {
                match runner.open_client_session() {
                    Ok(c) => chan = Some(c),
                    Err(e) => {
                        kprintln!("ssh: open session failed {:?}", e);
                        break 'outer;
                    }
                }
            }
        }

        // channel data -> VGA terminal (+ serial mirror)
        if let Some(c) = &chan {
            if winch {
                winch = false;
                let w = sunset::packets::WinChange { cols: 80, rows: 25, width: 0, height: 0 };
                kprintln!("ssh: window-change 80x25 ({:?})", runner.term_window_change(c, &w).is_ok());
            }
            let mut drew = false;
            while let Some((_num, dt, _len)) = runner.read_channel_ready() {
                let mut buf = [0u8; 512];
                match runner.read_channel(c, dt, &mut buf) {
                    Ok(0) => break,
                    Ok(n) => {
                        if !got_output {
                            kprintln!("ssh(d): first shell output ({} bytes)", n);
                            got_output = true;
                            out_t0 = time::now_ms();

                        out_bytes += n as u64;
                        for &b in &buf[..n] {
                            term::feed(b);
                            if b == 0x1B {
                                console::serial_raw(b'^');
                                console::serial_raw(b'[');
                            } else {
                                console::serial_raw(b);
                            }
                        }
                        drew = true;
                    }
                    Err(_) => break,
                }
            }
            if drew {
                term::flush();
            }
            // terminal replies (DSR/DA) go out ahead of keys typed later; they share the keyboard queue
            nkeys += term::take_reply(&mut keys[nkeys..]);

            // keyboard -> channel (whatever does not fit now stays queued; the i8042 holds the rest)
            if shell_open && !finished {
                if kbd::poll(&mut keys, &mut nkeys) {
                    kprintln!("\r\nkbd: Ctrl-Alt-Del -> hard reset");
                    wdt::hard_reset();
                }
                if nkeys > 0 {
                    match runner.write_channel(c, ChanData::Normal, &keys[..nkeys]) {
                        Ok(n) => {
                            keys.copy_within(n..nkeys, 0);
                            nkeys -= n;
                        }
                        Err(_) => nkeys = 0, // channel gone; the close check below ends the session
                    }
                }
            }

            if runner.is_channel_eof(c) || runner.is_channel_closed(c) {
                if !finished {
                    finished = true;
                    kprintln!("\r\nssh: shell channel closed, disconnecting ({} shell bytes in {} ms)", out_bytes, time::now_ms() - out_t0);
                    runner.close_output();
                }
            }
        }

        // runner -> network
        let sock = sockets.get_mut::<tcp::Socket>(tcp_h);
        while sock.can_send() {
            let out = runner.output_buf();
            if out.is_empty() {
                break;
            }
            match sock.send_slice(out) {
                Ok(0) | Err(_) => break,
                Ok(n) => runner.consume_output(n),
            }
        }
        if finished {
            sock.close();
            for _ in 0..50 {
                iface.poll(now(), &mut dev, &mut sockets);
            }
            break;
        }
    }
    kprintln!("ssh: session loop ended: {} shell bytes in {} ms, finished={}", out_bytes, if got_output { time::now_ms() - out_t0 } else { 0 }, finished);
    let (calls, reseeds, samples, failed) = rng::stats();
    kprintln!("kbd: {} keys dropped while the channel was not taking input", kbd::dropped());
    let rc = term::reply_counts();
    kprintln!("term: replies sent: DSR 6n x{}, DSR 5n x{}, DA1 x{}, DA2 x{}; {} dropped (reply buffer full)", rc[0], rc[1], rc[2], rc[3], rc[4]);
    kprintln!("rng: {} requests served, {} reseeds, {} raw samples health-tested, failure latched: {}", calls, reseeds, samples, failed);
    if got_output {
        kprintln!("ssh: M3 reached: interactive shell over SSH, drawn on VGA, typed on PS/2");
    } else {
        kprintln!("ssh: stopped before (d)");
    }
    if let (Some(c), false) = (ctl.as_mut(), bm_idle) {
        sockets.get_mut::<tcp::Socket>(tcp_h).abort();
        console::vga_log_on();
        kprintln!("ssh: session over; bmctl keeps serving (reboot / load to start another)");
        bmctl::serve(c, &mut iface, &mut dev, &mut sockets, udp_h);
    }
}

/// (QEMU's GCS powers up 0 with noreboot=false, so merely skipping the clear would prove nothing).

extern "C" {
    static tco_result: u32;
    static tco_gcs_pre: u32;
    static tco_gcs: u32;
    static tco_base: u32;
    static tco_sts1_pre: u16;
    static tco_sts2_pre: u16;
    static tco_cnt: u16;
}

fn rd<T: Copy>(p: *const T) -> T {
    unsafe { read_volatile(p) }
}

pub fn armed() -> bool {
    matches!(rd(addr_of!(tco_result)), 0 | 4 | 5)
}

pub fn pmbase() -> u16 {
    (pci::read32(pci::b0(31, 0), 0x40) & 0xFF80) as u16
}

pub fn base() -> u16 {
    pmbase() + 0x60
}

pub fn report() {
    let r = rd(addr_of!(tco_result));
    let what = match r {
        0 => "armed by boot.s",
        1 => "skipped (LPC 00:1f.0 is not ICH9 8086:2918 / NM10 8086:27bc)",
        2 => "NOT armed: PMBASE unset",
        3 => "NOT armed: RCBA disabled",
        4 => "left to GRUB (grub-armed build); live registers follow",
        5 => "NM10: boot.s does not arm (attended Rust arms it; else GRUB must); live registers follow",
        _ => "NOT armed: unknown",
    };
    kprintln!("wdt: {} (result={})", what, r);
    if r == 4 || r == 5 {
        let b = base();
        let gcs = unsafe { read_volatile(((pci::read32(pci::b0(31, 0), 0xF0) & 0xFFFF_C000) + 0x3410) as *const u32) };
        let (s1, s2, cnt, tmr) = unsafe { (inw(b + 4), inw(b + 6), inw(b + 8), inw(b + 0x12)) };
        kprintln!(
            "wdt: live TCOBASE={:#x} GCS={:#010x} NO_REBOOT={} TCO1_CNT={:#06x} HLT={} TCO1_STS={:#06x} TCO2_STS={:#06x} TCO_TMR={}",
            b, gcs, (gcs >> 5) & 1, cnt, (cnt >> 11) & 1, s1, s2, tmr & 0x3FF
        );
        return;
    }
    if r != 0 {
        return;
    }
    let (b, gcs, s1, s2, cnt) = {
        (rd(addr_of!(tco_base)) as u16, rd(addr_of!(tco_gcs)), rd(addr_of!(tco_sts1_pre)), rd(addr_of!(tco_sts2_pre)), rd(addr_of!(tco_cnt)))
    };
    kprintln!(
        "wdt: TCOBASE={:#x} GCS {:#010x} -> {:#010x} NO_REBOOT={} TCO1_CNT={:#06x} HLT={} LOCK={}",
        b, rd(addr_of!(tco_gcs_pre)), gcs, (gcs >> 5) & 1, cnt, (cnt >> 11) & 1, (cnt >> 12) & 1
    );
    kprintln!(
        "wdt: before-clear TCO1_STS={:#06x} TCO2_STS={:#06x} SECOND_TO_STS={} BOOT_STS={}",
        s1, s2, (s2 >> 1) & 1, (s2 >> 2) & 1
    );
    let tmr = unsafe { inw(b + 0x12) } & 0x3FF;
    kprintln!("wdt: TCO_TMR={} ticks x 0.6 s, reset expected after 2nd expiry (~{} ms)", tmr, tmr as u32 * 1200);

/// Live check of the reset path, whoever armed it: NO_REBOOT=0, HLT=0, TMR>=4, and the counter
/// (TCO_RLD reads back the current value on v2) decrements across ~1.3 ticks.
pub fn verify_live() -> bool {
    if !armed() {
        return false;
    }
    let b = base();
    let rcba = pci::read32(pci::b0(31, 0), 0xF0);
    if rcba & 1 == 0 {
        return false;
    }
    let gcs = unsafe { read_volatile(((rcba & 0xFFFF_C000) + 0x3410) as *const u32) };
    let (cnt, tmr) = unsafe { (inw(b + 8), inw(b + 0x12) & 0x3FF) };
    let c0 = unsafe { inw(b) } & 0x3FF;
    delay_ms(800);
    let c1 = unsafe { inw(b) } & 0x3FF;
    let ok = gcs & 0x20 == 0 && cnt & 0x0800 == 0 && tmr >= 4 && c1 != c0;
    kprintln!(
        "wdt: verify NO_REBOOT={} HLT={} TMR={} count {}->{} => {}",
        (gcs >> 5) & 1, (cnt >> 11) & 1, tmr, c0, c1, if ok { "VERIFIED" } else { "NOT VERIFIED" }
    );
    ok
}

/// Attended `wdt=probe`: arm the NM10 TCO ourselves without touching locked bits (SMI_EN.TCO_EN stays as the
/// BIOS left it): clear GCS.NO_REBOOT (the BIOS leaves it SET on the NC110), clear stale status, TMR=50 ticks
/// (30 s per stage, reset ~60 s after the last pet), reload, un-halt. Returns true if the readback says armed.
pub fn arm_probe() -> bool {
    let lpc = pci::read32(pci::b0(31, 0), 0);
    let rcba_raw = pci::read32(pci::b0(31, 0), 0xF0);
    if !(lpc == 0x27BC_8086 || lpc == 0x2918_8086) || pmbase() == 0 || rcba_raw & 1 == 0 {
        kprintln!("wdt: probe arm: prerequisites fail (LPC {:#010x}, PMBASE {:#x}, RCBA {:#x})", lpc, pmbase(), rcba_raw);
        return false;
    }
    let b = base();
    let rcba = rcba_raw & 0xFFFF_C000;
    let gcs_p = (rcba + 0x3410) as *mut u32;
    unsafe {
        let g0 = read_volatile(gcs_p);
        let (cnt0, tmr0, rld0) = (inw(b + 8), inw(b + 0x12) & 0x3FF, inw(b) & 0x3FF);
        kprintln!("wdt: inherited: GCS {:#x} NO_REBOOT={} TCO1_CNT={:#06x} HLT={} TMR={} RLD={} ({})", g0, (g0 >> 5) & 1, cnt0,
            (cnt0 >> 11) & 1, tmr0, rld0, if g0 & 0x20 == 0 && cnt0 & 0x800 == 0 { "inherited enabled: NO_REBOOT clear, timer running" } else { "inherited not enabled" });
        core::ptr::write_volatile(gcs_p, g0 & !0x20);
        outw(b + 4, 0x0008); // TCO1_STS.TIMEOUT (RW1C)
        outw(b + 6, 0x0002); // TCO2_STS.SECOND_TO_STS
        outw(b + 6, 0x0004); // TCO2_STS.BOOT_STS
        outw(b + 0x12, (inw(b + 0x12) & !0x3FF) | 50);
        outw(b, 1);
        outw(b + 8, inw(b + 8) & !0x0800); // un-halt; bit 12 (lock) written back as read
        let (g1, cnt, tmr, c0) = (read_volatile(gcs_p), inw(b + 8), inw(b + 0x12) & 0x3FF, inw(b) & 0x3FF);
        delay_ms(1300);
        let c1 = inw(b) & 0x3FF;
        let ok = g1 & 0x20 == 0 && cnt & 0x0800 == 0 && tmr == 50 && c1 != c0;
        kprintln!("wdt: probe arm: GCS {:#x} -> {:#x} TCO1_CNT={:#06x} TMR={} RLD {}->{} SMI_EN={:#x} => {}",
            g0, g1, cnt, tmr, c0, c1, inl(pmbase() + 0x30), if ok { "ARMED" } else { "NOT ARMED" });
        ok
    }
}

pub fn pet() {
    unsafe { outw(base(), 1) };
}

/// ACPI PM timer (PMBASE+0x08, 3.579545 MHz, 24-bit) based delay; same block on ICH9 and NM10.
fn pm_timer() -> u32 {
    unsafe { inl(pmbase() + 8) & 0x00FF_FFFF }
}

pub fn delay_ms(ms: u32) {
    let need = ms as u64 * 3_579_545 / 1000;
    let mut last = pm_timer();
    let mut acc = 0u64;
    while acc < need {
        let now = pm_timer();
        acc += (now.wrapping_sub(last) & 0x00FF_FFFF) as u64;
        last = now;
    }
}

/// `wdt=pet:N` pets every 500 ms for N s, then stops and hangs. `wdt=hang` hangs at once.
pub fn rehearse(cmdline: &str) -> bool {
    let Some(i) = cmdline.find("wdt=") else { return false };
    let arg = cmdline[i + 4..].split(' ').next().unwrap_or("");
    if arg == "probe" {
        return false; // attended net, handled in main (k5 hung here before the NIC: result 5 counts as armed)
    }
    if !armed() {
        kprintln!("wdt: mode {} requested but watchdog not armed", arg);
        return false;
    }
    let secs: u32 = arg.strip_prefix("pet:").and_then(|n| n.parse().ok()).unwrap_or(0);
    for s in 0..secs {
        for _ in 0..2 {
            pet();
            delay_ms(500);
        }
        kprintln!("wdt: alive t={}s (petting every 500 ms)", s + 1);
    }
    kprintln!("wdt: HANG now: cli; hlt with the timer running, nobody pets");
    loop {
        unsafe { core::arch::asm!("cli; hlt") };
    }
}
//! Monotonic milliseconds from the TSC, calibrated once against PIT channel 2 (gate via port 0x61).
use crate::port::{inb, outb, rdtsc};

static mut TSC_PER_MS: u64 = 1_000_000; // fallback guess (1 GHz) if calibration fails
static mut T0: u64 = 0;

pub fn init() -> u64 {
    unsafe {
        // PIT ch2, mode 0, lobyte/hibyte, count for 10 ms at 1.193182 MHz.
        const COUNT: u16 = 11932;
        let g = inb(0x61);
        outb(0x61, (g & 0xFD) | 0x01); // gate on, speaker off
        outb(0x43, 0b1011_0000);
        outb(0x42, COUNT as u8);
        outb(0x42, (COUNT >> 8) as u8);
        // Re-trigger: toggle gate low->high so mode 0 starts counting now.
        let g = inb(0x61);
        outb(0x61, g & 0xFE);
        outb(0x61, g | 0x01);
        let a = rdtsc();
        let mut spins: u64 = 0;
        while inb(0x61) & 0x20 == 0 && spins < 200_000_000 {
            spins += 1;
        }
        let b = rdtsc();
        if spins < 200_000_000 && b > a {
            TSC_PER_MS = ((b - a) / 10).max(1);
        }
        T0 = rdtsc();
        TSC_PER_MS
    }
}

pub fn now_ms() -> i64 {
    unsafe { ((rdtsc() - T0) / TSC_PER_MS) as i64 }
}

/// Busy-wait at least `us` microseconds (TSC, after init()).
pub fn delay_us(us: u64) {
    unsafe {
        let t = rdtsc();
        let n = us * TSC_PER_MS / 1000 + 1;
        while rdtsc().wrapping_sub(t) < n {
            core::hint::spin_loop();
        }
    }
}

/// Frames `receive` may hand out before the next `rx_budget` call. None = unlimited (SSH never sets it);
/// bmctl sets Some(32) before every poll so a flood cannot starve its deadline checks.
static mut RX_BUDGET: Option<u32> = None;

pub fn rx_budget(n: Option<u32>) {
    unsafe { RX_BUDGET = n };
}

pub struct Rx {
    idx: usize,
    len: usize,
    regs: &'static Regs,
}
pub struct Tx {
    bus: Bus,
    regs: &'static Regs,
}

impl Drop for Rx {
    fn drop(&mut self) {
        nic::rx_release(self.regs, self.idx);
    }
}

impl RxToken for Rx {
    fn consume<R, F: FnOnce(&[u8]) -> R>(self, f: F) -> R {
        f(nic::rx_buf(self.idx, self.len)) // descriptor goes back to the NIC when `self` drops
    }
}

impl TxToken for Tx {
    fn consume<R, F: FnOnce(&mut [u8]) -> R>(self, len: usize, f: F) -> R {
        nic::tx_with(self.bus, self.regs, len, f)
    }
}

impl Device for Nic {
    type RxToken<'a> = Rx;
    type TxToken<'a> = Tx;

    fn receive(&mut self, _t: Instant) -> Option<(Rx, Tx)> {
        unsafe {
            match RX_BUDGET {
                Some(0) => return None,
                Some(n) => RX_BUDGET = Some(n - 1),
                None => {}
            }
        }
        let (idx, len) = self.0.rx_poll()?;
        Some((Rx { idx, len, regs: self.0.regs }, Tx { bus: self.0.bus, regs: self.0.regs }))
    }

    fn transmit(&mut self, _t: Instant) -> Option<Tx> {
        Some(Tx { bus: self.0.bus, regs: self.0.regs })
    }

    fn capabilities(&self) -> DeviceCapabilities {
        let mut c = DeviceCapabilities::default();
        c.medium = Medium::Ethernet;
        c.max_transmission_unit = 1514;
        c.max_burst_size = Some(1);
        c
    }
}

//! Attended build: broadcast the captured console log as UDP (192.168.50.118:5514 -> 255.255.255.255:5514).
//! Broadcast needs no ARP and no DHCP lease. Static source IP = the netbook's Slot A address.
use crate::nic::{self, Bus, Rtl};
use crate::nic_regs::Regs;
use crate::{console, dhcp, kprintln, wdt};

const SRC_IP: [u8; 4] = [192, 168, 50, 118];
const PORT: u16 = 5514;

/// Copy of the NIC handles so the log can still be sent after ssh::run took the Rtl. Single-threaded and
/// tx_with is synchronous (waits for the descriptor), so interleaving with smoltcp between polls is safe.
static mut LOGNIC: Option<(Bus, &'static Regs, [u8; 6])> = None;

pub fn set_nic(n: &Rtl) {
    unsafe { LOGNIC = Some((n.bus, n.regs, n.mac)) };
}

fn send(data: &[u8]) {
    let Some((bus, regs, mac)) = (unsafe { *core::ptr::addr_of!(LOGNIC) }) else { return };
    let len = 42 + data.len();
    nic::tx_with(bus, regs, len, |f| {
        f[0..6].copy_from_slice(&[0xff; 6]);
        f[6..12].copy_from_slice(&mac);
        f[12..14].copy_from_slice(&[0x08, 0x00]);
        let ip = &mut f[14..34];
        ip.copy_from_slice(&[0x45, 0, 0, 0, 0, 0, 0x40, 0, 64, 17, 0, 0, 0, 0, 0, 0, 255, 255, 255, 255]);
        ip[2..4].copy_from_slice(&((20 + 8 + data.len()) as u16).to_be_bytes());
        ip[12..16].copy_from_slice(&SRC_IP);
        let c = dhcp::csum(ip);
        ip[10..12].copy_from_slice(&c.to_be_bytes());
        f[34..36].copy_from_slice(&PORT.to_be_bytes());
        f[36..38].copy_from_slice(&PORT.to_be_bytes());
        f[38..40].copy_from_slice(&((8 + data.len()) as u16).to_be_bytes());
        f[40..42].copy_from_slice(&[0, 0]); // UDP checksum 0 = none (IPv4)
        f[42..].copy_from_slice(data);
    });
}

/// Send the whole captured log, 1000-byte chunks, each prefixed "nc110 <round> <i>/<n> ".
pub fn dump(round: u8) {
    let log = console::captured();
    let total = log.len().div_ceil(1000).max(1);
    for (i, c) in log.chunks(1000).enumerate() {
        let mut b = [0u8; 1024];
        let h = [b'n', b'c', b'1', b'1', b'0', b' ', b'0' + round, b' ', b'0' + (i / 10) as u8, b'0' + (i % 10) as u8,
                 b'/', b'0' + (total / 10) as u8, b'0' + (total % 10) as u8, b' '];
        b[..h.len()].copy_from_slice(&h);
        b[h.len()..h.len() + c.len()].copy_from_slice(c);
        for _ in 0..2 { // chunks after the first went missing on the real LAN (k9, k11); the listener dedups
            send(&b[..h.len() + c.len()]);
            wdt::pet();
            wdt::delay_ms(50);
        }
    }
}

/// One raw DHCPDISCOVER, then poll RX for 4 s for an OFFER (proves RX DMA on the real NIC).
pub fn dhcp_probe(n: &mut Rtl) {
    let xid = 0x4e433131; // "NC11"
    nic::tx_with(n.bus, n.regs, 342, |f| dhcp::build_discover(f, n.mac, xid));
    kprintln!("dhcp: DISCOVER sent, waiting 4 s for an OFFER");
    let (mut frames, mut t) = (0u32, 0u32);
    while t < 4000 {
        while let Some((i, len)) = n.rx_poll() {
            frames += 1;
            let off = dhcp::parse_offer(nic::rx_buf(i, len), xid);
            nic::rx_release(n.regs, i);
            if let Some((ip, srv)) = off {
                kprintln!("dhcp: OFFER {} from {} after {} ms ({} frames seen)", dhcp::Ip(ip), dhcp::Ip(srv), t, frames);
                return;
            }
        }
        wdt::pet();
        wdt::delay_ms(10);
        t += 10;
    }
    kprintln!("dhcp: no OFFER in 4 s ({} frames received)", frames);
}
/// Back from the SSH terminal: blank the screen and log to it again.
pub fn vga_log_on() {
    unsafe {
        for i in 0..W * H {
            VGA.add(i).write_volatile(ATTR | b' ' as u16);
        }
        ROW = 0;
        COL = 0;
        VGA_LOG = true;
    }
}

/// Raw byte to COM1 (used to mirror the remote shell's output into the serial log).
pub fn serial_raw(b: u8) {
    serial_putc(b);
}

pub fn init() {
    unsafe {
        outb(COM1 + 1, 0x00); // no interrupts
        outb(COM1 + 3, 0x80); // DLAB
        outb(COM1 + 0, 0x01); // divisor 1 = 115200
        outb(COM1 + 1, 0x00);
        outb(COM1 + 3, 0x03); // 8N1
        outb(COM1 + 2, 0xC7); // FIFO
        outb(COM1 + 4, 0x03);
        for i in 0..W * H {
            VGA.add(i).write_volatile(ATTR | b' ' as u16);
        }
        ROW = 0;
        COL = 0;
    }
}

fn serial_putc(b: u8) {
    unsafe {
        // Bounded: a wedged or absent UART must never keep a failure path from reaching its 0xCF9 reset.
        for _ in 0..100_000 {
            if inb(COM1 + 5) & 0x20 != 0 {
                outb(COM1, b);
                return;
            }
        }
    }
}

fn vga_putc(b: u8) {
    unsafe {
        if b == b'\n' {
            COL = 0;
            ROW += 1;
        } else {
            VGA.add(ROW * W + COL).write_volatile(ATTR | b as u16);
            COL += 1;
            if COL == W {
                COL = 0;
                ROW += 1;
            }
        }
        if ROW == H {
            for i in 0..W * (H - 1) {
                VGA.add(i).write_volatile(VGA.add(i + W).read_volatile());
            }
            for i in 0..W {
                VGA.add((H - 1) * W + i).write_volatile(ATTR | b' ' as u16);
            }
            ROW = H - 1;
        }
    }
}

/// Point of no return: NIC quiesced (reset completed), trampoline entered. The TCO stays armed across the hop.
/// If the NIC reset does not complete, its DMA may still target memory the new image would overwrite: hard reset
/// instead of jumping (GRUB default, the agent OS, then the crash-loop escape).
pub fn jump(p: &Plan, nic: &Rtl) -> ! {
    if let Err(e) = nic.quiesce() {
        kprintln!("loader: NIC quiesce failed ({}): NOT jumping, hard reset", e);
        crate::wdt::hard_reset();
    }
    unsafe {
        for (i, v) in p.params.iter().enumerate() {
            core::ptr::write_volatile((PARAMS + 8 * i as u64) as *mut u64, *v);
        }
        crate::wdt::pet();
        let f: extern "sysv64" fn(u64) -> ! = core::mem::transmute(TRAMP as usize);
        f(PARAMS)
    }
}
/// Set once a TX descriptor fails to come back (OWN still set after the deadline). From then on the
/// TX DMA buffer and descriptor are treated as owned by the NIC forever: nothing is written to them.
static mut TX_WEDGED: bool = false;
static mut TX_SCRATCH: [u8; BUFSZ] = [0; BUFSZ];

/// Let `fill` build a `len`-byte frame directly in the DMA buffer, then send it.
/// Never touches the buffer/descriptor unless the NIC has cleared OWN (Codex M2 MUST-FIX).
pub fn tx_with<R2>(bus: Bus, r: &Regs, len: usize, fill: impl FnOnce(&mut [u8]) -> R2) -> R2 {
    let len = len.min(BUFSZ);
    unsafe {
        let d = addr_of_mut!(TX_DESC.0[0]);
        if TX_WEDGED || read_volatile(addr_of!((*d).opts1)) & r.desc_own != 0 {
            TX_WEDGED = true;
            // Frame is built into scratch and dropped: the NIC may still be DMAing the real buffer.
            let scratch = &mut *core::ptr::addr_of_mut!(TX_SCRATCH);
            return fill(&mut scratch[..len]);
        }
        let buf = core::slice::from_raw_parts_mut(addr_of_mut!(TX_BUF.0[0]) as *mut u8, BUFSZ);
        let res = fill(&mut buf[..len]);
        let wire = len.max(60);
        for b in &mut buf[len..wire] {
            *b = 0;
        }
        write_volatile(
            d,
            Desc {
                opts1: r.desc_own | r.ring_end | r.first_frag | r.last_frag | wire as u32,
                opts2: 0,
                addr: buf.as_ptr() as u64,
            },
        );
        fence(Ordering::SeqCst);
        bus.w8(r.tx_poll, r.npq);
        let t0 = rdtsc();
        while read_volatile(addr_of!((*d).opts1)) & r.desc_own != 0 {
            if rdtsc() - t0 > 2_000_000_000 {
                TX_WEDGED = true;
                kprintln!("nic: TX descriptor not returned by NIC, TX disabled");
                break;
            }
            core::hint::spin_loop();
        }
        fence(Ordering::SeqCst);
        let st = bus.r16(r.intr_status);
        bus.w16(r.intr_status, st);
        res
    }
}

Most is sound. **I found no commit-specific MUST-FIX under the stated “600 seconds without a fresh authenticated command” contract.** Removing `hard_end` is supported by the control flow. This was a source review; I did not rerun QEMU tests.

**MUST-FIX: none found.**

1. **Plain bmctl’s serving semantics are preserved.** Every replaced `continue` becomes `return None`, after which `serve` begins the next iteration with the same watchdog pet, RX budget of 32, and interface poll. It still processes at most one datagram per iteration. Fields retain their initial values and updates; deadline ordering remains **idle → jump → reboot → hello → receive/authenticate**. Retransmissions and rejected/replayed packets still do not refresh `last_ok`. See [bmctl.rs:278](~/nc110-baremetal/src/bmctl.rs:278).

   “Exactly preserved” needs a startup qualification: network randomness is now drawn before the boot ID; boot-ID generation, boot-info snapshotting, and key derivation move until after DHCP. This changes RNG consumption order and entropy-failure timing, but I found no authentication or recovery regression.

2. **The SSH outer loop cannot repeatedly pet while skipping bmctl.** With `ctl = Some`, each iteration sets RX budget 64, polls, then calls `step` before connection handling or SSH processing. `ctl` is never subsequently cleared. If polling or later processing wedges, it cannot return to the outer-loop pet. See [ssh.rs:194](~/nc110-baremetal/src/ssh.rs:194).

   The runner event loop, channel drain, output drain, keyboard handling, crypto, and terminal rendering contain no watchdog pets. Even an infinite `PollAgain`/`Progressed` cycle therefore leaves recovery to TCO. SSH DHCP lacks an RX budget and could remain inside polling under a flood, but it also does **not** pet: that is a watchdog-reset case, not an indefinite survival case.

3. **Exit paths preserve recovery.** Connection failure, protocol/authentication failure, startup deadline expiry, and ordinary closure all reach `serve` with the **existing** controller and `last_ok`. They do not restart the idle allowance. The `finished` path performs only 50 extra polls, without petting or replenishing the remaining RX budget, then reaches `serve`. Idle sets `bm_idle`, skips `serve`, and returns toward reset. Panics halt without petting. See [ssh.rs:463](~/nc110-baremetal/src/ssh.rs:463).

4. **Seed/key handling is correct.** `Ctl::new` receives a value copy of the original seed before SSH wipes its local copy. The controller retains the correct seed for chain-loading and derives the same bmctl key. Boot-ID authentication, session derivation, sequence handling, and host-key verification remain intact. Retaining the seed is intentional and already necessary in plain bmctl.

**SHOULD**

- **Distinguish authenticated ingress from bidirectional reachability.** There is an inherited limitation in [bmctl.rs:417](~/nc110-baremetal/src/bmctl.rs:417): fresh authenticated requests refresh idle even if every response is lost. Concrete scenario: TX becomes permanently wedged while RX works; an established client keeps sending increasing-sequence authenticated commands. Replies disappear, but idle never expires and the loop keeps petting. Ordinary retries of the same sequence cannot do this. If the hard rule includes this asymmetric failure despite continuing authenticated ingress, this is a **system-level MUST-FIX**, shared with plain bmctl; the removed SSH wall-clock bound previously limited its duration in SSH mode.

- **Expand recovery tests.** The new test covers overlap, post-session service, and idle during an open session. Add idle after early connection/authentication failure, invalid/replayed traffic under flood, and an injected inner-loop hang demonstrating TCO recovery. These directly exercise the replacement for the removed safety bound.

**NIT**

- Idle expiry does not mean immediate reset: the existing main epilogue logs, then pets for another 60 seconds before CF9 reset ([main.rs:111](~/nc110-baremetal/src/main.rs:111)).
- Update `net.rs`’s “SSH never sets [RX budget]” comment and the unconditional wall-clock-bound comment in `ssh.rs`.
