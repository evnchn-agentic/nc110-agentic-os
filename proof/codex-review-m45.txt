   344	        if f == b'c' && self.inter == 0 && self.arg(0, 0) == 0 {
   345	            match self.lead {

   348	                _ => {}

   350	            return;

   353	            match self.arg(0, 0) {
   354	                5 => self.send(1, b"\x1b[0n"),
   355	                6 => {
   356	                    // CPR, 1-based; a pending wrap reports the last column, as xterm does

   468	    pub fn feed(&mut self, b: u8) {
   469	        if self.need > 0 {
   470	            if b & 0xC0 == 0x80 {
   471	                self.utf = (self.utf << 6) | (b & 0x3F) as u32;
   472	                self.need -= 1;
   473	                if self.need == 0 {
   474	                    let cp = self.utf;
   475	                    let ok = cp >= self.min && !(0xD800..=0xDFFF).contains(&cp) && cp <= 0x10FFFF;
   476	                    self.print(if ok { cp } else { b'?' as u32 });
   477	                }
   478	                return;
   479	            }
   480	            self.need = 0; // sequence cut short (by a control, ESC, or a new lead byte): one '?', then this byte
   481	            self.print(b'?' as u32);
   482	        }
   483	        match self.st {
   484	            St::Ground => match b {
   485	                0x1B => self.st = St::Esc,
   486	                b'\r' => {
   487	                    self.col = 0;

/// Poll the controller and append the bytes typed so far to `buf[*len..]`. Mouse (AUX) bytes are dropped.
/// Always drains the controller so modifier releases are never missed; if `buf` is full (the SSH channel is not
/// taking input), the key's bytes are dropped and counted rather than left to overflow the controller.
/// Returns true on Ctrl-Alt-Del.
pub fn poll(buf: &mut [u8], len: &mut usize) -> bool {
    unsafe {
        if !ENABLED {
            drain();
            return false;
        }
        for _ in 0..64 {
            let st = inb(STATUS);
            if st & 1 == 0 {
                break;
            }
            let sc = inb(DATA);
            if st & 0x20 != 0 {
                continue; // AUX (touchpad) byte
            }
            if E0 && sc == 0x53 && (CTRL_L || CTRL_R) && (ALT_L || ALT_R) {
                return true;
            }
            let mut tmp = [0u8; 8];
            let n = translate(sc, &mut tmp);
            if buf.len() - *len >= n {
                buf[*len..*len + n].copy_from_slice(&tmp[..n]);
                *len += n;
            } else {
                DROPPED = DROPPED.wrapping_add(1);
            }
        }
        false
    }
}

Most is sound. I rebuilt and ran all 29 host tests successfully, and reproduced two rendering issues. I found no new release-build panic/OOB or arbitrary-byte reply injection.

- **MUST-FIX — Incomplete width tables desynchronize application columns.** [src/term.rs:789](~/nc110-baremetal/src/term.rs:789) declares `const ZERO: [(u32, u32); 8]`; line 791 classifies all `(0x3041, 0x33FF)` as wide. Consequently, `か\u3099X` renders `????X` instead of `??X`: the combining dakuten consumes two cells. Arabic U+064E similarly consumes one, while 🚀 incorrectly consumes one instead of two.
