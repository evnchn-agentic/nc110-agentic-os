+//! PS/2 keyboard, polled through the i8042 (data 0x60, status 0x64), scancode set 1.
+//!
+//! Set 1 because the firmware leaves the i8042's set-2 -> set-1 translation on (SeaBIOS in QEMU does;
+//! on the NC110 the internal keyboard sits behind the EC's i8042 emulation, which is the same 0x60/0x64
+//! interface; its translation state is unverified until M1-on-hardware). We never reprogram the controller.
+//! Output is what an xterm would send: Enter = CR, Backspace = DEL (0x7F), Ctrl+letter = letter & 0x1F,
+//! arrows/Home/End/Delete as CSI sequences. US layout only.

+const STATUS: u16 = 0x64;

+static mut SHIFT_R: bool = false;
+static mut CTRL: bool = false;
+static mut CAPS: bool = false;
+static mut E0: bool = false;

+    m[0x39] = (b' ', b' ');
+    m

+/// Drain whatever the controller has buffered (firmware leftovers) so the first key is a real one.
+pub fn init() -> usize {
+    let mut n = 0;
+    unsafe {
+        while inb(STATUS) & 1 != 0 && n < 64 {
+            let _ = inb(DATA);
+            n += 1;
+        }
+    }
+    n
+}

+/// Translate one scancode; appends 0..4 bytes to `out` and returns how many.
+fn translate(sc: u8, out: &mut [u8; 4]) -> usize {
+    unsafe {
+        if sc == 0xE0 {
+            E0 = true;
+            return 0;
+        }
+        let ext = E0;
+        E0 = false;
+        let brk = sc & 0x80 != 0;
+        let code = sc & 0x7F;
+        match (ext, code) {
+            (false, 0x2A) => SHIFT_L = !brk,
+            (false, 0x36) => SHIFT_R = !brk,
+            (_, 0x1D) => CTRL = !brk, // left ctrl, or E0 1D right ctrl
+            (false, 0x3A) if !brk => CAPS = !CAPS,
+            _ => {}
+        }
+        if brk {
+            return 0;
+        }
+        if ext {
+            let seq: &[u8] = match code {
+                0x48 => b"\x1b[A",
+                0x50 => b"\x1b[B",
+                0x4D => b"\x1b[C",
+                0x4B => b"\x1b[D",
+                0x47 => b"\x1b[H",
+                0x4F => b"\x1b[F",
+                0x53 => b"\x1b[3~",
+                0x1C => b"\r", // keypad Enter
+                0x35 => b"/",  // keypad /
+                _ => b"",
+            };
+            out[..seq.len()].copy_from_slice(seq);
+            return seq.len();
+        }
+        if (code as usize) >= MAP.len() {
+            return 0;
+        }
+        let (lo, hi) = MAP[code as usize];
+        if lo == 0 {
+            return 0;
+        }
+        let shift = SHIFT_L || SHIFT_R;
+        let mut c = if shift { hi } else { lo };
+        if CAPS && lo.is_ascii_lowercase() {
+            c = if shift { lo } else { hi };
+        }
+        if CTRL {
+            c = match c {
+                b'a'..=b'z' | b'A'..=b'Z' => c & 0x1F,
+                b'@' | b' ' | b'2' => 0x00,
+                b'[' => 0x1B,
+                b'\\' => 0x1C,
+                b']' => 0x1D,
+                b'_' | b'-' => 0x1F,
+                _ => c,
+            };
+        }
+        out[0] = c;
+        1
+    }
+}
+
+/// Poll the controller and append the bytes typed so far to `buf[*len..]`. Mouse (AUX) bytes are dropped.
+/// Stops reading scancodes once fewer than 4 bytes of room remain, so nothing typed is lost (the i8042 holds it).
+pub fn poll(buf: &mut [u8], len: &mut usize) {
+    unsafe {
+        while buf.len() - *len >= 4 {
+            let st = inb(STATUS);
+            if st & 1 == 0 {
+                break;
+            }
+            let sc = inb(DATA);
+            if st & 0x20 != 0 {
+                continue; // AUX (touchpad) byte
+            }
+            let mut tmp = [0u8; 4];
+            let n = translate(sc, &mut tmp);
+            buf[*len..*len + n].copy_from_slice(&tmp[..n]);
+            *len += n;
+        }
+    }
+}

+//! VGA text terminal (80x25 at 0xB8000) with the VT100/xterm subset a /bin/sh session needs:
+//! CR LF BS TAB BEL, deferred autowrap, scrolling, CSI A B C D G d H f J K P @ X L M, ESC 7/8/M/c.
+//! SGR (colours) and modes (?h/?l, bracketed paste) are parsed and ignored; OSC titles are swallowed;
+//! UTF-8 multibyte characters show as one '?'. The hardware cursor follows the text cursor.
+use crate::port::outb;
+
+const VGA: *mut u16 = 0xB8000 as *mut u16;
+const W: usize = 80;
+const H: usize = 25;
+const ATTR: u16 = 0x0700; // light grey on black (the log console before the shell is white on blue)
+const BLANK: u16 = ATTR | b' ' as u16;
+
+#[derive(Clone, Copy, PartialEq)]
+enum St {
+    Ground,
+    Esc,
+    EscSkip, // ESC ( B and friends: one charset byte follows
+    Csi,
+    Osc,
+    OscEsc,
+}
+
+struct T {
+    row: usize,
+    col: usize,
+    wrap: bool, // deferred wrap: last column written, next printable goes to the next line
+    st: St,
+    p: [u16; 8],
+    np: usize,
+    private: bool,
+    saved: (usize, usize),
+}
+
+static mut T: T = T { row: 0, col: 0, wrap: false, st: St::Ground, p: [0; 8], np: 0, private: false, saved: (0, 0) };
+
+fn put(r: usize, c: usize, v: u16) {
+    unsafe { VGA.add(r * W + c).write_volatile(v) }
+}
+fn get(r: usize, c: usize) -> u16 {
+    unsafe { VGA.add(r * W + c).read_volatile() }
+}
+fn clear_range(from: usize, to: usize) {
+    for i in from..to {
+        unsafe { VGA.add(i).write_volatile(BLANK) }
+    }
+}
+fn scroll_up(top: usize) {
+    for r in top..H - 1 {
+        for c in 0..W {
+            put(r, c, get(r + 1, c));
+        }
+    }
+    clear_range((H - 1) * W, H * W);
+}
+fn scroll_down(top: usize) {
+    for r in (top + 1..H).rev() {
+        for c in 0..W {
+            put(r, c, get(r - 1, c));
+        }
+    }
+    clear_range(top * W, top * W + W);
+}
+
+fn hw_cursor(t: &T) {
+    let pos = (t.row * W + t.col.min(W - 1)) as u16;
+    unsafe {
+        outb(0x3D4, 0x0A);
+        outb(0x3D5, 13); // cursor scanlines 13..14, enabled
+        outb(0x3D4, 0x0B);
+        outb(0x3D5, 14);
+        outb(0x3D4, 0x0F);
+        outb(0x3D5, pos as u8);
+        outb(0x3D4, 0x0E);
+        outb(0x3D5, (pos >> 8) as u8);
+    }
+}
+
+/// Take over the screen: clear it and home the cursor.
+pub fn init() {
+    clear_range(0, W * H);
+    unsafe {
+        let t = &mut *core::ptr::addr_of_mut!(T);
+        *t = T { row: 0, col: 0, wrap: false, st: St::Ground, p: [0; 8], np: 0, private: false, saved: (0, 0) };
+        hw_cursor(t);
+    }
+}
+
+fn lf(t: &mut T) {
+    if t.row == H - 1 {
+        scroll_up(0);
+    } else {
+        t.row += 1;
+    }
+}
+
+fn arg(t: &T, i: usize, def: u16) -> usize {
+    let v = if i < t.np { t.p[i] } else { 0 };
+    (if v == 0 { def } else { v }) as usize
+}
+
+fn csi(t: &mut T, f: u8) {
+    if t.private {
+        return; // ?25h/l, ?2004h/l, ?1h ... : modes we do not model
+    }
+    let n = arg(t, 0, 1);
+    match f {
+        b'A' => t.row = t.row.saturating_sub(n),
+        b'B' | b'e' => t.row = (t.row + n).min(H - 1),
+        b'C' | b'a' => t.col = (t.col + n).min(W - 1),
+        b'D' => t.col = t.col.saturating_sub(n),
+        b'G' | b'`' => t.col = (n - 1).min(W - 1),
+        b'd' => t.row = (n - 1).min(H - 1),
+        b'H' | b'f' => {
+            t.row = (arg(t, 0, 1) - 1).min(H - 1);
+            t.col = (arg(t, 1, 1) - 1).min(W - 1);
+        }
+        b'J' => {
+            let cur = t.row * W + t.col;
+            match arg(t, 0, 0) {
+                0 => clear_range(cur, W * H),
+                1 => clear_range(0, cur + 1),
+                _ => clear_range(0, W * H),
+            }
+        }
+        b'K' => {
+            let s = t.row * W;
+            match arg(t, 0, 0) {
+                0 => clear_range(s + t.col, s + W),
+                1 => clear_range(s, s + t.col + 1),
+                _ => clear_range(s, s + W),
+            }
+        }
+        b'P' => {
+            // delete n chars, shift the rest of the line left
+            let n = n.min(W - t.col);
+            for c in t.col..W {
+                put(t.row, c, if c + n < W { get(t.row, c + n) } else { BLANK });
+            }
+        }
+        b'@' => {
+            let n = n.min(W - t.col);
+            for c in (t.col..W).rev() {
+                put(t.row, c, if c >= t.col + n { get(t.row, c - n) } else { BLANK });
+            }
+        }
+        b'X' => {
+            let s = t.row * W + t.col;
+            clear_range(s, s + n.min(W - t.col));
+        }
+        b'L' => {
+            for _ in 0..n.min(H - t.row) {
+                scroll_down(t.row);
+            }
+        }
+        b'M' => {
+            for _ in 0..n.min(H - t.row) {
+                scroll_up(t.row);
+            }
+        }
+        _ => {} // 'm' (SGR), 'r' (scroll region), 'h'/'l', 'n' (DSR: no reply), ...: ignored
+    }
+    t.wrap = false;
+}
+
+/// Feed one byte of the remote's output.
+pub fn feed(b: u8) {
+    let t = unsafe { &mut *core::ptr::addr_of_mut!(T) };
+    match t.st {
+        St::Ground => match b {
+            0x1B => t.st = St::Esc,
+            b'\r' => {
+                t.col = 0;
+                t.wrap = false;
+            }
+            b'\n' | 0x0B | 0x0C => {
+                lf(t);
+                t.wrap = false;
+            }
+            0x08 => {
+                t.col = t.col.saturating_sub(1);
+                t.wrap = false;
+            }
+            b'\t' => t.col = ((t.col + 8) & !7).min(W - 1),
+            0x80..=0xBF => {} // UTF-8 continuation byte
+            _ if b < 0x20 || b == 0x7F => {} // BEL and other controls
+            _ => {
+                if t.wrap {
+                    t.col = 0;
+                    lf(t);
+                    t.wrap = false;
+                }
+                let ch = if b >= 0x80 { b'?' } else { b };
+                put(t.row, t.col, ATTR | ch as u16);
+                if t.col == W - 1 {
+                    t.wrap = true;
+                } else {
+                    t.col += 1;
+                }
+            }
+        },
+        St::Esc => {
+            t.st = St::Ground;
+            match b {
+                b'[' => {
+                    t.st = St::Csi;
+                    t.p = [0; 8];
+                    t.np = 0;
+                    t.private = false;
+                }
+                b']' => t.st = St::Osc,
+                b'(' | b')' | b'*' | b'+' | b'#' | b'%' => t.st = St::EscSkip,
+                b'7' => t.saved = (t.row, t.col),
+                b'8' => (t.row, t.col) = t.saved,
+                b'D' => lf(t),
+                b'E' => {
+                    t.col = 0;
+                    lf(t);
+                }
+                b'M' => {
+                    if t.row == 0 {
+                        scroll_down(0);
+                    } else {
+                        t.row -= 1;
+                    }
+                }
+                b'c' => init(),
+                _ => {} // '=', '>', ...
+            }
+        }
+        St::EscSkip => t.st = St::Ground,
+        St::Csi => match b {
+            b'0'..=b'9' => {
+                if t.np == 0 {
+                    t.np = 1;
+                }
+                let i = t.np - 1;
+                if i < t.p.len() {
+                    t.p[i] = t.p[i].saturating_mul(10).saturating_add((b - b'0') as u16);
+                }
+            }
+            b';' => {
+                if t.np == 0 {
+                    t.np = 1;
+                }
+                t.np += 1;
+            }
+            b'?' | b'>' | b'<' | b'=' => t.private = true,
+            0x20..=0x2F => {} // intermediates
+            0x40..=0x7E => {
+                t.st = St::Ground;
+                csi(t, b);
+            }
+            0x1B => t.st = St::Esc,
+            _ => {} // controls inside CSI: ignored
+        },
+        St::Osc => match b {
+            0x07 => t.st = St::Ground,
+            0x1B => t.st = St::OscEsc,
+            _ => {}
+        },
+        St::OscEsc => t.st = if b == b'\\' { St::Ground } else { St::Osc },
+    }
+}
+
+/// Move the hardware cursor to the text cursor (once per batch of output, not per byte).
+pub fn sync_cursor() {
+    hw_cursor(unsafe { &*core::ptr::addr_of!(T) });
+}

fn rd<T: Copy>(p: *const T) -> T {
    unsafe { read_volatile(p) }

pub fn armed() -> bool {
    matches!(rd(addr_of!(tco_result)), 0 | 4 | 5)
}

fn pmbase() -> u16 {
    (pci::read32(pci::b0(31, 0), 0x40) & 0xFF80) as u16
}

    pmbase() + 0x60

pub fn report() {
    let r = rd(addr_of!(tco_result));
    let what = match r {
        0 => "armed by boot.s",
        1 => "skipped (LPC 00:1f.0 is not ICH9 8086:2918 / NM10 8086:27bc)",
        2 => "NOT armed: PMBASE unset",
        3 => "NOT armed: RCBA disabled",
        4 => "left to GRUB (grub-armed build); live registers follow",
        5 => "NM10: GRUB owns the watchdog, kernel hands off; live registers follow",
        _ => "NOT armed: unknown",
    };
    kprintln!("wdt: {} (result={})", what, r);
    if r == 4 || r == 5 {
        let b = base();
        let gcs = unsafe { read_volatile(((pci::read32(pci::b0(31, 0), 0xF0) & 0xFFFF_C000) + 0x3410) as *const u32) };
        let (s1, s2, cnt, tmr) = unsafe { (inw(b + 4), inw(b + 6), inw(b + 8), inw(b + 0x12)) };

            "wdt: live TCOBASE={:#x} GCS={:#010x} NO_REBOOT={} TCO1_CNT={:#06x} HLT={} TCO1_STS={:#06x} TCO2_STS={:#06x} TCO_TMR={}",
            b, gcs, (gcs >> 5) & 1, cnt, (cnt >> 11) & 1, s1, s2, tmr & 0x3FF
        );
        return;
    }
    if r != 0 {
        return;
    }
    let (b, gcs, s1, s2, cnt) = {
        (rd(addr_of!(tco_base)) as u16, rd(addr_of!(tco_gcs)), rd(addr_of!(tco_sts1_pre)), rd(addr_of!(tco_sts2_pre)), rd(addr_of!(tco_cnt)))
    };

/// Live check of the reset path, whoever armed it: NO_REBOOT=0, HLT=0, TMR>=4, and the counter
/// (TCO_RLD reads back the current value on v2) decrements across ~1.3 ticks.
pub fn verify_live() -> bool {
    if !armed() {
        return false;
    }
    let b = base();
    let rcba = pci::read32(pci::b0(31, 0), 0xF0);
    if rcba & 1 == 0 {
        return false;
    }
    let gcs = unsafe { read_volatile(((rcba & 0xFFFF_C000) + 0x3410) as *const u32) };
    let (cnt, tmr) = unsafe { (inw(b + 8), inw(b + 0x12) & 0x3FF) };
    let c0 = unsafe { inw(b) } & 0x3FF;
    delay_ms(800);
    let c1 = unsafe { inw(b) } & 0x3FF;
    let ok = gcs & 0x20 == 0 && cnt & 0x0800 == 0 && tmr >= 4 && c1 != c0;
    kprintln!(
        "wdt: verify NO_REBOOT={} HLT={} TMR={} count {}->{} => {}",
        (gcs >> 5) & 1, (cnt >> 11) & 1, tmr, c0, c1, if ok { "VERIFIED" } else { "NOT VERIFIED" }
    );
    ok
}

/// ACPI PM timer (PMBASE+0x08, 3.579545 MHz, 24-bit) based delay; same block on ICH9 and NM10.
fn pm_timer() -> u32 {
    unsafe { inl(pmbase() + 8) & 0x00FF_FFFF }
}

pub fn delay_ms(ms: u32) {
    let need = ms as u64 * 3_579_545 / 1000;
    let mut last = pm_timer();
    let mut acc = 0u64;
    while acc < need {
        let now = pm_timer();
        acc += (now.wrapping_sub(last) & 0x00FF_FFFF) as u64;
        last = now;
    }
}

/// `wdt=pet:N` pets every 500 ms for N s, then stops and hangs. `wdt=hang` hangs at once.

    let arg = cmdline[i + 4..].split(' ').next().unwrap_or("");
    if !armed() {
        kprintln!("wdt: mode {} requested but watchdog not armed", arg);
        return false;
    }

        for _ in 0..2 {
            pet();
            delay_ms(500);
        }
        kprintln!("wdt: alive t={}s (petting every 500 ms)", s + 1);
    }
    kprintln!("wdt: HANG now: cli; hlt with the timer running, nobody pets");
    loop {
        unsafe { core::arch::asm!("cli; hlt") };
    }
}

/// OpenSSH-style "SHA256:<base64, no padding>" of the ssh-ed25519 public key blob.
fn fingerprint(key: &[u8; 32], out: &mut [u8; 43]) {
    let mut h = Sha256::new();
    h.update(11u32.to_be_bytes());
    h.update(b"ssh-ed25519");
    h.update(32u32.to_be_bytes());
    h.update(key);
    let d = h.finalize();
    const A: &[u8; 64] = b"ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/";
    let mut bits: u32 = 0;
    let mut n = 0;
    let mut o = 0;
    for &b in d.iter() {
        bits = (bits << 8) | b as u32;
        n += 8;
        while n >= 6 {
            n -= 6;
            out[o] = A[((bits >> n) & 63) as usize];
            o += 1;
        }
    }
    if n > 0 {
        out[o] = A[((bits << (6 - n)) & 63) as usize];
    }
}

const W: usize = 80;
const H: usize = 25;
const ATTR: u16 = 0x1F00; // white on blue

static mut ROW: usize = 0;
static mut COL: usize = 0;
/// Cleared when the SSH terminal takes over the screen: log lines then go to COM1 only.
static mut VGA_LOG: bool = true;

pub fn vga_log_off() {
    unsafe { VGA_LOG = false };
}

/// Raw byte to COM1 (used to mirror the remote shell's output into the serial log).
pub fn serial_raw(b: u8) {
    serial_putc(b);
}

pub fn init() {
    unsafe {
        outb(COM1 + 1, 0x00); // no interrupts
        outb(COM1 + 3, 0x80); // DLAB
        outb(COM1 + 0, 0x01); // divisor 1 = 115200
        outb(COM1 + 1, 0x00);
        outb(COM1 + 3, 0x03); // 8N1
        outb(COM1 + 2, 0xC7); // FIFO
        outb(COM1 + 4, 0x03);
        for i in 0..W * H {
            VGA.add(i).write_volatile(ATTR | b' ' as u16);
        }
        ROW = 0;
        COL = 0;
    }
}

fn serial_putc(b: u8) {
    unsafe {
        while inb(COM1 + 5) & 0x20 == 0 {}
        outb(COM1, b);
    }
}

fn vga_putc(b: u8) {
    unsafe {
        if b == b'\n' {
            COL = 0;
            ROW += 1;
        } else {
            VGA.add(ROW * W + COL).write_volatile(ATTR | b as u16);
            COL += 1;
            if COL == W {
                COL = 0;
                ROW += 1;
            }
        }
        if ROW == H {
            for i in 0..W * (H - 1) {
                VGA.add(i).write_volatile(VGA.add(i + W).read_volatile());
            }
            for i in 0..W {
                VGA.add((H - 1) * W + i).write_volatile(ATTR | b' ' as u16);
            }
            ROW = H - 1;
        }
    }
}

    fn write_str(&mut self, s: &str) -> fmt::Result {
        for &b in s.as_bytes() {
            if b == b'\n' {
                serial_putc(b'\r');
            }
            serial_putc(b);
            if unsafe { VGA_LOG } {
                vga_putc(b);
            }
        }
        Ok(())
    }
}

Most of the bounds and data movement code is sound. I found one significant termination defect and three smaller issues. Reviewed commit `7cbb8e3`, its installed dependencies, and the existing proof log; no files changed and no new QEMU run.

1. **MUST-FIX — Dead sessions can spin forever while petting the watchdog.** [ssh.rs:210](~/nc110-baremetal/src/ssh.rs:210), [ssh.rs:164](~/nc110-baremetal/src/ssh.rs:164)

   A TCP FIN without an SSH channel-close/disconnect leaves smoltcp in `CloseWait`. Its `is_active()` remains **true**, so this code never calls `runner.close_input()`. With `shell_open == true`, the deadline no longer applies. The loop continues petting TCO indefinitely.

   A blackholed connection has the same recovery problem: neither TCP timeout nor keepalive is configured. Also, Sunset 0.6.0 sends PTY/shell requests with `want_reply: false`; their `Ok` results mean locally queued, not remotely accepted. Thus `shell_open` disables the startup deadline before shell startup is established.

   **Fix:** detect drained receive-side closure using `may_recv()`, retain a startup deadline, and provide bounded peer-liveness detection plus a local escape/reset path. A healthy idle shell should remain open; merely iterating the loop must not be mistaken for a usable connection. Normal channel EOF/close termination does work.

2. **SHOULD-FIX — Keyboard backpressure can lose scancodes and leave modifiers stuck.** [kbd.rs:137](~/nc110-baremetal/src/kbd.rs:137)

   Once the 64-byte input queue has fewer than four free bytes, polling stops entirely—including processing Shift/Ctrl releases. If SSH stays blocked and typing continues, upstream buffering eventually fills. Losing a modifier release leaves subsequent input incorrectly modified.

   The comment claiming “nothing typed is lost” is false even for QEMU: its PS/2 implementation has a finite queue and drops input when full. [QEMU source](https://raw.githubusercontent.com/qemu/qemu/master/hw/input/ps2.c)

   **Fix:** continue draining and tracking keyboard state independently of channel capacity, with a bounded queue and explicit overflow/recovery policy. Increasing the queue alone cannot guarantee losslessness under indefinite backpressure.

3. **SHOULD-FIX — Ignored SGR sequences cancel deferred autowrap and overwrite text.** [term.rs:160](~/nc110-baremetal/src/term.rs:160)

   Deterministic example from a cleared screen: feed 80 `A` bytes, then `ESC[0mB`. The 80th `A` sets pending wrap; the ignored SGR clears it; `B` overwrites column 80 instead of appearing at the start of the next row. Ordinary coloured output can trigger this even when displaying colours is deliberately unsupported.

   **Fix:** preserve pending wrap for SGR and other operations that should not change cursor/wrap state.

4. **SHOULD-FIX before hardware acceptance — Set-1 decoding depends on unverified firmware state.** [kbd.rs:61](~/nc110-baremetal/src/kbd.rs:61)

   Initialization only drains data. It neither checks translation nor establishes the keyboard’s scan mode. If the NC110 delivers untranslated set 2, decoding becomes incorrect immediately—for example, set-2 `A` make code `0x1C` becomes Enter here. Linux explicitly distinguishes translated and untranslated controller configurations. [Linux i8042 source](https://raw.githubusercontent.com/torvalds/linux/master/drivers/input/serio/i8042.c)

   This is a **conditional hardware risk**, not evidence that this NC110 actually boots in the wrong mode. Validate the actual GRUB-to-kernel handoff with raw make/break captures and controller state; then enforce or explicitly gate the supported configuration.

The suspected memory-corruption paths did **not** hold up: row/column invariants stay within 25×80; numeric CSI parameters saturate; P/@/X clamp to remaining columns; L/M clamp to remaining rows; copy directions are correct. Excess CSI parameters do not cause array indexing beyond the eight slots with the current callers.

Likewise, `write_channel` partial writes—including `Ok(0)` during backpressure/rekey—retain the unsent suffix correctly. Keyboard appends respect available capacity at the current call site.

**NIT:** None worth adding. The existing proof demonstrates the happy path, but its machine has no armed TCO and it does not exercise these failure cases.
