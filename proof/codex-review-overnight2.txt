# Shared iTCO v2 arm + verify for GRUB 2 (needs insmod iorw memrw setpci regexp). Used verbatim by the QEMU
# rehearsal (grub-test/iso/boot/grub/grub.cfg) and the proposed NC110 entry (grub-test/m1-nc110-entry.cfg).
# Caller sets: lpc_expect pmbase_expect rcba_expect gcs_addr smi_en tco_rld tco1_sts tco2_sts tco1_cnt tco_tmr
#              tmr_val tmr_re. GRUB prints values as %x (lowercase, no 0x, no padding); bit tests are regexps.
# Optional caller override: lock_mode (default refuse, set below). Callers source this file FIRST, then set it.
#   Policy is consumed per call: tco_arm resets lock_mode to refuse first thing, so set it before EACH call.
#   refuse : TCO_LOCK set -> do not arm (the original M1 design: it needs SMI_EN.TCO_EN cleared).
#   accept : TCO_LOCK set -> arm anyway, preserving TCO_LOCK and never writing SMI_EN: TCO_EN is not checked, so the
#            first timeout raises an SMI to the firmware and the reset relies on the second timeout. Only valid
#            once a hardware test showed this firmware's SMI handler does NOT swallow the timeout (NC110 step 1).
#            Linux iTCO_wdt does the same on v2 by default (turn_SMI_watchdog_clear_off=1 < version 2).
set lock_mode=refuse
# tco_arm sets wdt_ok=yes ONLY if every live predicate holds. The caller must boot only inside that branch.

# Bit predicates on %x strings:
#   GCS bit 5 (NO_REBOOT) clear      : nibble 1 in {0,1,4,5,8,9,c,d}
#   TCO1_CNT bit 11 (TMR_HLT) clear  : nibble 2 < 8
#   TCO1_CNT bit 12 (TCO_LOCK) clear : nibble 3 even
#   SMI_EN bit 13 (TCO_EN) clear     : nibble 3 in {0,1,4,5,8,9,c,d}
set re_noreboot_clear='^([0-9a-f]|[0-9a-f]*[014589cd][0-9a-f])$'
set re_hlt_clear='^([0-9a-f]{1,2}|[0-7][0-9a-f]{2}|[0-9a-f]*[0-9a-f][0-7][0-9a-f]{2})$'
set re_lock_clear='^([0-9a-f]{1,3}|[0-9a-f]*[02468ace][0-9a-f]{3})$'
set re_tcoen_clear='^([0-9a-f]{1,3}|[0-9a-f]*[014589cd][0-9a-f]{3})$'
set re_hex16='^[0-9a-f]{1,4}$'

function tco_re_check {
  # $1 = regexp, $2 = value, $3 = expected (yes/no). Sets re_bad=yes on a wrong answer.
  if regexp "$1" "$2"; then set got=yes; else set got=no; fi
  if [ "$got" != "$3" ]; then echo "wdt: selftest FAILED: $2 vs $1 gave $got"; set re_bad=yes; fi
}

function tco_selftest {
  # Proves THIS GRUB's regexp/test behave as the predicates assume, before they guard anything.
  set re_bad=no
  tco_re_check "$re_noreboot_clear" "0" yes
  tco_re_check "$re_noreboot_clear" "fed1f401" yes
  tco_re_check "$re_noreboot_clear" "20" no
  tco_re_check "$re_noreboot_clear" "fed1f421" no
  tco_re_check "$re_hlt_clear" "0" yes
  tco_re_check "$re_hlt_clear" "1200" yes
  tco_re_check "$re_hlt_clear" "800" no
  tco_re_check "$re_hlt_clear" "1a00" no
  tco_re_check "$re_lock_clear" "800" yes
  tco_re_check "$re_lock_clear" "2000" yes
  tco_re_check "$re_lock_clear" "1000" no
  tco_re_check "$re_lock_clear" "3800" no
  tco_re_check "$re_tcoen_clear" "21" yes
  tco_re_check "$re_tcoen_clear" "1021" yes
  tco_re_check "$re_tcoen_clear" "2021" no
  tco_re_check "$re_tcoen_clear" "f021" no
  tco_re_check "$re_hex16" "1000" yes
  tco_re_check "$re_hex16" "unset" no
  tco_re_check "$re_hex16" "11000" no
  tco_re_check '^5$' "5" yes
  tco_re_check '^5$' "4005" no
  if [ "a" != "a" -o "b" != "c" ]; then set got=yes; else set got=no; fi
  if [ "$got" != "yes" ]; then echo "wdt: selftest FAILED: test -o"; set re_bad=yes; fi
  if [ "$re_bad" = "no" ]; then echo "wdt: selftest OK (21 regexp cases + test -o)"; fi
}

function tco_arm {
  set wdt_ok=no
  set tco_mode="$lock_mode"
  set lock_mode=refuse
  tco_selftest
  if [ "$re_bad" != "no" ]; then echo "wdt: GRUB predicates untrustworthy, NOT arming"; return 1; fi
  set lpcid=unset; set pmbase=unset; set rcba=unset
  setpci -s 00:1f.0 -v lpcid 0.l
  setpci -s 00:1f.0 -v pmbase 40.l
  setpci -s 00:1f.0 -v rcba f0.l
  echo "wdt: lpcid=$lpcid pmbase=$pmbase rcba=$rcba"
  if [ "$lpcid" != "$lpc_expect" -o "$pmbase" != "$pmbase_expect" -o "$rcba" != "$rcba_expect" ]; then
    echo "wdt: chipset mismatch, NOT arming"; return 1
  fi
  set cntpre=unset; set sts2pre=unset; set gcspre=unset
  inw -v cntpre $tco1_cnt
  inw -v sts2pre $tco2_sts
  read_dword -v gcspre $gcs_addr
  echo "wdt: before: GCS=$gcspre TCO1_CNT=$cntpre TCO2_STS=$sts2pre (6 = last reset was the watchdog)"
  if regexp "$re_hex16" "$cntpre"; then true; else echo "wdt: TCO1_CNT read invalid ($cntpre), NOT arming"; return 1; fi
  if regexp "$re_lock_clear" "$cntpre"; then set tco_locked=no; else set tco_locked=yes; fi
  if [ "$tco_locked" = "yes" ]; then
    if [ "$tco_mode" != "accept" ]; then echo "wdt: TCO_LOCK set, NOT arming (lock_mode=$tco_mode)"; return 1; fi
    echo "wdt: TCO_LOCK set, lock_mode=accept: TCO_EN left as read, 1st timeout goes to firmware SMI, reset on 2nd"
  fi
  write_dword $gcs_addr 0 0x20       # GCS: clear NO_REBOOT (bit 5), masked RMW
  if [ "$tco_locked" = "no" ]; then
    outl $smi_en 0 0x2000            # SMI_EN: clear TCO_EN (bit 13), masked RMW; locked: never written
  fi
  outw $tco1_sts 0x0008              # TCO1_STS: clear TIMEOUT (RW1C)
  outw $tco2_sts 0x0002              # TCO2_STS: clear SECOND_TO_STS first ...
  outw $tco2_sts 0x0004              # ... then BOOT_STS (Intel order)
  outw $tco_tmr $tmr_val 0x03ff      # TCO_TMR: ticks of 0.6 s in bits 9:0 only
  outw $tco_rld 1                    # TCO_RLD: reload
  outw $tco1_cnt 0 0x0800            # TCO1_CNT: clear TCO_TMR_HLT (bit 11), masked RMW; bit 12 TCO_LOCK written back as read
  set gcs=unset; set cnt=unset; set tmr=unset; set smien=unset; set c0=unset; set c1=unset
  read_dword -v gcs $gcs_addr
  inw -v cnt $tco1_cnt
  inw -v tmr $tco_tmr
  inl -v smien $smi_en
  inw -v c0 $tco_rld
  sleep 1
  inw -v c1 $tco_rld
  echo "wdt: after: GCS=$gcs TCO1_CNT=$cnt TCO_TMR=$tmr SMI_EN=$smien count $c0 -> $c1 locked=$tco_locked mode=$tco_mode"
  if regexp "$re_noreboot_clear" "$gcs"; then true; else echo "wdt: NO_REBOOT still set"; return 1; fi
  if regexp "$re_hlt_clear" "$cnt"; then true; else echo "wdt: timer still halted"; return 1; fi
  if [ "$tco_locked" = "no" ]; then
    if regexp "$re_tcoen_clear" "$smien"; then true; else echo "wdt: SMI_EN.TCO_EN still set"; return 1; fi
  fi
  if regexp "$tmr_re" "$tmr"; then true; else echo "wdt: TCO_TMR not $tmr_val"; return 1; fi
  if [ "$c0" = "$c1" -o "$c0" = "unset" -o "$c1" = "unset" ]; then echo "wdt: counter not moving"; return 1; fi
  set wdt_ok=yes
}

pub fn probe(prefer_mmio: bool) -> Result<Rtl, &'static str> {
    if let Some(b) = pci::find(0x10EC, 0x8136) {
        return probe_8101e(b);
    }
    if let Some(b) = pci::find(0x10EC, 0x8139) {
        return probe_8139cp(b, prefer_mmio);
    }
    Err("no 10ec:8136 or 10ec:8139 found (bridges walked)")
}

fn probe_8139cp(b: Bdf, mmio: bool) -> Result<Rtl, &'static str> {
    let bus = if mmio {
        set_pci_cmd(b, 0, PCI_CMD_BM);
        let m = mmio_window(b)?;
        set_pci_cmd(b, PCI_CMD_MEM, PCI_CMD_BM);
        Bus::Mmio(m)
    } else {
        set_pci_cmd(b, PCI_CMD_IO, PCI_CMD_BM);
        let bar0 = pci::read32(b, 0x10);
        if bar0 & 1 == 0 {
            return Err("rtl8139 BAR0 is not I/O");
        }
        Bus::Pio((bar0 & !0x3) as u16)
    };
    let mut mac = [0u8; 6];
    for (i, m) in mac.iter_mut().enumerate() {
        *m = unsafe { bus.r8(R::RTL8139CP.mac0 + i as u16) };
    }
    Ok(Rtl { bus, regs: &R::RTL8139CP, bdf: b, mac, rx_next: 0, rx_cfg: 0x0000_E70A, ver: 0 })
}

fn probe_8101e(b: Bdf) -> Result<Rtl, &'static str> {
    set_pci_cmd(b, 0, PCI_CMD_BM | PCI_CMD_IO); // stop any DMA left running by iPXE before anything else
    let m = mmio_window(b)?;
    set_pci_cmd(b, PCI_CMD_MEM, PCI_CMD_BM | PCI_CMD_IO);
    let bus = Bus::Mmio(m);
    let r = &R::RTL8101E;
    let txc = unsafe { bus.r32(r.tx_config) };
    if txc == 0xFFFF_FFFF {
        return Err("TxConfig reads all-ones (PCI read failed, r8169_main.c:5419)");
    }
    let xid = (txc >> 20) & 0xFCF; // r8169_main.c:5422
    let ver = match xid_version(xid) {
        Some(v) => v,
        None => {
            kprintln!("nic: 10ec:8136 XID {:#05x} is not RTL8101E/8102E/8103E/8105E (VER_07..10, 29, 30): refusing", xid);
            return Err("unsupported 10ec:8136 chip version");
        }
    };
    kprintln!("nic: 10ec:8136 XID {:#05x} -> r8169 RTL_GIGA_MAC_VER_{:02}", xid, ver);
    // rtl_init_rxcfg (r8169_main.c:2486-2490 and default 2510): VER_10 FIFO_THRESH|DMA_BURST, VER_07..09 default.
    let rx_cfg = if ver == 10 { R::RX_FIFO_THRESH | R::RX_DMA_BURST } else { R::RX128_INT_EN | R::RX_DMA_BURST };
    let mut mac = [0u8; 6];
    for (i, x) in mac.iter_mut().enumerate() {
        *x = unsafe { bus.r8(r.mac0 + i as u16) };
    }
    Ok(Rtl { bus, regs: r, bdf: b, mac, rx_next: 0, rx_cfg: rx_cfg | R::ACCEPT_BROADCAST | R::ACCEPT_MY_PHYS, ver })
}

impl Rtl {
    pub fn name(&self) -> &'static str {
        self.regs.name
    }

    fn mdio_write(&self, reg: u32, val: u16) -> Result<(), &'static str> {
        // r8169_mdio_write, r8169_main.c:1154-1164: flag set, wait for it to clear (25 x 20 us), then 20 us.
        unsafe { self.bus.w32(self.regs.phyar, R::PHYAR_FLAG | (reg & 0x1F) << 16 | val as u32) };
        for _ in 0..25 {
            time::delay_us(20);
            if unsafe { self.bus.r32(self.regs.phyar) } & R::PHYAR_FLAG == 0 {
                time::delay_us(20);
                return Ok(());
            }
        }
        Err("PHYAR write timeout")
    }

    fn mdio_read(&self, reg: u32) -> Result<u16, &'static str> {
        // r8169_mdio_read, r8169_main.c:1166-1180: flag clear, wait for it to SET, data in 15:0, then 20 us.
        unsafe { self.bus.w32(self.regs.phyar, (reg & 0x1F) << 16) };
        for _ in 0..25 {
            time::delay_us(20);
            let v = unsafe { self.bus.r32(self.regs.phyar) };
            if v & R::PHYAR_FLAG != 0 {
                time::delay_us(20);
                return Ok(v as u16);
            }
        }
        Err("PHYAR read timeout")
    }

    pub fn skip_8105e(&mut self) {
        if self.ver == 29 || self.ver == 30 {
            kprintln!("nic: nic=no8105e: skipping rtl_hw_start_8105e (generic 810x path only)");
            self.ver += 100;
        }
    }

    /// rtl_ephy_write / rtl_ephy_read (r8169_main.c:1261-1277): EPHYAR 0x80, flag bit 31, reg 20:16, data 15:0.
    fn ephy_write(&self, reg: u32, val: u16) -> Result<(), &'static str> {
        unsafe { self.bus.w32(0x80, 0x8000_0000 | val as u32 | (reg & 0x1F) << 16) };
        for _ in 0..100 {
            time::delay_us(10);
            if unsafe { self.bus.r32(0x80) } & 0x8000_0000 == 0 {
                time::delay_us(10);
                return Ok(());
            }
        }
        Err("EPHYAR write timeout")
    }

    fn ephy_read(&self, reg: u32) -> Result<u16, &'static str> {
        unsafe { self.bus.w32(0x80, (reg & 0x1F) << 16) };
        for _ in 0..100 {
            time::delay_us(10);
            let v = unsafe { self.bus.r32(0x80) };
            if v & 0x8000_0000 != 0 {
                return Ok(v as u16);
            }
        }
        Err("EPHYAR read timeout")
    }

    /// rtl_hw_start_8105e_1 / _2 (r8169_main.c:3619-3650), VER_29/30 only. Called with Cfg9346 unlocked.
    fn hw_start_8105e(&self) -> Result<(), &'static str> {
        const E: [(u32, u16, u16); 8] =
            [(0x07, 0, 0x4000), (0x19, 0, 0x0200), (0x19, 0, 0x0020), (0x1e, 0, 0x2000), (0x03, 0, 0x0001), (0x19, 0, 0x0100), (0x19, 0, 0x0004), (0x0a, 0, 0x0020)];
        unsafe {
            let fe = self.bus.r32(0xF0); // FuncEvent: force LAN exit from ASPM if Rx/Tx not idle; no early tally
            self.bus.w32(0xF0, (fe | 0x0000_2800) & !0x0001_0000);
            self.bus.w8(0xD3, self.bus.r8(0xD3) | (1 << 3) | (1 << 2)); // MCU: EN_NDP | EN_OOB_RESET
            self.bus.w8(0xD0, self.bus.r8(0xD0) | (1 << 6)); // DLLPR: PFM_EN
        }
        for (reg, mask, bits) in E {
            let w = (self.ephy_read(reg)? & !mask) | bits;
            self.ephy_write(reg, w)?;
        }
        unsafe { self.bus.w8(0x54, self.bus.r8(0x54) & !(1 << 1)) }; // Config3: clear Rdy_to_L23 (l2l3 disable)
        if self.ver == 30 {
            let w = self.ephy_read(0x1e)? | 0x8000;
            self.ephy_write(0x1e, w)?;
        }
        kprintln!("nic: rtl_hw_start_8105e_{} applied (FuncEvent, MCU, DLLPR, 8 ephy writes, L2/L3 off)", self.ver - 28);
        Ok(())
    }

    fn reset(&self) -> Result<(), &'static str> {
        let r = self.regs;
        unsafe {

    fn program_rings(&self) {
        let r = self.regs;
        unsafe {
            for i in 0..NRX {
                let d = addr_of_mut!(RX_DESC.0[i]);
                let mut o = r.desc_own | BUFSZ as u32;
                if i == NRX - 1 {
                    o |= r.ring_end;
                }
                write_volatile(d, Desc { opts1: o, opts2: 0, addr: addr_of!(RX_BUF.0[i]) as u64 });
            }
            write_volatile(addr_of_mut!(TX_DESC.0[0]), Desc { opts1: r.ring_end, opts2: 0, addr: 0 });
            fence(Ordering::SeqCst);
            let tx = addr_of!(TX_DESC) as u64;
            let rx = addr_of!(RX_DESC) as u64;
            // high dword first (r8169_main.c:2758-2767 "magic spell")
            self.bus.w32(r.tx_ring + 4, (tx >> 32) as u32);
            self.bus.w32(r.tx_ring, tx as u32);
            self.bus.w32(r.rx_ring + 4, (rx >> 32) as u32);
            self.bus.w32(r.rx_ring, rx as u32);
        }
    }

/// ACPI PM timer (PMBASE+0x08, 3.579545 MHz, 24-bit) based delay; same block on ICH9 and NM10.
fn pm_timer() -> u32 {
    unsafe { inl(pmbase() + 8) & 0x00FF_FFFF }
}

pub fn delay_ms(ms: u32) {
    let need = ms as u64 * 3_579_545 / 1000;
    let mut last = pm_timer();
    let mut acc = 0u64;
    while acc < need {
        let now = pm_timer();
        acc += (now.wrapping_sub(last) & 0x00FF_FFFF) as u64;
        last = now;
    }
}
