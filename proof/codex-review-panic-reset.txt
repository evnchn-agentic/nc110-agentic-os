fn vga_putc(b: u8) {
    unsafe {
        if b == b'\n' {
            COL = 0;
            ROW += 1;
        } else {
            cell(ROW * cols() + COL, ATTR | b as u16);
            COL += 1;
            if COL == cols() {
                COL = 0;
                ROW += 1;
            }
        }
        if ROW == rows() {
            for i in 0..cols() * (rows() - 1) {
                cell(i, cell_get(i + cols()));
            }
            for i in 0..cols() {
                cell((rows() - 1) * cols() + i, ATTR | b' ' as u16);
            }
            ROW = rows() - 1;
        }
    }
}

pub struct Console;
impl fmt::Write for Console {
    fn write_str(&mut self, s: &str) -> fmt::Result {
        if unsafe { AT_LINE_START } && crate::time::ready() && !s.is_empty() && !s.starts_with(['\r', '\n']) {
            unsafe { AT_LINE_START = false };
            let t = crate::time::now_ms();
            write!(self, "[{:3}.{:03}] ", t / 1000, t % 1000)?;
        }
        for &b in s.as_bytes() {
            if b == b'\n' {
                serial_putc(b'\r');
            }
            serial_putc(b);
            unsafe {
                if CAP_LEN < 8192 {
                    (*core::ptr::addr_of_mut!(CAP))[CAP_LEN] = b;
                    CAP_LEN += 1;
                }
            }
            if unsafe { VGA_LOG } {
                vga_putc(b);
            }
            if b == b'\n' {
                unsafe { AT_LINE_START = true };
            }
        }
        Ok(())
    }
}

//! PCI config mechanism #1 (0xCF8/0xCFC), with a bridge walk (M2.5: the NC110's RTL8101E is expected to sit
//! behind a PCIe root port, i.e. on a secondary bus). The walk only READS config space. The one config write
//! in this kernel goes to the matched NIC function (command register), see nic.rs.
use crate::port::{inl, outl};

#[derive(Clone, Copy, PartialEq, Eq)]
pub struct Bdf {
    pub bus: u8,
    pub dev: u8,
    pub func: u8,
}

impl core::fmt::Display for Bdf {
    fn fmt(&self, f: &mut core::fmt::Formatter) -> core::fmt::Result {
        write!(f, "{:02x}:{:02x}.{}", self.bus, self.dev, self.func)
    }
}

fn addr(b: Bdf, off: u8) -> u32 {
    0x8000_0000 | (b.bus as u32) << 16 | (b.dev as u32) << 11 | (b.func as u32) << 8 | (off as u32 & 0xFC)
}

pub fn read32(b: Bdf, off: u8) -> u32 {
    unsafe {
        outl(0xCF8, addr(b, off));
        inl(0xCFC)
    }
}

pub fn write32(b: Bdf, off: u8, v: u32) {
    unsafe {
        outl(0xCF8, addr(b, off));
        outl(0xCFC, v);
    }
}

/// Visit every present function: bus 0 first, then the secondary bus of every type-1 (bridge) header,
/// depth first. `visit` returns true to stop. Bounded: at most 32 buses visited, depth 8.
pub fn walk(visit: &mut dyn FnMut(Bdf, u32) -> bool) -> bool {
    let mut seen = [false; 256];
    let mut budget = 32;
    walk_bus(0, 0, &mut seen, &mut budget, visit)
}

fn walk_bus(bus: u8, depth: u32, seen: &mut [bool; 256], budget: &mut u32, visit: &mut dyn FnMut(Bdf, u32) -> bool) -> bool {
    if seen[bus as usize] || depth > 8 || *budget == 0 {
        return false;
    }
    seen[bus as usize] = true;
    *budget -= 1;
    for dev in 0..32u8 {
        let f0 = Bdf { bus, dev, func: 0 };
        if read32(f0, 0) & 0xFFFF == 0xFFFF {
            continue;
        }
        let multi = (read32(f0, 0x0C) >> 16) & 0x80 != 0;
        for func in 0..if multi { 8u8 } else { 1 } {
            let b = Bdf { bus, dev, func };
            let id = read32(b, 0);
            if id & 0xFFFF == 0xFFFF {
                continue;
            }
            if visit(b, id) {
                return true;
            }
            let htype = (read32(b, 0x0C) >> 16) & 0x7F;
            if htype == 1 {
                let sec = ((read32(b, 0x18) >> 8) & 0xFF) as u8;
                if sec != 0 && sec != bus && walk_bus(sec, depth + 1, seen, budget, visit) {
                    return true;
                }
            }
        }
    }
    false
}

/// First function matching vendor:device anywhere in the walked hierarchy.
pub fn find(vendor: u16, device: u16) -> Option<Bdf> {
    let want = (device as u32) << 16 | vendor as u32;
    let mut hit = None;
    walk(&mut |b, id| {
        if id == want {
            hit = Some(b);
            true
        } else {
            false
        }
    });
    hit
}

/// (base, is_64bit) of the first MEMORY BAR of a type-0 function (mirrors r8169's "first MMIO region",
/// r8169_main.c:5408). Read-only: no BAR sizing writes.
pub fn first_mem_bar(b: Bdf) -> Option<(u64, bool)> {
    let mut i = 0u8;
    while i < 6 {
        let off = 0x10 + 4 * i;
        let lo = read32(b, off);
        if lo != 0 && lo & 1 == 0 { // 0 = unimplemented slot (RTL8102E BAR1), not a memory BAR at 0
            let ty = (lo >> 1) & 3;
            let base = (lo & !0xF) as u64;
            if ty == 2 {
                if i == 5 {
                    return None;
                }
                let hi = read32(b, off + 4) as u64;
                return Some((base | hi << 32, true));
            }
            return Some((base, false));
        }
        i += 1;
    }
    None
}

/// Memory BAR bases of every mass-storage function (class 0x01: IDE/SATA/AHCI). Read-only; used so the NIC's
/// MMIO window is refused if it is anywhere near the disk controller's.
pub fn storage_mem_bars(out: &mut [u64; 16]) -> usize {
    let mut n = 0;
    walk(&mut |b, _| {
        let class = read32(b, 0x08) >> 24;
        let htype = (read32(b, 0x0C) >> 16) & 0x7F;
        if class == 0x01 && htype == 0 {
            let mut i = 0u8;
            while i < 6 {
                let lo = read32(b, 0x10 + 4 * i);
                if lo & 1 == 0 {
                    let mut base = (lo & !0xF) as u64;
                    if (lo >> 1) & 3 == 2 && i < 5 {
                        base |= (read32(b, 0x10 + 4 * (i + 1)) as u64) << 32;
                        i += 1;
                    }
                    if base != 0 && n < out.len() {
                        out[n] = base;
                        n += 1;
                    }
                }
                i += 1;
            }
        }
        false
    });
    n
}

//! Monotonic milliseconds from the TSC, calibrated once against PIT channel 2 (gate via port 0x61).
use crate::port::{inb, outb, rdtsc};

static mut TSC_PER_MS: u64 = 1_000_000; // fallback guess (1 GHz) if calibration fails
static mut T0: u64 = 0;

pub fn init() -> u64 {
    unsafe {
        // PIT ch2, mode 0, lobyte/hibyte, count for 10 ms at 1.193182 MHz.
        const COUNT: u16 = 11932;
        let g = inb(0x61);
        outb(0x61, (g & 0xFD) | 0x01); // gate on, speaker off
        outb(0x43, 0b1011_0000);
        outb(0x42, COUNT as u8);
        outb(0x42, (COUNT >> 8) as u8);
        // Re-trigger: toggle gate low->high so mode 0 starts counting now.
        let g = inb(0x61);
        outb(0x61, g & 0xFE);
        outb(0x61, g | 0x01);
        let a = rdtsc();
        let mut spins: u64 = 0;
        while inb(0x61) & 0x20 == 0 && spins < 200_000_000 {
            spins += 1;
        }
        let b = rdtsc();
        if spins < 200_000_000 && b > a {
            TSC_PER_MS = ((b - a) / 10).max(1);
        }
        T0 = rdtsc();
        TSC_PER_MS
    }
}

/// ms the TSC had counted when `init` ran: time since the last CPU reset (firmware + GRUB + our early boot),
/// since a hard reset zeroes the TSC; after an in-RAM chain-load it also includes the previous kernel's run.
pub fn since_reset_ms() -> u64 {
    unsafe { T0 / TSC_PER_MS }
}

pub fn tsc_to_ms(t: u64) -> u64 {
    unsafe { t / TSC_PER_MS }
}

pub fn ready() -> bool {
    unsafe { T0 != 0 }
}

pub fn now_ms() -> i64 {
    unsafe { ((rdtsc() - T0) / TSC_PER_MS) as i64 }
}

/// Busy-wait at least `us` microseconds (TSC, after init()).
pub fn delay_us(us: u64) {
    unsafe {
        let t = rdtsc();
        let n = us * TSC_PER_MS / 1000 + 1;
        while rdtsc().wrapping_sub(t) < n {
            core::hint::spin_loop();
        }
    }
}

/// CMOS RTC hours and minutes (the NC110's RTC keeps local time: Linux there labels HKT as UTC). Waits out an update in
/// progress (bounded), handles BCD and 12-hour formats. None if the RTC never settles.
pub fn rtc_hm() -> Option<(u8, u8)> {
    use crate::port::{inb, outb};
    let rd = |r: u8| unsafe {
        outb(0x70, r);
        inb(0x71)
    };
    for _ in 0..1000 {
        if rd(0x0A) & 0x80 == 0 {
            let (m, h, b) = (rd(0x02), rd(0x04), rd(0x0B));
            let bcd = |v: u8| if b & 0x04 == 0 { (v & 0x0F) + (v >> 4) * 10 } else { v };
            let pm = h & 0x80 != 0;
            let mut hh = bcd(h & 0x7F);
            if b & 0x02 == 0 {
                hh = hh % 12 + if pm { 12 } else { 0 };
            }
            return Some((hh, bcd(m)));
        }
    }
    None
}

**MUST-FIX: none.**

**SHOULD**

1. **Bound the panic delay independently of the PM timer.** [main.rs:205](~/nc110-baremetal/src/main.rs:205) does:
   ```rust
   for _ in 0..20 {
       wdt::pet();
       wdt::delay_ms(500);
   }
   wdt::hard_reset();
   ```
   But [wdt.rs:190](~/nc110-baremetal/src/wdt.rs:190) waits without a bound:
   ```rust
   while acc < need {
       let now = pm_timer();
       acc += (now.wrapping_sub(last) & 0x00FF_FFFF) as u64;
       last = now;
   }
   ```
   A frozen or inaccessible PM timer returning a constant leaves `acc` unchanged forever. **This does not pet forever:** it stalls inside one delay, so the armed TCO eventually expires. However, that returns you to the reset mechanism observed to leave this board dark. Before watchdog arming, it could hang indefinitely. A panic-specific polling bound that falls through to CF9 would close this gap. This is conditional hardening, not evidence that the NC110’s PM timer currently fails.

2. **The QEMU test can falsely certify a reset.** [panic-q35-test.py:28](~/nc110-baremetal/tools/panic-q35-test.py:28):
   ```python
   ok = "PANIC" in log() and gone
   ```
   Any QEMU exit after that text passes—including a QEMU error or a triple-fault reset immediately after printing. There is no lower timing bound, successful-exit check, or reset-cause check. Require the expected panic text, successful exit, and a reasonable timing window; use QEMU reset tracing to distinguish CF9 from other reset causes. Printing “exited (reset)” currently overstates the evidence.

**NIT**

Use the same raw `0x02; 0x06` sequence in the nested path, preferably through a logging-free reset helper. This removes dependence on the previous `RST_CPU` bit and keeps the two paths consistent. **I would not characterize the existing `0x06` as invalid.**

The specific concerns:

- **Infinite loop while petting? No ordinary control-flow path.** There are at most 20 pets. Neither `delay_ms()` nor the console pets internally. `PANICKING.swap(true, …)` makes another Rust panic bypass formatting and the countdown. A stall can prevent CF9, but cannot sustain watchdog feeding.

- **Is `0x06` alone a hard reset? Yes, provided `RST_CPU` was zero.** It sets `SYS_RST=1` and raises `RST_CPU`; `FULL_RST=0` requests a hard reset without power cycling. Both Intel’s [ICH7 datasheet, §10.7.5](https://studylib.net/doc/18311038/intel%C2%AE-i-o-controller-hub-7--ich7--family-external-design) and [NM10 datasheet, §13.7.5, p.404](https://www.mouser.com/pdfdocs/82NM10datasheet.pdf#page=404) describe that edge-triggered behavior and default register value zero. `0x02` first guarantees the low-to-high edge. I found no reachable preceding kernel write that leaves bit 2 high and then enters this nested-panic path.

- **Can `pet()` or `delay_ms(500)` panic? No evident arithmetic or indexing panic.** `pet()` is `outw(base(), 1)`. PMBASE is masked to `0xFF80`, so adding `0x60` or `8` fits `u16`. The delay’s `u64` arithmetic is safely bounded for 500 ms; subtraction explicitly wraps. The credible issue is the unbounded hardware poll above. A hardware exception is not necessarily a Rust panic and is not covered by `PANICKING`.

- **Crash-at-boot reboot loop? Yes, intentionally faster recovery.** The [GRUB entry generator](~/nc110-baremetal/tools/nc110-install-agentos.sh:31) appends `"x"` and accepts only `'^x{1,5}$'`, otherwise calling `nc110_linux_now2`. With the stated deployed recovery setup, five failed kernel boots followed by Linux is preferable to a board that never reaches POST. The ten-second display window also preserves some diagnostic opportunity.

- **Production hook reachable? Yes, through the boot command line.** `wdt::rehearse(cmdline)` is unconditional, and `arg == "panic"` deliberately panics before its `armed()` check. An accidentally deployed `wdt=panic` causes repeated resets until recovery. The production entry currently passes `bmctl grubtsc=…`, not this option; this is not a network-exposed trigger. Existing rehearsal modes already permit command-line-induced hangs.
