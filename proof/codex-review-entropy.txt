   101	/// TSC granularity: smallest nonzero step between back-to-back reads (QEMU/TCG: 1000; real CPUs: tens).
   102	/// None = the timer never moved (broken/constant): no jitter source at all.
   103	fn tsc_granularity() -> Option<u64> {
   104	    let mut g = u64::MAX;
   105	    let mut last = tsc();
   106	    for _ in 0..100_000 {
   107	        let t = tsc();
   108	        let d = t.wrapping_sub(last);
   109	        if d != 0 && d < g {
   110	            g = d;
   111	        }
   112	        last = t;
   113	    }
   114	    if g == u64::MAX { None } else { Some(g) }
   115	}

   117	/// One raw sample = how many walk steps (strided over 2 MiB) complete while the TSC advances by WINDOW
   118	/// ticks (max(64 x granularity, 32768)), starting on a TSC edge. It races two clocks: execution speed (cache, bus, and in
   119	/// QEMU the host's scheduling/caches) against the timer. A first attempt timed a FIXED walk instead; under
   120	/// TCG the TSC moves in 1000-tick steps, so the delta was nearly constant and both the RCT and the in-kernel
   121	/// estimate refused (proof/m25-short-walk-serial.log, proof/m25-fixedwalk-serial.log).
   122	/// If no TSC edge is seen the raw reading is 0 (a dead timer reads as a constant, so the RCT fires).
   123	fn raw_delta() -> u64 {
   124	    let w = unsafe { WINDOW };
   125	    let t0 = tsc();
   126	    let mut start = t0;
   127	    let mut spins = 0u32;
   128	    while start == t0 {
   129	        start = tsc();
   130	        spins += 1;
   131	        if spins > 1 << 16 {
   132	            return 0;
   133	        }
   134	    }
   135	    let mut pos = unsafe { WALK_POS };
   136	    let mut n = 0u64;
   137	    while tsc().wrapping_sub(start) < w && n < 1 << 20 {
   138	        touch(&mut pos);
   139	        n += 1;
   140	    }
   141	    unsafe { WALK_POS = pos };
   142	    n
   143	}

   220	fn sqrt(x: f64) -> f64 {
   221	    if x <= 0.0 {
   222	        return 0.0;
   223	    }
   224	    let mut g = if x > 1.0 { x } else { 1.0 };
   225	    for _ in 0..60 {
   226	        g = 0.5 * (g + x / g);
   227	    }
   228	    g
   229	}

   231	fn log2(x: f64) -> f64 {
   232	    let b = x.to_bits();
   233	    let e = ((b >> 52) & 0x7ff) as i64 - 1023;
   234	    let m = f64::from_bits((b & 0x000f_ffff_ffff_ffff) | (1023u64 << 52)); // [1, 2)
   235	    let y = (m - 1.0) / (m + 1.0);
   236	    let (mut s, mut t) = (0.0, y);
   237	    let mut k = 1.0;
   238	    while k < 60.0 {
   239	        s += t / k;
   240	        t *= y * y;
   241	        k += 2.0;
   242	    }
   243	    e as f64 + 2.0 * s / core::f64::consts::LN_2
   244	}

   248	/// Lag-1 bound: max over previous values a (>= 20 successors) of the 99% upper bound on max_b P(b | a).
   249	/// Not a 90B estimator; a cheap guard against the serial correlation MCV (an IID estimate) cannot see.
   250	fn lag1_milli(samples: &[u8]) -> u32 {
   251	    let t = unsafe { &mut *core::ptr::addr_of_mut!(TRANS) };
   252	    for row in t.iter_mut() {
   253	        row.fill(0);
   254	    }
   255	    for w in samples.windows(2) {
   256	        t[w[0] as usize][w[1] as usize] += 1;
   257	    }
   258	    let mut worst = 0.0f64;
   259	    for row in t.iter() {
   260	        let n: u32 = row.iter().map(|&x| x as u32).sum();
   261	        if n >= 20 {
   262	            let p = *row.iter().max().unwrap() as f64 / n as f64;
   263	            let pu = (p + 2.576 * sqrt(p * (1.0 - p) / (n as f64 - 1.0))).min(1.0);
   264	            if pu > worst {
   265	                worst = pu;
   266	            }
   267	        }
   268	    }
   269	    if worst == 0.0 {
   270	        return 8000;
   271	    }
   272	    (-log2(worst) * 1000.0) as u32
   273	}

   275	/// SP 800-90B 6.3.1 most-common-value estimate, in milli-bits per 8-bit sample.
   276	fn mcv_milli(samples: &[u8]) -> (u32, u32, u32) {
   277	    let mut c = [0u32; 256];
   278	    for &s in samples {
   279	        c[s as usize] += 1;
   280	    }
   281	    let max = *c.iter().max().unwrap();
   282	    let distinct = c.iter().filter(|&&x| x > 0).count() as u32;
   283	    let n = samples.len() as f64;
   284	    let p = max as f64 / n;
   285	    let pu = (p + 2.576 * sqrt(p * (1.0 - p) / (n - 1.0))).min(1.0);
   286	    ((-log2(pu) * 1000.0) as u32, max, distinct)
   287	}

   311	/// Startup: health-test STARTUP_N samples, estimate min-entropy, seed the key. Err = do not run SSH.
   312	pub fn init(dump: bool) -> Result<(), ()> {
   313	    unsafe {
   314	        if SEEDED {
   315	            return Ok(());
   316	        }
   317	        if FAILED {
   318	            return Err(());
   319	        }
   320	    }
   321	    // A dead timer is not refused here: the samples then all read 0 and the health tests (the one authority
   322	    // on the source) must refuse. That is the path the `badjitter` negative control exercises.
   323	    let g = match tsc_granularity() {
   324	        Some(g) => g,
   325	        None => {
   326	            kprintln!("rng: TSC never advanced in 100000 reads (dead timer?)");
   327	            1
   328	        }
   329	    };
   330	    // 64 timer steps, but never under 32768 ticks (~20 us on the N570): on a cycle-exact TSC, 64 x granularity
   331	    // would be a window of a few DRAM accesses and the count close to constant.
   332	    let w = (64 * g).max(32768);
   333	    unsafe { WINDOW = w };
   334	    kprintln!("rng: TSC granularity {} ticks, sample window {} ticks", g, w);
   335	    if dump {
   336	        dump_raw();
   337	    }
   338	    let mut h = Sha256::new();
   339	    h.update(b"nc110-m2.5 seed");
   340	    let t0 = rdtsc();
   341	    if let Err(e) = gather(&mut h, STARTUP_N, true) {
   342	        kprintln!("rng: jitter startup health test FAILED ({}) -> refusing to run SSH", e);
   343	        return Err(());
   344	    }
   345	    let cycles = rdtsc().wrapping_sub(t0);
   346	    let low = unsafe { &*core::ptr::addr_of!(LOWB) };
   347	    let (hm, max, distinct) = mcv_milli(low);
   348	    let hl = lag1_milli(low);
   349	    kprintln!(
   350	        "rng: jitter startup {} samples passed RCT(cut {}) + APT(W {}, cut {}); {} TSC ticks/sample",
   351	        STARTUP_N, RCT_CUTOFF, APT_W, APT_CUTOFF, cycles / STARTUP_N as u64
   352	    );
   353	    kprintln!(
   354	        "rng: min-entropy per 8-bit sample: MCV {}.{:03} (max {}/{}, {} distinct), lag-1 {}.{:03}; claimed {}.{:03}",
   355	        hm / 1000, hm % 1000, max, STARTUP_N, distinct, hl / 1000, hl % 1000, H_CLAIM_MILLI / 1000, H_CLAIM_MILLI % 1000
   356	    );
   357	    if hm.min(hl) < H_CLAIM_MILLI {
   358	        unsafe { FAILED = true };
   359	        kprintln!("rng: estimate below the claimed 0.1 bit/sample -> refusing to run SSH");
   360	        return Err(());
   361	    }
   362	    if let Err(e) = gather(&mut h, BLOCK_N - STARTUP_N, false) {
   363	        kprintln!("rng: jitter health test FAILED while seeding ({}) -> refusing to run SSH", e);
   364	        return Err(());
   365	    }
   366	    if has_rdrand() {
   367	        match rdrand32() {
   368	            Ok(r) => {
   369	                h.update(r);
   370	                unsafe { USE_RDRAND = true };
   371	                kprintln!("rng: RDRAND present, mixed in (jitter alone already passed)");
   372	            }
   373	            Err(()) => {
   374	                unsafe { FAILED = true };
   375	                kprintln!("rng: RDRAND present but failing -> refusing to run SSH");
   376	                return Err(());
   377	            }
   378	        }
   379	    } else {
   380	        kprintln!("rng: no RDRAND (as on the Atom N570): jitter is the only source");
   381	    }
   382	    unsafe {
   383	        KEY = h.finalize().into();
   384	        SEEDED = true;
   385	    }
   386	    Ok(())
   387	}

   389	/// Fill `buf` from the ratchet. Errors if not seeded, if a health test ever failed, or if a reseed fails.
   390	pub fn fill(buf: &mut [u8]) -> Result<(), ()> {
   391	    unsafe {
   392	        if !SEEDED || FAILED {
   393	            return Err(());
   394	        }
   395	        CALLS += 1;
   396	        if CALLS % RESEED_EVERY == 0 {
   397	            let mut h = Sha256::new();
   398	            h.update(b"reseed");
   399	            h.update(KEY);
   400	            if gather(&mut h, BLOCK_N, false).is_err() {
   401	                return Err(());
   402	            }
   403	            RESEEDS += 1;
   404	            KEY = h.finalize().into();
   405	        }
   406	        if USE_RDRAND {
   407	            let r = rdrand32().map_err(|_| FAILED = true)?;
   408	            let mut h = Sha256::new();
   409	            h.update(b"rdrand");
   410	            h.update(KEY);
   411	            h.update(r);
   412	            KEY = h.finalize().into();
   413	        }
   414	        for chunk in buf.chunks_mut(32) {
   415	            let mut h = Sha256::new();
   416	            h.update(b"out");
   417	            h.update(KEY);
   418	            h.update(CTR.to_le_bytes());
   419	            CTR += 1;
   420	            chunk.copy_from_slice(&h.finalize()[..chunk.len()]);
   421	        }
   422	        let mut h = Sha256::new();
   423	        h.update(b"next");
   424	        h.update(KEY);
   425	        KEY = h.finalize().into();
   426	    }
   427	    Ok(())
   428	}

   107	def selftest():
   108	    """Mutation check: a source with no noise must score ~0, real randomness ~8, or the estimator is broken."""
   109	    n = 200_000
   110	    cases = [
   111	        ("constant timer (raw_delta returns the same count)", bytes([0x5a]) * n, lambda h: h < 0.05),
   112	        ("TSC frozen (raw_delta returns 0)", bytes(n), lambda h: h < 0.05),
   113	        ("deterministic counter (i mod 256)", bytes(i % 256 for i in range(n)), lambda h: h < 0.5),
   114	        ("two-value alternation", bytes((0x40, 0x41) * (n // 2)), lambda h: h < 0.5),

   119	    ]

   121	    for name, data, ok in cases:
   122	        print(f"== {name}")

   124	        verdict = ok(h)
   125	        print(f"  {'ok  ' if verdict else 'FAIL'} expectation\n")
