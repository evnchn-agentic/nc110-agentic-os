    56	/// RFC 8439 tag over aad and ct with the one-time key from keystream block 0.
    57	fn tag(pk: &[u8; 32], aad: &[u8], ct: &[u8]) -> [u8; 16] {
    58	    let mut m = Poly1305::new(pk.into());
    59	    m.update_padded(aad);
    60	    m.update_padded(ct);
    61	    let mut lens = [0u8; 16];
    62	    lens[..8].copy_from_slice(&(aad.len() as u64).to_le_bytes());
    63	    lens[8..].copy_from_slice(&(ct.len() as u64).to_le_bytes());
    64	    m.update_padded(&lens);
    65	    m.finalize().into()
    66	}

    68	fn cipher(key: &[u8; 32], n: &[u8; 12]) -> (ChaCha20, [u8; 32]) {
    69	    let mut c = ChaCha20::new(key.into(), n.into());
    70	    let mut b0 = [0u8; 64];
    71	    c.apply_keystream(&mut b0); // block 0 -> Poly1305 key; the stream continues at block 1
    72	    let mut pk = [0u8; 32];
    73	    pk.copy_from_slice(&b0[..32]);
    74	    (c, pk)
    75	}

    77	pub fn seal(key: &[u8; 32], n: &[u8; 12], aad: &[u8], buf: &mut [u8]) -> [u8; 16] {
    78	    let (mut c, pk) = cipher(key, n);
    79	    c.apply_keystream(buf);
    80	    tag(&pk, aad, buf)
    81	}

    83	pub fn open(key: &[u8; 32], n: &[u8; 12], aad: &[u8], buf: &mut [u8], t: &[u8]) -> bool {
    84	    let (mut c, pk) = cipher(key, n);
    85	    let want = tag(&pk, aad, buf);
    86	    if t.len() != 16 {
    87	        return false;
    88	    }
    89	    let diff = want.iter().zip(t).fold(0u8, |a, (x, y)| a | (x ^ y));
    90	    if diff != 0 {
    91	        return false;
    92	    }
    93	    c.apply_keystream(buf);
    94	    true
    95	}

   153	/// Allowlisted read. Returns Err(reason) for anything outside the policy; never touches refused addresses.
   154	fn peek(w: &mut Writer, nic_mmio: Option<u64>, what: &str, a: &str, b: &str) -> Result<(), &'static str> {
   155	    let addr = u64::from_str_radix(a.trim_start_matches("0x"), 16).map_err(|_| "addr: hex")?;
   156	    let n: u64 = b.parse().map_err(|_| "len/width: decimal")?;
   157	    match what {
   158	        "mem" => {
   159	            if n == 0 || n > 512 {
   160	                return Err("len 1..512");
   161	            }
   162	            let end = addr.checked_add(n).ok_or("overflow")?;
   163	            let (ks, ke) = (core::ptr::addr_of!(__load_start) as u64, core::ptr::addr_of!(__bss_end) as u64);
   164	            if addr < ke && ks < end {
   165	                return Err("refused: kernel image (code, stack, keys)");
   166	            }
   167	            if end > 0x1_0000_0000 || !crate::mb::ram_contains(addr, n) {
   168	                return Err("refused: not inside one multiboot RAM range below 4 GiB");
   169	            }
   170	            for i in 0..n {
   171	                let v = unsafe { core::ptr::read_volatile((addr + i) as *const u8) };
   172	                let _ = core::fmt::write(w, format_args!("{:02x}", v));
   173	            }
   174	        }
   175	        "nic" => {
   176	            let base = nic_mmio.ok_or("refused: NIC not on MMIO")?;
   177	            if addr % 4 != 0 || n % 4 != 0 || n == 0 || addr.checked_add(n).is_none_or(|e| e > 0x100) {
   178	                return Err("nic: dword-aligned offset/len inside 0..0x100");
   179	            }
   180	            for o in (addr..addr + n).step_by(4) {
   181	                let v = unsafe { core::ptr::read_volatile((base + o) as *const u32) };
   182	                let _ = core::fmt::write(w, format_args!("{:08x} ", v));
   183	            }
   184	        }
   185	        "io" => {
   186	            // Explicit (offset from PMBASE, width) allowlist from the NM10 datasheet (13.8.3, 13.9). Deliberately
   187	            // absent: LV2/LV3/LV4 at +0x14..+0x16, whose READ requests C2/C3/C4 (Codex MUST-FIX), and anything else.
   188	            const OK: [(u64, u64); 16] = [
   189	                (0x00, 2), (0x02, 2), (0x04, 4), (0x08, 4), (0x10, 4), (0x28, 4), (0x2c, 4), (0x30, 4), (0x34, 4),
   190	                (0x60, 2), (0x64, 2), (0x66, 2), (0x68, 2), (0x6a, 2), (0x72, 2), (0x62, 1),
   191	            ];
   192	            let pm = wdt::pmbase() as u64;
   193	            let off = addr.checked_sub(pm).ok_or("io: below PMBASE")?;
   194	            if pm == 0 || !OK.contains(&(off, n)) {
   195	                return Err("io: only allowlisted PM/TCO registers (PMBASE+off,width): 0/2 2/2 4/4 8/4 10/4 28/4 2c/4 30/4 34/4 60/2 62/1 64/2 66/2 68/2 6a/2 72/2");
   196	            }
   197	            let p = u16::try_from(addr).map_err(|_| "io: port > 0xffff")?;
   198	            let v = unsafe { match n { 1 => inb(p) as u32, 2 => inw(p) as u32, _ => inl(p) } };
   199	            let _ = core::fmt::write(w, format_args!("{:#x}", v));
   200	        }
   201	        _ => return Err("peek mem <hexaddr> <len> | peek nic <hexoff> <len> | peek io <hexport> <1|2|4>"),
   202	    }
   203	    Ok(())
   204	}

   206	/// What the caller must do after `Ctl::step`; the caller owns the NIC, so it does the hop/reset itself.
   207	pub enum Act {
   208	    Idle,
   209	    Reboot,
   210	    Sleep(u32),
   300	            Bus::Mmio(b) => Some(b),
   301	            Bus::Pio(_) => None,
   302	        };
   303	        let mut boot_id = [0u8; 16];
   304	        if rng::fill(&mut boot_id).is_err() {
   305	            kprintln!("bmctl: entropy failed -> not starting");
   306	            return None;
   307	        }
   308	        let mut bid = heapless::String::<64>::new();
   309	        hex(&boot_id, &mut bid);
   310	        sock.bind(PORT).expect("bind");
   311	        kprintln!("bmctl: listening on {}:{} boot id {} (idle reset after {} s)", ip, PORT, bid, idle_ms / 1000);
   312	        crate::udplog::dump(5);
   313	        console::status("ready");
   314	        let t = time::now_ms();
   315	        Some(Ctl {
   316	            seed, key: derive_key(&seed), boot_id, bid, ip, idle_ms, nic_mmio, last_ok: t, last_hello: t - 2001, last_footer: t - 1000, battery: None,
   317	            ok: 0, bad: 0, replay: 0, retx: 0, next_sess: 1, who_window: 0, who_count: 0, reboot_at: None, sleep_at: None, stage: None, jump_at: None,
   318	        })
   319	    }

   321	    /// Carries out an `Act`: reboot and chain-load never return; false = idle timeout, the caller must reset.
   322	    pub fn act(&mut self, a: Act, dev: &Nic) -> bool {
   323	        match a {
   324	            Act::Idle => false,
   325	            Act::Reboot => wdt::hard_reset(),
   326	            Act::Sleep(m) => {
   327	                if let Err(e) = crate::sleep::enter(m, &dev.0) {
   328	                    kprintln!("sleep: {}", e);
   329	                    console::status("sleep failed (see log)");
   330	                }
   331	                true
   332	            }
   333	            Act::Jump(p) => crate::loader::jump(&p, &dev.0),
   334	        }
   335	    }

   337	    /// Deadlines first, then at most one datagram. Only an authenticated command counts as activity (`last_ok`).
   338	    pub fn step(&mut self, sock: &mut udp::Socket, t: i64) -> Option<Act> {
   339	        crate::hda::poll(t);
   340	        if t - self.last_footer >= 1000 {
   341	            self.last_footer = t;
   342	            let mut r = heapless::String::<72>::new();
   343	            let s = t / 1000;
   344	            let _ = core::fmt::write(&mut r, format_args!("{}   up {}:{:02}:{:02}", self.ip, s / 3600, s / 60 % 60, s % 60));
   345	            if s % 5 == 0 || self.battery.is_none() {
   346	                self.battery = Some(crate::ec::battery());
   347	            }
   348	            if let Some(Some((pct, ac, chg))) = self.battery {
   349	                let _ = core::fmt::write(&mut r, format_args!("   BAT {}%{}", pct, if chg { " charging" } else if ac { " AC" } else { "" }));
   350	            }
   351	            if let Some((h, m)) = time::rtc_hm() {
   352	                let _ = core::fmt::write(&mut r, format_args!("   {:02}:{:02}", h, m));
   353	            }
   354	            console::footer(b"NC110 agentic OS", r.as_bytes());
   355	        }
   356	        if t - self.last_ok > self.idle_ms {
   357	            kprintln!("bmctl: idle {} s -> reset", self.idle_ms / 1000);
   358	            console::status("no control traffic: restarting");
   359	            return Some(Act::Idle);
   360	        }
   361	        if self.jump_at.as_ref().is_some_and(|(j, _)| t > *j) {
   362	            let (_, p) = self.jump_at.take().unwrap();
   363	            kprintln!("bmctl: chain-loading the staged kernel");
   364	            console::status("loading new kernel");
   365	            crate::udplog::dump(7);
   366	            return Some(Act::Jump(p));
   367	        }
   368	        if let Some((at, m)) = self.sleep_at {
   369	            if t > at {
   370	                self.sleep_at = None;
   371	                console::status("sleeping (wake: magic packet, power button, or the RTC alarm)");
   372	                return Some(Act::Sleep(m));
   373	            }
   374	        }
   375	        if self.reboot_at.is_some_and(|r| t > r) {
   376	            kprintln!("bmctl: reboot requested");
   377	            console::status("restarting");
   378	            return Some(Act::Reboot);
   379	        }
   380	        if t - self.last_hello > 2000 {
   381	            self.last_hello = t;
   382	            let mut h = heapless::String::<128>::new();
   383	            let _ = core::fmt::write(&mut h, format_args!("nc110 hello bmctl {}:{} boot {}", self.ip, PORT, self.bid));
   384	            let _ = sock.send_slice(h.as_bytes(), IpEndpoint::new(IpAddress::Ipv4(Ipv4Address::BROADCAST), 5514));
   385	        }
   386	        let sess = unsafe { &mut *core::ptr::addr_of_mut!(SESS) };
   387	        let pend = unsafe { &mut *core::ptr::addr_of_mut!(PEND) };
   388	        let mut buf = [0u8; 1500];
   389	        let Ok((n, meta)) = sock.recv_slice(&mut buf) else { return None };
   390	        let from = meta.endpoint;
   391	        if n == 3 + 16 && &buf[..3] == b"who" {
   392	            if t - self.who_window > 1000 {
   393	                (self.who_window, self.who_count) = (t, 0);
   394	            }
   395	            self.who_count += 1;
   396	            // ponytail: one global ceiling (CPU and reflection; bmctl is LAN-only), not per sender. A flood above it, or
   397	            // one spoofing many sources, still starves `who`; the full fix is a stateless cookie handshake (protocol v2).
   398	            if self.who_count > 1000 {
   399	                return None;
   400	            }
   401	            // Unauthenticated `who` only ever writes the pending table (oldest entry out), never an authenticated session.
   402	            let slot = (0..pend.len()).min_by_key(|&j| pend[j].as_ref().map_or(i64::MIN, |x| x.t));
   403	            let (Some(slot), Some(next)) = (slot, self.next_sess.checked_add(1)) else { return None };
   404	            let id = self.next_sess;
   405	            self.next_sess = next;
   406	            let mut nc = [0u8; 16];
   407	            nc.copy_from_slice(&buf[3..19]);
   408	            pend[slot] = Some(Pend { id, nc, peer: from, t });
   409	            let mut chk = heapless::String::<64>::new();
   410	            hex(&who_check(&self.key, &nc, &self.boot_id, id)[..16], &mut chk);
   411	            let mut h = heapless::String::<160>::new();
   412	            let _ = core::fmt::write(&mut h, format_args!("nc110 boot {} session {} check {}", self.bid, id, chk));
   413	            let _ = sock.send_slice(h.as_bytes(), from);
   414	            return None;
   415	        }
   416	        // Frame: [session u32 LE][seq u64 LE][ct][tag]
   417	        if n < 4 + 8 + 16 {
   418	            self.bad += 1;
   419	            return None;
   420	        }
   421	        let sid = u32::from_le_bytes(buf[..4].try_into().unwrap());
   422	        // Authenticated session, or a pending one from the endpoint that asked `who` (promoted below once the frame
   423	        // authenticates; only then may it evict the least recently used authenticated session).
   424	        let in_sess = sess.iter().position(|x| x.as_ref().is_some_and(|x| x.id == sid && x.peer == Some(from)));
   425	        let in_pend = pend.iter().position(|x| x.as_ref().is_some_and(|x| x.id == sid && x.peer == from));
   426	        let mut cand = None;
   427	        let se = match (in_sess, in_pend) {
   428	            (Some(i), _) => sess[i].as_mut().unwrap(),
   429	            (None, Some(i)) => {
   430	                let p = pend[i].as_ref().unwrap();
   431	                cand.insert(Sess { id: p.id, key: session_key(&self.key, &self.boot_id, p.id, &p.nc), last_seq: 0, peer: Some(from), last_t: t, retx: 0, reply: [0; 8 + 1300 + 16], reply_len: 0 })
   432	            }
   433	            _ => {
   434	                self.bad += 1;
   435	                return None;
   436	            }
   437	        };
   438	        let seq = u64::from_le_bytes(buf[4..12].try_into().unwrap());
   439	        let (body, tg) = buf[12..n].split_at_mut(n - 12 - 16);
   440	        if !open(&se.key, &nonce(b"C2K\0", seq), &self.boot_id, body, tg) {
   441	            self.bad += 1;
   442	            return None;
   443	        }
   444	        if seq == se.last_seq && se.reply_len > 0 {
   445	            if se.retx >= 5 {
   446	                return None; // cap resends of one reply
   447	            }
   448	            se.retx += 1;
   449	            self.retx += 1; // retry of the last request: same bytes again (to the bound peer), nothing re-executed
   450	            let _ = sock.send_slice(&se.reply[..se.reply_len], from);
   451	            return None;
   452	        }
   453	        if seq <= se.last_seq {
   454	            self.replay += 1;
   455	            return None;
   456	        }
   457	        se.last_seq = seq;
   458	        se.last_t = t;
   459	        se.retx = 0;
   460	        self.last_ok = t;
   461	        let se = if let (None, Some(i)) = (in_sess, in_pend) {
   462	            let dst = (0..sess.len()).find(|&j| sess[j].is_none())
   463	                .unwrap_or_else(|| (0..sess.len()).min_by_key(|&j| sess[j].as_ref().map_or(i64::MIN, |x| x.last_t)).unwrap());
   464	            pend[i] = None;
   465	            sess[dst] = cand.take();
   466	            sess[dst].as_mut().unwrap()
   467	        } else {
   468	            se
   469	        };
   470	        self.ok += 1;
   471	        let mut out = [0u8; 1300];
   472	        let len = {
   473	            let mut w = Writer { b: &mut out, n: 0 };
   474	            let loading = |c: &[u8]| c.starts_with(b"ldat") || c.starts_with(b"lbeg") || c.starts_with(b"lcom"); // incl. lcomlinux
   475	            if self.jump_at.is_some() && loading(body) {
   476	                // committed: the staged image is frozen until the hop (Codex loader MUST-FIX 2)
   477	                let _ = core::fmt::write(&mut w, format_args!("err committed, jump pending"));
   478	            } else if body.len() >= 8 && &body[..4] == b"ldat" {
   479	                // binary chunk: "ldat" ++ u32 LE offset ++ bytes (checked before the text parser)
   480	                let off = u32::from_le_bytes(body[4..8].try_into().unwrap()) as u64;
   481	                let r = match self.stage.as_mut() {
   482	                    Some(st) => crate::loader::data(st, off, &body[8..]),
   483	                    None => Err("no lbeg"),
   484	                };
   485	                match r {
   486	                    Ok(got) => { let _ = core::fmt::write(&mut w, format_args!("ok {}", got)); }
   487	                    Err(e) => { let _ = core::fmt::write(&mut w, format_args!("err {}", e)); }
   488	                }
   489	            }
   490	            let cmd = if w.n > 0 { "" } else { core::str::from_utf8(body).unwrap_or("") };
   491	            let mut it = cmd.split(' ');
   492	            match it.next().unwrap_or("") {
   493	                "" if w.n > 0 => {} // already answered (ldat)
   494	                "lbeg" => {
   495	                    let size: u64 = it.next().and_then(|x| x.parse().ok()).unwrap_or(0);
   496	                    match crate::loader::begin(size, it.next().unwrap_or("")) {
   497	                        Ok(st) => {
   498	                            self.stage = Some(st);
   499	                            let _ = core::fmt::write(&mut w, format_args!("ok staging {} B at {:#x}", size, crate::loader::STAGE));
   500	                        }
   501	                        Err(e) => { let _ = core::fmt::write(&mut w, format_args!("err {}", e)); }
   502	                    }
   503	                }
   504	                "lcomlinux" => {
   505	                    let rest = cmd.splitn(2, ' ').nth(1).unwrap_or("");
   506	                    let r = match self.stage.as_ref() {
   507	                        Some(st) => crate::loader::prepare_linux(st, rest),
   508	                        None => Err("no lbeg"),
   509	                    };
   510	                    match r {
   511	                        Ok(p) => {
   512	                            let _ = core::fmt::write(&mut w, format_args!("ok booting Linux in 300 ms"));
   513	                            self.jump_at = Some((t + 300, p));
   514	                        }
   515	                        Err(e) => { let _ = core::fmt::write(&mut w, format_args!("err {}", e)); }
   516	                    }
   517	                }
   518	                "lcom" => {
   519	                    let rest = cmd.splitn(2, ' ').nth(1).unwrap_or("");
   520	                    let r = match self.stage.as_ref() {
   521	                        Some(st) => crate::loader::prepare(st, rest, &self.seed),
   522	                        None => Err("no lbeg"),
   523	                    };
   524	                    match r {
   525	                        Ok(p) => {
   526	                            let _ = core::fmt::write(&mut w, format_args!("ok jumping in 300 ms"));
   527	                            self.jump_at = Some((t + 300, p)); // the reply leaves first, then the hop
   528	                        }
   529	                        Err(e) => { let _ = core::fmt::write(&mut w, format_args!("err {}", e)); }
   530	                    }
   531	                }
   532	                "ping" => {
   533	                    let _ = core::fmt::write(&mut w, format_args!("pong uptime_ms={} ip={}", t, self.ip));
   534	                }
   535	                "stats" => {
   350	                    }
   351	                    if st & (1 << 23) != 0 {
   352	                        active = true;
   353	                        break;
   354	                    }
   355	                    let a = st & 0x7FF;
   356	                    let a = if a == 0x7FF { 0 } else { a as usize + 1 };
   357	                    got += a;
   358	                    done_toggles += 1;
   359	                    if inp && a < mps as usize {
   360	                        break; // short packet ends the transfer
   361	                    }
   362	                }
   363	                if !active {
   364	                    if res.is_ok() || res == Err("timeout") {
   365	                        d.eps[e].2 = ((d.eps[e].2 as u32 + done_toggles) & 1) as u8;
   366	                        res = Ok(got);
   367	                    }
   368	                    break;
   369	                }
   370	            }
   371	            write_volatile(&mut qh.0[1], 1);
   372	            self.w16(0, 0);
   373	            res
   374	        }
   375	    }
   376	}

   378	/// bmctl `usb ctl <dev> <setup: 16 hex> [len]`, `usb in <dev> <ep hex> <len>`, `usb out <dev> <ep hex> <data hex>`.
   379	pub fn cmd<'a>(w: &mut dyn Write, it: &mut impl Iterator<Item = &'a str>, sub: &str) {
   380	    let devs = unsafe { &mut *core::ptr::addr_of_mut!(DEVS) };
   381	    let Some(d) = it.next().and_then(|x| x.parse::<usize>().ok()).and_then(|i| devs.get_mut(i)).and_then(|x| x.as_mut()) else {
   382	        let _ = write!(w, "err no such device (run usb first)");
   383	        return;

        self.b[self.n..self.n + k].copy_from_slice(&s[..k]);

//! Monotonic milliseconds from the TSC, calibrated once against PIT channel 2 (gate via port 0x61).
use crate::port::{inb, outb, rdtsc};

static mut TSC_PER_MS: u64 = 1_000_000; // fallback guess (1 GHz) if calibration fails
static mut T0: u64 = 0;

pub fn init() -> u64 {
    unsafe {
        // PIT ch2, mode 0, lobyte/hibyte, count for 10 ms at 1.193182 MHz.
        const COUNT: u16 = 11932;
        let g = inb(0x61);
        outb(0x61, (g & 0xFD) | 0x01); // gate on, speaker off
        outb(0x43, 0b1011_0000);
        outb(0x42, COUNT as u8);
        outb(0x42, (COUNT >> 8) as u8);
        // Re-trigger: toggle gate low->high so mode 0 starts counting now.
        let g = inb(0x61);
        outb(0x61, g & 0xFE);
        outb(0x61, g | 0x01);
        let a = rdtsc();
        let mut spins: u64 = 0;
        while inb(0x61) & 0x20 == 0 && spins < 200_000_000 {
            spins += 1;
        }
        let b = rdtsc();
        if spins < 200_000_000 && b > a {
            TSC_PER_MS = ((b - a) / 10).max(1);
        }
        T0 = rdtsc();
        TSC_PER_MS
    }
}

/// ms the TSC had counted when `init` ran: time since the last CPU reset (firmware + GRUB + our early boot),
/// since a hard reset zeroes the TSC; after an in-RAM chain-load it also includes the previous kernel's run.
pub fn since_reset_ms() -> u64 {
    unsafe { T0 / TSC_PER_MS }
}

pub fn tsc_to_ms(t: u64) -> u64 {
    unsafe { t / TSC_PER_MS }
}

pub fn ready() -> bool {
    unsafe { T0 != 0 }
}

pub fn now_ms() -> i64 {
    unsafe { ((rdtsc() - T0) / TSC_PER_MS) as i64 }
}

/// Busy-wait at least `us` microseconds (TSC, after init()).
pub fn delay_us(us: u64) {
    unsafe {
        let t = rdtsc();
        let n = us * TSC_PER_MS / 1000 + 1;
        while rdtsc().wrapping_sub(t) < n {
            core::hint::spin_loop();
        }
    }
}

/// CMOS RTC hours and minutes (the NC110's RTC keeps local time: Linux there labels HKT as UTC). Waits out an update in
/// progress (bounded), handles BCD and 12-hour formats. None if the RTC never settles.
pub fn rtc_hm() -> Option<(u8, u8)> {
    use crate::port::{inb, outb};
    let rd = |r: u8| unsafe {
        outb(0x70, r);
        inb(0x71)
    };
    for _ in 0..1000 {
        if rd(0x0A) & 0x80 == 0 {
            let (m, h, b) = (rd(0x02), rd(0x04), rd(0x0B));
            let bcd = |v: u8| if b & 0x04 == 0 { (v & 0x0F) + (v >> 4) * 10 } else { v };
            let pm = h & 0x80 != 0;
            let mut hh = bcd(h & 0x7F);
            if b & 0x02 == 0 {
                hh = hh % 12 + if pm { 12 } else { 0 };
            }
            return Some((hh, bcd(m)));
        }
    }
    None
}

def rpc(raw, want=lambda d: True):
    """Send `raw`, return the first reply from the target that `want` accepts. Stale datagrams (late answers to
    an earlier timed-out attempt) are drained before sending and skipped while waiting, so a resend can never
    leave the client one reply behind."""
    host = socket.gethostbyname(a.host)
    for _ in range(3):  # a retry of the same frame gets the kernel's cached reply, nothing runs twice
        s.setblocking(False)
        try:
            while True:
                s.recvfrom(2048)
        except (BlockingIOError, OSError):

        s.sendto(raw, (a.host, a.port))
        end = time.monotonic() + TIMEOUT[0]
        try:
            while time.monotonic() < end:
                d, src = s.recvfrom(2048)
                if src[0] == host and want(d):
                    return d
        except socket.timeout:
            continue
    sys.exit(f"no reply from {a.host}:{a.port}")

None found in commit `1186a01`. Most is sound. This is a source review; I did not run QEMU or benchmark the N570.

The new promotion path preserves the relevant security invariants:

- **Peer binding:** [bmctl.rs:425](~/nc110-baremetal/src/bmctl.rs:425) requires `x.id == sid && x.peer == from`. Matching a SID from another UDP endpoint cannot select that pending entry.
- **Forged frames leave pending state intact:** the temporary session is created with `cand.insert(...)`; failed `open(...)` returns before `pend[i] = None`. It neither consumes the entry nor changes its nonce, peer, timestamp, or sequence state.
- **Replay state survives promotion:** authentication precedes `if seq <= se.last_seq`, then `se.last_seq = seq` precedes `sess[dst] = cand.take()`. Sequence zero is rejected; a duplicate first frame subsequently uses the authenticated session’s cached reply. Eviction cannot resurrect the pending session because promotion removed it.
- **No SID wraparound:** `self.next_sess.checked_add(1)` refuses new issuance at exhaustion. Different pending records cannot reuse an issued SID within this controller’s lifetime.
- **No new aliasing defect:** on the promotion branch, `se` borrows the local candidate. Its last use precedes `cand.take()`; on the other branch it borrows the existing session. The branch-specific borrows do not introduce overlapping live mutable access. The existing unsafe static-table access still relies on serialized callers.
- **Both slicing fixes are correct:** `x.bytes().all(|c| c.is_ascii_hexdigit())` guarantees every subsequent two-byte string slice lands on UTF-8 boundaries.

**SHOULD**

1. **Make the flood test assert that a flood actually occurred.**  
   [dos-q35-test.py:88](~/nc110-baremetal/tools/dos-q35-test.py:88) decides success using only:
   ```python
   good = alive and ok >= 0.8 * tries
   ```
   A failed flood thread or severely underloaded sender can therefore produce PASS. Also, `sleep(0.002)` every five sends nominally targets approximately **2,500 pps**, not the comment’s “a few hundred”; actual scheduling and QEMU throughput determine the load.

   Use explicit rate pacing, propagate thread failures, and assert achieved load—preferably also the kernel’s `sessions_issued` increase. Run a controlled below-ceiling case separately from overload. The existing test meaningfully exercises fresh handshakes under contention, but does not establish a reproducible 380-pps or 1000-pps guarantee.

2. **Add direct promotion regression cases.**  
   None of the three named tests checks: wrong endpoint → valid original endpoint; bad tag → valid frame on the same pending SID; authenticated sequence zero → valid sequence one; duplicate first frame after promotion; or replay after authenticated-session eviction. The older `bmctl-q35-test.sh` checks retries and tampering, but its tamper checks occur after promotion. These cases would directly guard the new candidate lifecycle.

3. **Qualify the pending-table availability claim.**  
   [bmctl.rs:37](~/nc110-baremetal/src/bmctl.rs:37) says a flood “cannot push a real client's entry out within its round trip.” That is stronger than the implementation guarantees:
   ```rust
   min_by_key(|&j| pend[j].as_ref().map_or(i64::MIN, |x| x.t))
   ```
   At evenly spaced 1000 accepted requests/s, 256 entries provide roughly 256 ms of retention. A fixed-window ceiling permits bursts; a client can lose its entry much sooner. Equal millisecond timestamps also resolve by index rather than issuance order. Describe this as mitigation for the measured load and latency. If strict oldest-issued eviction matters, the nonwrapping SID is a suitable ordering key.

   **`i64::MIN` itself is correct:** empty entries sort ahead of ordinary uptime timestamps. There is no subtraction or overflow in this comparison.

4. **Measure before optimizing the scan.**  
   The stated load entails **256,000 entry inspections/s** for `who` eviction. That alone is not persuasive evidence of an N570 performance problem. Forged frames matching a pending SID now additionally trigger key derivation and candidate initialization, so benchmark that path too. QEMU success does not measure target-hardware CPU headroom. I would not require a more complicated data structure without measurements.
