+an implementation independent of the kernel's, so agreement is a real interop check.
+"""
+import argparse, hashlib, socket, struct, sys, time
+from cryptography.hazmat.primitives.ciphers.aead import ChaCha20Poly1305
+
+ap = argparse.ArgumentParser()
+ap.add_argument("--host", default="192.168.50.120")
+ap.add_argument("--port", type=int, default=5515)
+ap.add_argument("--key", default="test-keys/client.key")
+ap.add_argument("cmd", nargs="+")
+a = ap.parse_args()
+km = open(a.key, "rb").read()
+assert len(km) == 72, "key module must be 72 bytes"
+key = hashlib.sha256(b"nc110-bmctl-v1" + km[8:40]).digest()
+aead = ChaCha20Poly1305(key)
+s = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
+s.settimeout(2.0)
+
+
+def rpc(raw):
+    for _ in range(3):
+        s.sendto(raw, (a.host, a.port))
+        try:
+            return s.recvfrom(2048)[0]
+        except socket.timeout:
+            continue
+    sys.exit(f"no reply from {a.host}:{a.port}")
+
+
+who = rpc(b"who").decode()
+assert who.startswith("nc110 boot "), who
+boot = bytes.fromhex(who.split()[2])
+
+
+def call(cmd):
+    seq = time.time_ns() // 1000  # strictly increasing across invocations on this Mac
+    ct = aead.encrypt(b"C2K\0" + struct.pack("<Q", seq), cmd.encode(), boot)
+    r = rpc(struct.pack("<Q", seq) + ct)
+    rseq = struct.unpack("<Q", r[:8])[0]
+    if rseq != seq:
+        sys.exit(f"reply seq {rseq} != {seq}")
+    return aead.decrypt(b"K2C\0" + struct.pack("<Q", seq), r[8:], boot)
+
+
+cmd = " ".join(a.cmd)
+if cmd == "logall":
+    off, out = 0, b""
+    while True:
+        chunk = call(f"log {off}")
+        if not chunk:
+            break
+        out += chunk
+        off += len(chunk)
+    sys.stdout.write(out.decode(errors="replace"))
+else:
+    print(call(cmd).decode(errors="replace"))

//! Entropy for Sunset (getrandom custom backend). M2.5: CPU-jitter source first, RDRAND only as an extra.
//!
//! The NC110's Atom N570 has NO RDRAND. The noise source races execution against the TSC: count strided
//! walk steps (2 MiB buffer) completed while the TSC advances by a fixed window (see `raw_delta`).
//! Interrupts are off, so all jitter comes from the microarchitecture (cache/DRAM/bus timing) or, in QEMU,
//! from the HOST: TCG's TSC moves in 1000-tick (1 us host clock) steps and the count reflects host speed
//! noise, so QEMU numbers say nothing about the N570. In-order Atoms are the known weak case
//! for jitter sources; the claim below must be re-measured on the real board (cmdline `rng=dump`).
//!
//! H_CLAIM = 0.1 bit per 8-bit raw sample (low byte of each count). Chosen BELOW the worst QEMU estimate
//! (lag-1 0.198, tools/entropy-est.py on proof/m25-rngdump-serial.log), not derived from hardware.
//! SP 800-90B-style health tests, false-positive rate alpha = 2^-30, cutoffs from H_CLAIM:
//!   Repetition Count Test (4.4.1): cutoff 1 + ceil(30 / 0.1) = 301.
//!   Adaptive Proportion Test (4.4.2): W = 512, cutoff 507 = smallest C with P(1 + Bin(511, 2^-0.1) >= C)
//!   <= 2^-30 (computed: tools/entropy-est.py --cutoffs 0.1). At this H the APT is weak by construction.
//! Startup: 4096 samples must pass both tests AND both in-kernel estimates (MCV, SP 800-90B 6.3.1, and a
//! lag-1 conditional most-common-value bound that catches serial correlation) must be >= H_CLAIM, else refuse.
//! Continuous: every sample is tested; one failure latches FAILED and every later request errors.
//!
//! Conditioning: SHA-256 (the pinned soft `sha2`) over the full 64-bit counts. Credit 0.1 bit/sample, 10240
//! samples per key = 1024 credited bits vs 320 needed (256 + 64, 90B 3.1.5.1.2), ~3.2x oversampling as in
//! jitterentropy. Output is a hash ratchet (out = H(key|"out"|ctr), then key = H(key|"next")), reseeded from
//! 10240 fresh samples every 8 requests. It is an extractor + ratchet, NOT a certified SP 800-90A DRBG.

pub fn has_rdrand() -> bool {
    let r = __cpuid(1);
    r.ecx & (1 << 30) != 0

#[target_feature(enable = "rdrand")]
unsafe fn rdrand64() -> Option<u64> {
    let mut v = 0u64;
    for _ in 0..32 {
        if _rdrand64_step(&mut v) == 1 {
            return Some(v);
        }
    }
    None
}

/// 32 bytes from RDRAND, or Err if it keeps failing (CF=0) or looks stuck (same value twice in a row).
fn rdrand32() -> Result<[u8; 32], ()> {
    let mut out = [0u8; 32];
    let mut last = None;
    for c in out.chunks_mut(8) {
        let v = unsafe { rdrand64() }.ok_or(())?;
        if last == Some(v) {
            return Err(());
        }
        last = Some(v);
        c.copy_from_slice(&v.to_le_bytes());
    }
    Ok(out)
}

        let p = core::ptr::addr_of_mut!(WALK) as *mut u8;
        *pos = (*pos + 4160) % WALK_SZ;
        let v = core::ptr::read_volatile(p.add(*pos));
        core::ptr::write_volatile(p.add(*pos), v.wrapping_add(1));
    }
}

/// TSC granularity: smallest nonzero step between back-to-back reads (QEMU/TCG: 1000; real CPUs: tens).
/// None = the timer never moved (broken/constant): no jitter source at all.
fn tsc_granularity() -> Option<u64> {
    let mut g = u64::MAX;
    let mut last = tsc();
    for _ in 0..100_000 {
        let t = tsc();
        let d = t.wrapping_sub(last);
        if d != 0 && d < g {
            g = d;
        }
        last = t;
    }
    if g == u64::MAX { None } else { Some(g) }
}

/// One raw sample = how many walk steps (strided over 2 MiB) complete while the TSC advances by WINDOW
/// ticks (max(64 x granularity, 32768)), starting on a TSC edge. It races two clocks: execution speed (cache, bus, and in
/// QEMU the host's scheduling/caches) against the timer. A first attempt timed a FIXED walk instead; under
/// TCG the TSC moves in 1000-tick steps, so the delta was nearly constant and both the RCT and the in-kernel
/// estimate refused (proof/m25-short-walk-serial.log, proof/m25-fixedwalk-serial.log).
/// If no TSC edge is seen the raw reading is 0 (a dead timer reads as a constant, so the RCT fires).
fn raw_delta() -> u64 {
    let w = unsafe { WINDOW };
    let t0 = tsc();
    let mut start = t0;
    let mut spins = 0u32;
    while start == t0 {
        start = tsc();
        spins += 1;
        if spins > 1 << 16 {
            return 0;
        }
    }
    let mut pos = unsafe { WALK_POS };
    let mut n = 0u64;
    while tsc().wrapping_sub(start) < w && n < 1 << 20 {
        touch(&mut pos);
        n += 1;
    }
    unsafe { WALK_POS = pos };
    n
}

// ---------------- health tests (SP 800-90B 4.4) ----------------

struct Health {
    rct_last: u8,
    rct_n: u32,
    apt_first: u8,
    apt_n: u32,
    apt_i: u32,
    started: bool,
}

impl Health {
    const fn new() -> Self {
        Self { rct_last: 0, rct_n: 0, apt_first: 0, apt_n: 0, apt_i: 0, started: false }
    }
    /// Err(name) when a test fires.
    fn feed(&mut self, s: u8) -> Result<(), &'static str> {
        if self.started && s == self.rct_last {
            self.rct_n += 1;
            if self.rct_n >= RCT_CUTOFF {
                return Err("repetition count");
            }
        } else {
            self.rct_last = s;
            self.rct_n = 1;
        }
        if !self.started || self.apt_i == APT_W {
            self.apt_first = s;
            self.apt_n = 1;
            self.apt_i = 1;
        } else {
            self.apt_i += 1;
            if s == self.apt_first {
                self.apt_n += 1;
                if self.apt_n >= APT_CUTOFF {
                    return Err("adaptive proportion");
                }
            }
        }
        self.started = true;
        Ok(())
    }
}

// ---------------- pool ----------------

static mut HEALTH: Health = Health::new();
static mut KEY: [u8; 32] = [0; 32];
static mut CTR: u64 = 0;
static mut CALLS: u32 = 0;
static mut RESEEDS: u32 = 0;
static mut SAMPLES: u64 = 0;
static mut SEEDED: bool = false;
static mut FAILED: bool = false;
static mut USE_RDRAND: bool = false;
static mut LOWB: [u8; STARTUP_N] = [0; STARTUP_N];

/// Collect `n` samples through the continuous health tests into `h`. Latches FAILED on any failure.
fn gather(h: &mut Sha256, n: usize, keep_low: bool) -> Result<(), &'static str> {
    for i in 0..n {
        let d = raw_delta();
        let s = d as u8;
        unsafe { SAMPLES += 1 };
        if let Err(e) = unsafe { (*core::ptr::addr_of_mut!(HEALTH)).feed(s) } {
            unsafe { FAILED = true };
            return Err(e);
        }
        if keep_low {
            unsafe { LOWB[i] = s };
        }
        h.update(d.to_le_bytes());
    }
    Ok(())
}

fn sqrt(x: f64) -> f64 {
    if x <= 0.0 {
        return 0.0;
    }
    let mut g = if x > 1.0 { x } else { 1.0 };
    for _ in 0..60 {
        g = 0.5 * (g + x / g);
    }
    g
}

fn log2(x: f64) -> f64 {
    let b = x.to_bits();
    let e = ((b >> 52) & 0x7ff) as i64 - 1023;
    let m = f64::from_bits((b & 0x000f_ffff_ffff_ffff) | (1023u64 << 52)); // [1, 2)
    let y = (m - 1.0) / (m + 1.0);
    let (mut s, mut t) = (0.0, y);
    let mut k = 1.0;
    while k < 60.0 {
        s += t / k;
        t *= y * y;
        k += 2.0;
    }
    e as f64 + 2.0 * s / core::f64::consts::LN_2
}

static mut TRANS: [[u16; 256]; 256] = [[0; 256]; 256];

/// Lag-1 bound: max over previous values a (>= 20 successors) of the 99% upper bound on max_b P(b | a).
/// Not a 90B estimator; a cheap guard against the serial correlation MCV (an IID estimate) cannot see.
fn lag1_milli(samples: &[u8]) -> u32 {
    let t = unsafe { &mut *core::ptr::addr_of_mut!(TRANS) };
    for row in t.iter_mut() {
        row.fill(0);
    }
    for w in samples.windows(2) {
        t[w[0] as usize][w[1] as usize] += 1;
    }
    let mut worst = 0.0f64;
    for row in t.iter() {
        let n: u32 = row.iter().map(|&x| x as u32).sum();
        if n >= 20 {
            let p = *row.iter().max().unwrap() as f64 / n as f64;
            let pu = (p + 2.576 * sqrt(p * (1.0 - p) / (n as f64 - 1.0))).min(1.0);
            if pu > worst {
                worst = pu;
            }
        }
    }
    if worst == 0.0 {
        return 8000;
    }
    (-log2(worst) * 1000.0) as u32
}

/// SP 800-90B 6.3.1 most-common-value estimate, in milli-bits per 8-bit sample.
fn mcv_milli(samples: &[u8]) -> (u32, u32, u32) {
    let mut c = [0u32; 256];
    for &s in samples {
        c[s as usize] += 1;
    }
    let max = *c.iter().max().unwrap();
    let distinct = c.iter().filter(|&&x| x > 0).count() as u32;
    let n = samples.len() as f64;
    let p = max as f64 / n;
    let pu = (p + 2.576 * sqrt(p * (1.0 - p) / (n - 1.0))).min(1.0);
    ((-log2(pu) * 1000.0) as u32, max, distinct)
}

/// `rng=dump`: print a SEPARATE batch of STARTUP_N raw samples (never used for keys) for offline estimation
/// (tools/entropy-est.py, or NIST SP800-90B ea_non_iid). Printed before the health tests so a failing source
/// can still be measured. Full deltas for the first 64, then the 8-bit samples.
fn dump_raw() {
    kprint!("rawdelta:");
    for _ in 0..64 {
        kprint!(" {}", raw_delta());
    }
    kprintln!("");
    let mut line = [0u8; 64];
    for i in 0..STARTUP_N / 64 {
        for b in line.iter_mut() {
            *b = raw_delta() as u8;
        }
        kprint!("rawjit {:04}:", i * 64);
        for b in line {
            kprint!(" {:02x}", b);
        }
        kprintln!("");
    }
}

/// Startup: health-test STARTUP_N samples, estimate min-entropy, seed the key. Err = do not run SSH.
pub fn init(dump: bool) -> Result<(), ()> {
    unsafe {
        if SEEDED {
            return Ok(());
        }
        if FAILED {
            return Err(());
        }
    }
    // A dead timer is not refused here: the samples then all read 0 and the health tests (the one authority
    // on the source) must refuse. That is the path the `badjitter` negative control exercises.
    let g = match tsc_granularity() {
        Some(g) => g,
        None => {
            kprintln!("rng: TSC never advanced in 100000 reads (dead timer?)");
            1
        }
    };
    // 64 timer steps, but never under 32768 ticks (~20 us on the N570): on a cycle-exact TSC, 64 x granularity
    // would be a window of a few DRAM accesses and the count close to constant.
    let w = (64 * g).max(32768);
    unsafe { WINDOW = w };
    kprintln!("rng: TSC granularity {} ticks, sample window {} ticks", g, w);
    if dump {
        dump_raw();
    }
    let mut h = Sha256::new();
    h.update(b"nc110-m2.5 seed");
    let t0 = rdtsc();
    if let Err(e) = gather(&mut h, STARTUP_N, true) {
        kprintln!("rng: jitter startup health test FAILED ({}) -> refusing to run SSH", e);
        return Err(());
    }
    let cycles = rdtsc().wrapping_sub(t0);
    let low = unsafe { &*core::ptr::addr_of!(LOWB) };
    let (hm, max, distinct) = mcv_milli(low);
    let hl = lag1_milli(low);
    kprintln!(
        "rng: jitter startup {} samples passed RCT(cut {}) + APT(W {}, cut {}); {} TSC ticks/sample",
        STARTUP_N, RCT_CUTOFF, APT_W, APT_CUTOFF, cycles / STARTUP_N as u64
    );
    kprintln!(
        "rng: min-entropy per 8-bit sample: MCV {}.{:03} (max {}/{}, {} distinct), lag-1 {}.{:03}; claimed {}.{:03}",
        hm / 1000, hm % 1000, max, STARTUP_N, distinct, hl / 1000, hl % 1000, H_CLAIM_MILLI / 1000, H_CLAIM_MILLI % 1000
    );
    if hm.min(hl) < H_CLAIM_MILLI {
        unsafe { FAILED = true };
        kprintln!("rng: estimate below the claimed 0.1 bit/sample -> refusing to run SSH");
        return Err(());
    }
    if let Err(e) = gather(&mut h, BLOCK_N - STARTUP_N, false) {
        kprintln!("rng: jitter health test FAILED while seeding ({}) -> refusing to run SSH", e);
        return Err(());
    }
    if has_rdrand() {
        match rdrand32() {
            Ok(r) => {
                h.update(r);
                unsafe { USE_RDRAND = true };
                kprintln!("rng: RDRAND present, mixed in (jitter alone already passed)");
            }
            Err(()) => {
                unsafe { FAILED = true };
                kprintln!("rng: RDRAND present but failing -> refusing to run SSH");
                return Err(());
            }
        }
    } else {
        kprintln!("rng: no RDRAND (as on the Atom N570): jitter is the only source");
    }
    unsafe {
        KEY = h.finalize().into();
        SEEDED = true;
    }
    Ok(())
}

/// Fill `buf` from the ratchet. Errors if not seeded, if a health test ever failed, or if a reseed fails.
pub fn fill(buf: &mut [u8]) -> Result<(), ()> {
    unsafe {
        if !SEEDED || FAILED {
            return Err(());
        }
        CALLS += 1;
        if CALLS % RESEED_EVERY == 0 {
            let mut h = Sha256::new();
            h.update(b"reseed");
            h.update(KEY);
            if gather(&mut h, BLOCK_N, false).is_err() {
                return Err(());
            }
            RESEEDS += 1;
            KEY = h.finalize().into();
        }
        if USE_RDRAND {
            let r = rdrand32().map_err(|_| FAILED = true)?;
            let mut h = Sha256::new();
            h.update(b"rdrand");
            h.update(KEY);
            h.update(r);
            KEY = h.finalize().into();
        }
        for chunk in buf.chunks_mut(32) {
            let mut h = Sha256::new();
            h.update(b"out");
            h.update(KEY);
            h.update(CTR.to_le_bytes());
            CTR += 1;
            chunk.copy_from_slice(&h.finalize()[..chunk.len()]);
        }
        let mut h = Sha256::new();
        h.update(b"next");
        h.update(KEY);
        KEY = h.finalize().into();
    }
    Ok(())
}

/// (requests served, reseeds, raw samples taken (all health-tested), health failure latched)
pub fn stats() -> (u32, u32, u64, bool) {
    unsafe { (CALLS, RESEEDS, SAMPLES, FAILED) }
}

#[no_mangle]
unsafe extern "Rust" fn __getrandom_v03_custom(dest: *mut u8, len: usize) -> Result<(), getrandom::Error> {
    core::ptr::write_bytes(dest, 0, len);
    let buf = core::slice::from_raw_parts_mut(dest, len);
    fill(buf).map_err(|_| getrandom::Error::UNSUPPORTED)
}
//! TCO watchdog (ICH9 in QEMU q35, NM10 on the NC110; both iTCO v2). boot.s arms it before any
//! Rust runs; this module reports what boot.s did, pets it, and runs the M0.5 rehearsal modes.
use crate::port::{inl, inw, outb, outw};
use crate::{kprintln, pci};
use core::ptr::{addr_of, read_volatile};

#[cfg(all(feature = "neg-control", feature = "grub-armed"))]
compile_error!("neg-control is QEMU-only and must never be combined with grub-armed (the hardware artifact)");

/// Negative control (`--features neg-control`, QEMU q35 only; boot.s never self-arms on NM10): boot.s SETS GCS.NO_REBOOT instead of clearing it
/// (QEMU's GCS powers up 0 with noreboot=false, so merely skipping the clear would prove nothing).
#[no_mangle]
static TCO_NEG_CONTROL: u8 = cfg!(feature = "neg-control") as u8;

/// `--features grub-armed`: boot.s does NOT arm; used to test the GRUB iorw/memrw arming sequence.
#[no_mangle]
static TCO_GRUB_ARMED: u8 = cfg!(feature = "grub-armed") as u8;

extern "C" {
    static tco_result: u32;
    static tco_gcs_pre: u32;
    static tco_gcs: u32;
    static tco_base: u32;
    static tco_sts1_pre: u16;
    static tco_sts2_pre: u16;
    static tco_cnt: u16;
}

fn rd<T: Copy>(p: *const T) -> T {
    unsafe { read_volatile(p) }
}

pub fn armed() -> bool {
    matches!(rd(addr_of!(tco_result)), 0 | 4 | 5)
}

fn pmbase() -> u16 {
    (pci::read32(pci::b0(31, 0), 0x40) & 0xFF80) as u16
}

pub fn base() -> u16 {
    pmbase() + 0x60
}

pub fn report() {
    let r = rd(addr_of!(tco_result));
    let what = match r {
        0 => "armed by boot.s",
        1 => "skipped (LPC 00:1f.0 is not ICH9 8086:2918 / NM10 8086:27bc)",
        2 => "NOT armed: PMBASE unset",
        3 => "NOT armed: RCBA disabled",
        4 => "left to GRUB (grub-armed build); live registers follow",
        5 => "NM10: boot.s does not arm (attended Rust arms it; else GRUB must); live registers follow",
        _ => "NOT armed: unknown",
    };
    kprintln!("wdt: {} (result={})", what, r);
    if r == 4 || r == 5 {
        let b = base();
        let gcs = unsafe { read_volatile(((pci::read32(pci::b0(31, 0), 0xF0) & 0xFFFF_C000) + 0x3410) as *const u32) };
        let (s1, s2, cnt, tmr) = unsafe { (inw(b + 4), inw(b + 6), inw(b + 8), inw(b + 0x12)) };
        kprintln!(
            "wdt: live TCOBASE={:#x} GCS={:#010x} NO_REBOOT={} TCO1_CNT={:#06x} HLT={} TCO1_STS={:#06x} TCO2_STS={:#06x} TCO_TMR={}",
            b, gcs, (gcs >> 5) & 1, cnt, (cnt >> 11) & 1, s1, s2, tmr & 0x3FF
        );
        return;
    }
    if r != 0 {
        return;
    }
    let (b, gcs, s1, s2, cnt) = {
        (rd(addr_of!(tco_base)) as u16, rd(addr_of!(tco_gcs)), rd(addr_of!(tco_sts1_pre)), rd(addr_of!(tco_sts2_pre)), rd(addr_of!(tco_cnt)))
    };
    kprintln!(
        "wdt: TCOBASE={:#x} GCS {:#010x} -> {:#010x} NO_REBOOT={} TCO1_CNT={:#06x} HLT={} LOCK={}",
        b, rd(addr_of!(tco_gcs_pre)), gcs, (gcs >> 5) & 1, cnt, (cnt >> 11) & 1, (cnt >> 12) & 1
    );
    kprintln!(
        "wdt: before-clear TCO1_STS={:#06x} TCO2_STS={:#06x} SECOND_TO_STS={} BOOT_STS={}",
        s1, s2, (s2 >> 1) & 1, (s2 >> 2) & 1
    );
    let tmr = unsafe { inw(b + 0x12) } & 0x3FF;
    kprintln!("wdt: TCO_TMR={} ticks x 0.6 s, reset expected after 2nd expiry (~{} ms)", tmr, tmr as u32 * 1200);
}

/// Live check of the reset path, whoever armed it: NO_REBOOT=0, HLT=0, TMR>=4, and the counter
/// (TCO_RLD reads back the current value on v2) decrements across ~1.3 ticks.
pub fn verify_live() -> bool {
    if !armed() {
        return false;
    }
    let b = base();
    let rcba = pci::read32(pci::b0(31, 0), 0xF0);
    if rcba & 1 == 0 {
        return false;
    }
    let gcs = unsafe { read_volatile(((rcba & 0xFFFF_C000) + 0x3410) as *const u32) };
    let (cnt, tmr) = unsafe { (inw(b + 8), inw(b + 0x12) & 0x3FF) };
    let c0 = unsafe { inw(b) } & 0x3FF;
    delay_ms(800);
    let c1 = unsafe { inw(b) } & 0x3FF;
    let ok = gcs & 0x20 == 0 && cnt & 0x0800 == 0 && tmr >= 4 && c1 != c0;
    kprintln!(
        "wdt: verify NO_REBOOT={} HLT={} TMR={} count {}->{} => {}",
        (gcs >> 5) & 1, (cnt >> 11) & 1, tmr, c0, c1, if ok { "VERIFIED" } else { "NOT VERIFIED" }
    );
    ok
}

/// QEMU `-machine pc` (PIIX3 8086:7000 at 00:01.0) has no TCO; the M0 NIC demo may run unprotected there.
pub fn is_qemu_pc() -> bool {
    pci::read32(pci::b0(1, 0), 0) == 0x7000_8086 && rd(addr_of!(tco_result)) == 1
}

/// The negative control must reach its hang unverified (that is the point); only on q35, never NM10.
pub fn neg_control_on_q35() -> bool {
    let on = cfg!(feature = "neg-control") && rd(addr_of!(tco_result)) == 0;
    if on {
        kprintln!("wdt: neg-control build on q35: proceeding unverified on purpose");
    }
    on
}

/// Hard reset via the reset control register (spec M2 `reboot`): back to the firmware / Slot A.
pub fn hard_reset() -> ! {
    kprintln!("wdt: hard reset via 0xCF9");
    unsafe {
        outb(0xCF9, 0x02);
        outb(0xCF9, 0x06);
    }
    loop {
        unsafe { core::arch::asm!("cli; hlt") };
    }
}

/// Attended `wdt=probe`: arm the NM10 TCO ourselves without touching locked bits (SMI_EN.TCO_EN stays as the
/// BIOS left it): clear GCS.NO_REBOOT (the BIOS leaves it SET on the NC110), clear stale status, TMR=50 ticks
/// (30 s per stage, reset ~60 s after the last pet), reload, un-halt. Returns true if the readback says armed.
pub fn arm_probe() -> bool {
    let lpc = pci::read32(pci::b0(31, 0), 0);
    let rcba_raw = pci::read32(pci::b0(31, 0), 0xF0);
    if !(lpc == 0x27BC_8086 || lpc == 0x2918_8086) || pmbase() == 0 || rcba_raw & 1 == 0 {
        kprintln!("wdt: probe arm: prerequisites fail (LPC {:#010x}, PMBASE {:#x}, RCBA {:#x})", lpc, pmbase(), rcba_raw);
        return false;
    }
    let b = base();
    let rcba = rcba_raw & 0xFFFF_C000;
    let gcs_p = (rcba + 0x3410) as *mut u32;
    unsafe {
        let g0 = read_volatile(gcs_p);
        let (cnt0, tmr0, rld0) = (inw(b + 8), inw(b + 0x12) & 0x3FF, inw(b) & 0x3FF);
        kprintln!("wdt: inherited: GCS {:#x} NO_REBOOT={} TCO1_CNT={:#06x} HLT={} TMR={} RLD={} ({})", g0, (g0 >> 5) & 1, cnt0,
            (cnt0 >> 11) & 1, tmr0, rld0, if g0 & 0x20 == 0 && cnt0 & 0x800 == 0 { "inherited enabled: NO_REBOOT clear, timer running" } else { "inherited not enabled" });
        core::ptr::write_volatile(gcs_p, g0 & !0x20);
        outw(b + 4, 0x0008); // TCO1_STS.TIMEOUT (RW1C)
        outw(b + 6, 0x0002); // TCO2_STS.SECOND_TO_STS
        outw(b + 6, 0x0004); // TCO2_STS.BOOT_STS
        outw(b + 0x12, (inw(b + 0x12) & !0x3FF) | 50);
        outw(b, 1);
        outw(b + 8, inw(b + 8) & !0x0800); // un-halt; bit 12 (lock) written back as read
        let (g1, cnt, tmr, c0) = (read_volatile(gcs_p), inw(b + 8), inw(b + 0x12) & 0x3FF, inw(b) & 0x3FF);
        delay_ms(1300);
        let c1 = inw(b) & 0x3FF;
        let ok = g1 & 0x20 == 0 && cnt & 0x0800 == 0 && tmr == 50 && c1 != c0;
        kprintln!("wdt: probe arm: GCS {:#x} -> {:#x} TCO1_CNT={:#06x} TMR={} RLD {}->{} SMI_EN={:#x} => {}",
            g0, g1, cnt, tmr, c0, c1, inl(pmbase() + 0x30), if ok { "ARMED" } else { "NOT ARMED" });
        ok
    }
}

pub fn pet() {
    unsafe { outw(base(), 1) };
}

/// ACPI PM timer (PMBASE+0x08, 3.579545 MHz, 24-bit) based delay; same block on ICH9 and NM10.
fn pm_timer() -> u32 {
    unsafe { inl(pmbase() + 8) & 0x00FF_FFFF }
}

pub fn delay_ms(ms: u32) {
    let need = ms as u64 * 3_579_545 / 1000;
    let mut last = pm_timer();
    let mut acc = 0u64;
    while acc < need {
        let now = pm_timer();
        acc += (now.wrapping_sub(last) & 0x00FF_FFFF) as u64;
        last = now;
    }
}

/// `wdt=pet:N` pets every 500 ms for N s, then stops and hangs. `wdt=hang` hangs at once.
pub fn rehearse(cmdline: &str) -> bool {
    let Some(i) = cmdline.find("wdt=") else { return false };
    let arg = cmdline[i + 4..].split(' ').next().unwrap_or("");
    if arg == "probe" {
        return false; // attended net, handled in main (k5 hung here before the NIC: result 5 counts as armed)
    }
    if !armed() {
        kprintln!("wdt: mode {} requested but watchdog not armed", arg);
        return false;
    }
    let secs: u32 = arg.strip_prefix("pet:").and_then(|n| n.parse().ok()).unwrap_or(0);
    for s in 0..secs {
        for _ in 0..2 {
            pet();
            delay_ms(500);
        }
        kprintln!("wdt: alive t={}s (petting every 500 ms)", s + 1);
    }
    kprintln!("wdt: HANG now: cli; hlt with the timer running, nobody pets");
    loop {
        unsafe { core::arch::asm!("cli; hlt") };
    }
}

/// ACPI PM timer (PMBASE+0x08, 3.579545 MHz, 24-bit) based delay; same block on ICH9 and NM10.
fn pm_timer() -> u32 {
    unsafe { inl(pmbase() + 8) & 0x00FF_FFFF }
}

pub fn delay_ms(ms: u32) {
    let need = ms as u64 * 3_579_545 / 1000;
    let mut last = pm_timer();
    let mut acc = 0u64;
    while acc < need {
        let now = pm_timer();
        acc += (now.wrapping_sub(last) & 0x00FF_FFFF) as u64;
        last = now;
    }
}

codex
The AEAD assembly looks correct, but **nonce allocation is not safe enough to extend to `poke` or kernel loading yet**. I found one cryptographic MUST-FIX and several bounded availability issues. No files edited; this was source review plus a small independent Python check, not a hardware flood test.

**MUST-FIX**

1. **Nonce uniqueness depends on wall-clock behavior, across every boot and client sharing the key.** [bmctl.py:43](~/nc110-baremetal/tools/bmctl.py:43), [bmctl.rs:218](~/nc110-baremetal/src/bmctl.rs:218)

   `time_ns() // 1000` is not guaranteed strictly increasing: clock rollback, concurrent invocations, or multiple controllers can repeat it. The kernel rejects duplicate sequences only **after the client has encrypted and transmitted**; that cannot undo C2K nonce reuse visible to a sniffer.

   Across boots, `last_seq` resets while the key remains unchanged. A repeated sequence can therefore also produce a second K2C encryption under the same nonce. **Changing boot-ID AAD does not change the keystream or Poly1305 one-time key.** I confirmed the unchanged ciphertext with Python’s independent AEAD implementation. Reuse exposes plaintext XOR and can enable authentication forgery; this is the nonce-uniqueness requirement in [RFC 8439](https://www.rfc-editor.org/rfc/rfc8439.html).

   This is conditional on sequence reuse—not an immediate keyless exploit from observing one normal exchange. Nevertheless, the protocol must enforce the invariant rather than assume clocks provide it.

   **Fix:** derive boot/session-specific traffic keys using HKDF with explicit protocol/direction labels and boot/session identifiers. Allocate sequences monotonically within each session, with coordination or distinct sessions for concurrent controllers. Boot-specific keys alone do not fix same-boot client collisions.

**SHOULD-FIX**

2. **Unbounded network polling can starve software reset deadlines.** [bmctl.rs:143](~/nc110-baremetal/src/bmctl.rs:143)

   The pinned smoltcp `Interface::poll()` drains ingress until empty and explicitly documents potentially unbounded work under sustained traffic. Both idle and requested-reboot checks occur afterward.

   An adequately fast LAN flood could therefore prevent reaching those checks. It also prevents further watchdog pets, so a functioning hardware watchdog should eventually fire; this is **not** an attack that keeps petting forever. The repository’s warning that TCO can hang this real board makes reliance on that fallback particularly concerning.

   Bound ingress work per iteration and check deadlines between batches. Actual exploitability at the netbook’s receive rate remains untested.

3. **UDP retries lose successful results, which matters for future mutations.** [bmctl.py:26](~/nc110-baremetal/tools/bmctl.py:26)

   If a command executes but its reply is lost, `rpc()` resends the identical request; the kernel rejects it as replay without returning the result. The caller reports failure despite successful execution.

   Before adding writes/loading, cache the exact encrypted response for an authenticated duplicate request and resend those bytes without executing again. Do not generate a different response under the original nonce.

4. **`who` is an unrestricted small reflector.** [bmctl.rs:164](~/nc110-baremetal/src/bmctl.rs:164)

   Three request bytes produce 43 response bytes: roughly 14× UDP-payload amplification, substantially less including packet overhead. Add a global response-rate limit; source-only limits are weak against spoofing.

   `log` is different: it requires authentication. A sniffer can race a captured, not-yet-accepted request with a spoofed source and redirect its one encrypted response, but accepted requests cannot be replayed repeatedly for amplification. I would not classify `log` as an unauthenticated bulk reflector.

5. **An authenticated log offset can panic the kernel.** [bmctl.rs:202](~/nc110-baremetal/src/bmctl.rs:202)

   `off + 1200` can overflow. For `usize::MAX`, release arithmetic wraps to 1199; with a sufficiently long log, the slice start exceeds its end and panics. The panic handler halts.

   Clamp the start first, then use saturating addition for the end. Low security severity today because the caller already has reboot authority, but worth fixing before expanding the parser.

**NIT / hardening**

- Use an established constant-time equality primitive and reject non-16-byte tags first. The current XOR/OR fold has no source-level mismatch early exit, but compiler-level constant-time behavior was not verified. **There is no truncated-tag acceptance bug:** length is checked before success, and the network caller always supplies exactly 16 bytes.
- Validate discovery response shape and boot-ID length, and filter reply endpoints. Today any sender can win `recvfrom()`, spoof discovery, or cause parsing/decryption failure. Endpoint filtering improves robustness but cannot authenticate against LAN source spoofing.
- Make the Q35 script fail its process on failed expectations. Currently several failures merely print `FAIL`; interop coverage also does not substitute for RFC vectors and boundary-length tests.

The remaining requested checks:

- **Construction:** consuming all 64 bytes of ChaCha block 0 correctly leaves encryption at counter 1; the first 32 bytes supply the Poly1305 key. Separate zero-padding of AAD/ciphertext and little-endian byte lengths are correct. Authentication precedes decryption.
- **Reflection:** `C2K\0` and `K2C\0` provide distinct nonces. Feeding a reply back as a command fails authentication.
- **Replay/petting:** only authenticated, increasing sequences advance `last_seq` and `last_ok`. `who`, junk, and accepted-frame replays cannot keep the idle timer alive. Captured but undelivered commands can be delayed and accepted once; sequence ordering provides no expiration.
- **Boot ID:** secrecy or unpredictability is unnecessary for its present AAD role; non-repetition matters. Prediction alone does not forge commands. Repetition permits old-boot replay after sequence reset. The RNG explicitly says its entropy claim has not been established on this Atom; health tests do not prove boot uniqueness.
- **SSH seed reuse:** the fixed domain prefix and high-entropy seed are reasonable separation for this narrow derivation; I see no direct cross-protocol break. Separate provisioning would improve compromise isolation. Arbitrary RAM reads or loaded code can expose the SSH seed and control keys.
- **Future authority:** unrestricted RAM writes, NIC DMA programming, and kernel loading amount to full kernel authority. “RAM + NIC MMIO only” does not meaningfully contain a compromised controller or loader.

Fix nonce/session design before extending authority, and make retries and reset deadlines dependable before relying on it remotely.

**MUST-FIX**

1. **Nonce uniqueness depends on wall-clock behavior, across every boot and client sharing the key.** [bmctl.py:43](~/nc110-baremetal/tools/bmctl.py:43), [bmctl.rs:218](~/nc110-baremetal/src/bmctl.rs:218)

   `time_ns() // 1000` is not guaranteed strictly increasing: clock rollback, concurrent invocations, or multiple controllers can repeat it. The kernel rejects duplicate sequences only **after the client has encrypted and transmitted**; that cannot undo C2K nonce reuse visible to a sniffer.

   Across boots, `last_seq` resets while the key remains unchanged. A repeated sequence can therefore also produce a second K2C encryption under the same nonce. **Changing boot-ID AAD does not change the keystream or Poly1305 one-time key.** I confirmed the unchanged ciphertext with Python’s independent AEAD implementation. Reuse exposes plaintext XOR and can enable authentication forgery; this is the nonce-uniqueness requirement in [RFC 8439](https://www.rfc-editor.org/rfc/rfc8439.html).

   This is conditional on sequence reuse—not an immediate keyless exploit from observing one normal exchange. Nevertheless, the protocol must enforce the invariant rather than assume clocks provide it.

   **Fix:** derive boot/session-specific traffic keys using HKDF with explicit protocol/direction labels and boot/session identifiers. Allocate sequences monotonically within each session, with coordination or distinct sessions for concurrent controllers. Boot-specific keys alone do not fix same-boot client collisions.

**SHOULD-FIX**

2. **Unbounded network polling can starve software reset deadlines.** [bmctl.rs:143](~/nc110-baremetal/src/bmctl.rs:143)

   The pinned smoltcp `Interface::poll()` drains ingress until empty and explicitly documents potentially unbounded work under sustained traffic. Both idle and requested-reboot checks occur afterward.

   An adequately fast LAN flood could therefore prevent reaching those checks. It also prevents further watchdog pets, so a functioning hardware watchdog should eventually fire; this is **not** an attack that keeps petting forever. The repository’s warning that TCO can hang this real board makes reliance on that fallback particularly concerning.

   Bound ingress work per iteration and check deadlines between batches. Actual exploitability at the netbook’s receive rate remains untested.

3. **UDP retries lose successful results, which matters for future mutations.** [bmctl.py:26](~/nc110-baremetal/tools/bmctl.py:26)

   If a command executes but its reply is lost, `rpc()` resends the identical request; the kernel rejects it as replay without returning the result. The caller reports failure despite successful execution.

   Before adding writes/loading, cache the exact encrypted response for an authenticated duplicate request and resend those bytes without executing again. Do not generate a different response under the original nonce.

4. **`who` is an unrestricted small reflector.** [bmctl.rs:164](~/nc110-baremetal/src/bmctl.rs:164)

   `log` is different: it requires authentication. A sniffer can race a captured, not-yet-accepted request with a spoofed source and redirect its one encrypted response, but accepted requests cannot be replayed repeatedly for amplification. I would not classify `log` as an unauthenticated bulk reflector.
