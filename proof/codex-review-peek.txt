//! Multiboot v1 info: cmdline, memory map, and the client-key module (M2.5).
//!
//! The SSH client key is NOT compiled into the ELF. GRUB loads it with `module /baremetal/<file>` (QEMU:
//! `-initrd <file>`), so the key lives on /boot and can be rotated without a rebuild.
//! Module format (72 bytes): b"NC110K1\n" | ed25519 seed (32) | ed25519 public key (32).
//! The kernel derives the public key from the seed and refuses on mismatch (truncation / corruption / wrong
//! file), copies the seed into a static, then zeroes the module bytes in RAM.
use crate::kprintln;
use core::ptr::{read_volatile, write_volatile};

const MAGIC: &[u8; 8] = b"NC110K1\n";
const KEYMOD_LEN: u64 = 72;

static mut RAM: [(u64, u64); 32] = [(0, 0); 32];
static mut NRAM: usize = 0;
static mut HAVE_MAP: bool = false;

unsafe fn rd32(a: u64) -> u32 {
    read_volatile(a as *const u32)
}

/// Record available-RAM ranges (mmap type 1, or mem_lower/mem_upper as a fallback).
pub fn parse_memory(info: u64) {
    unsafe {
        if info == 0 {
            return;
        }
        let flags = rd32(info);
        if flags & (1 << 6) != 0 {
            let len = rd32(info + 44) as u64;
            let mut p = rd32(info + 48) as u64;
            let end = p + len;
            while p + 24 <= end && NRAM < 32 {
                let size = rd32(p) as u64;
                let base = read_volatile((p + 4) as *const u64);
                let l = read_volatile((p + 12) as *const u64);
                let ty = rd32(p + 20);
                if ty == 1 && l > 0 {
                    RAM[NRAM] = (base, l);
                    NRAM += 1;
                }
                p += size + 4;
            }
            HAVE_MAP = true;
        } else if flags & 1 != 0 {
            RAM[0] = (0, rd32(info + 4) as u64 * 1024);
            RAM[1] = (0x10_0000, rd32(info + 8) as u64 * 1024);
            NRAM = 2;
            HAVE_MAP = true;
        }
    }
}

/// True if [base, base+len) touches available RAM, or if no memory map was given (fail closed).
pub fn ram_overlaps(base: u64, len: u64) -> bool {
    unsafe {
        if !HAVE_MAP {
            return true;
        }
        let ram = &*core::ptr::addr_of!(RAM);
        ram[..NRAM].iter().any(|&(b, l)| base < b + l && b < base + len)
    }
}

/// True only if [base, base+len) lies entirely inside ONE available-RAM range of the multiboot map.
pub fn ram_contains(base: u64, len: u64) -> bool {
    let Some(end) = base.checked_add(len) else { return false };
    unsafe {
        let ram = &*core::ptr::addr_of!(RAM);
        HAVE_MAP && ram[..NRAM].iter().any(|&(b, l)| base >= b && end <= b.saturating_add(l))
    }
}

pub fn cmdline(info: u64) -> &'static str {
    unsafe {
        if info == 0 || rd32(info) & 4 == 0 {
            return "";
        }
        let p = rd32(info + 16) as *const u8;
        let mut n = 0;
        while n < 256 && *p.add(n) != 0 {
            n += 1;
        }
        core::str::from_utf8(core::slice::from_raw_parts(p, n)).unwrap_or("")
    }
}

/// Find the key module, validate it, copy the seed out, wipe the module. None (with a log line) if absent/bad.
pub fn take_client_key(info: u64) -> Option<[u8; 32]> {
    unsafe {
        if info == 0 || rd32(info) & (1 << 3) == 0 || rd32(info + 20) == 0 {
            kprintln!("keymod: no multiboot module");
            return None;
        }
        let count = rd32(info + 20).min(16) as u64;
        let mods = rd32(info + 24) as u64;
        for i in 0..count {
            let e = mods + 16 * i;
            let (start, end) = (rd32(e) as u64, rd32(e + 4) as u64);
            if end <= start || end - start != KEYMOD_LEN {
                continue;
            }
            let m = core::slice::from_raw_parts(start as *const u8, KEYMOD_LEN as usize);
            if &m[..8] != MAGIC {
                continue;
            }
            let mut seed = [0u8; 32];
            let mut publ = [0u8; 32];
            seed.copy_from_slice(&m[8..40]);
            publ.copy_from_slice(&m[40..72]);
            for j in 0..KEYMOD_LEN {
                write_volatile((start + j) as *mut u8, 0);
            }
            let derived = sunset::ed25519_dalek::SigningKey::from_bytes(&seed).verifying_key().to_bytes();
            if derived != publ {
                kprintln!("keymod: module {} public key does not match its seed -> ignored (wiped)", i);
                return None;
            }
            kprintln!("keymod: client key from module {} at {:#x} (72 B), public key verified, module wiped", i, start);
            return Some(seed);
        }
        kprintln!("keymod: {} module(s), none is a 72-byte NC110K1 key", count);
        None
    }
}

//! Realtek C+ descriptor-ring NIC driver, two chips behind a PCI device-id match (M2.5):
//!   10ec:8136 RTL8101E family (the NC110's wired NIC): MMIO only, r8169-style init. NOT RUNNABLE IN QEMU.
//!   10ec:8139 rtl8139 in C+ mode (QEMU stand-in): port I/O BAR0, or MMIO BAR1 with cmdline `nic=mmio`.
//! Both use the same 16-byte descriptors and ring registers; the per-chip deltas live in nic_regs.rs and are
//! checked against pinned Linux v6.12 sources on the host (tools/test-nic-regs.sh).
//!
//! Safety rules this file keeps (spec rule 4: no disk access, ever):
//! - The only PCI config write is to the matched NIC's command register (low 16 bits; status written as 0).
//! - MMIO base must be a 32-bit-reachable memory BAR, 256-byte aligned, outside multiboot RAM, and unable to
//!   overlap any mass-storage (class 01h, e.g. SATA/AHCI) BAR (span bounded by its alignment, read-only).
//!   Otherwise the NIC is refused.
//! - Bus mastering is OFF across the chip reset and ring programming (iPXE leaves the NIC DMAing), and only
//!   turned on after the rings point at our buffers.

#[repr(C)]
#[derive(Clone, Copy)]
struct Desc {
    opts1: u32,
    opts2: u32,
    addr: u64,
}

#[repr(C, align(256))]
struct Ring<const N: usize>([Desc; N]);

#[repr(C, align(16))]
struct Bufs<const N: usize>([[u8; BUFSZ]; N]);

const Z: Desc = Desc { opts1: 0, opts2: 0, addr: 0 };
static mut TX_DESC: Ring<1> = Ring([Z; 1]);
static mut RX_DESC: Ring<NRX> = Ring([Z; NRX]);
static mut TX_BUF: Bufs<1> = Bufs([[0; BUFSZ]; 1]);
static mut RX_BUF: Bufs<NRX> = Bufs([[0; BUFSZ]; NRX]);

#[derive(Clone, Copy)]
pub enum Bus {
    Pio(u16),
    Mmio(u64),
}

impl Bus {
    unsafe fn r8(self, o: u16) -> u8 {
        match self {
            Bus::Pio(p) => inb(p + o),
            Bus::Mmio(m) => read_volatile((m + o as u64) as *const u8),
        }
    }
    unsafe fn r16(self, o: u16) -> u16 {
        match self {
            Bus::Pio(p) => inw(p + o),
            Bus::Mmio(m) => read_volatile((m + o as u64) as *const u16),
        }
    }
    unsafe fn r32(self, o: u16) -> u32 {
        match self {
            Bus::Pio(p) => inl(p + o),
            Bus::Mmio(m) => read_volatile((m + o as u64) as *const u32),
        }
    }
    unsafe fn w8(self, o: u16, v: u8) {
        match self {
            Bus::Pio(p) => outb(p + o, v),
            Bus::Mmio(m) => write_volatile((m + o as u64) as *mut u8, v),
        }
    }
    unsafe fn w16(self, o: u16, v: u16) {
        match self {
            Bus::Pio(p) => outw(p + o, v),
            Bus::Mmio(m) => write_volatile((m + o as u64) as *mut u16, v),
        }
    }
    unsafe fn w32(self, o: u16, v: u32) {
        match self {
            Bus::Pio(p) => outl(p + o, v),
            Bus::Mmio(m) => write_volatile((m + o as u64) as *mut u32, v),
        }
    }
}

const PCI_CMD_IO: u32 = 1 << 0;
const PCI_CMD_MEM: u32 = 1 << 1;
const PCI_CMD_BM: u32 = 1 << 2;

/// The one PCI config write: the NIC's command register. Upper half (status, RW1C) written as 0.
fn set_pci_cmd(b: Bdf, set: u32, clear: u32) {
    let cmd = pci::read32(b, 0x04) & 0xFFFF;
    pci::write32(b, 0x04, (cmd & !clear) | set);
}

extern "C" {
    static mut pd: [u64; 2048]; // boot.s: 4 x 512 2-MiB identity entries covering 0..4 GiB
}

#[repr(C, align(4096))]
struct Pt([u64; 512]);
static mut UC_PT: Pt = Pt([0; 512]);

/// Make ONLY the 4 KiB page holding the NIC registers uncached (PCD|PWT): split the 2 MiB identity PDE that
/// covers it into 512 identity 4 KiB entries with the PDE's own flags, so nothing else in that 2 MiB (on the
/// NM10 possibly the AHCI ABAR) changes attributes. Review fix (GLM-5.2 B1): previously the whole 2 MiB went UC.
fn map_uc_4k(base: u64) {
    unsafe {
        let pde = addr_of_mut!(pd[(base >> 21) as usize]);
        let old = read_volatile(pde);
        let region = base & !0x1F_FFFF;
        let flags = old & 0x1F & !(1 << 7); // P, RW, US, PWT, PCD from the PDE; PS dropped for a PTE
        let pt = &mut *addr_of_mut!(UC_PT);
        for (i, e) in pt.0.iter_mut().enumerate() {
            *e = (region + (i as u64) * 4096) | flags;
        }
        pt.0[((base >> 12) & 0x1FF) as usize] |= (1 << 4) | (1 << 3);
        fence(Ordering::SeqCst);
        write_volatile(pde, (addr_of!(UC_PT) as u64) | 0x3); // present | rw, points at UC_PT
        core::arch::asm!("mov {t}, cr3", "mov cr3, {t}", t = out(reg) _, options(nostack)); // flush incl. the 2 MiB entry
    }
}

/// Validate a memory BAR for use as the NIC register window, then map its 2 MiB page uncached (PCD|PWT).
fn mmio_window(b: Bdf) -> Result<u64, &'static str> {
    let (base, _is64) = pci::first_mem_bar(b).ok_or("no memory BAR")?;
    if base == 0 {
        return Err("memory BAR unassigned (0)");
    }
    if base & 0xFF != 0 || base + 0x100 > 0x1_0000_0000 {
        return Err("memory BAR not 256-byte aligned or not below 4 GiB");
    }
    if crate::ram_overlaps(base, 0x100) {
        return Err("memory BAR overlaps RAM in the multiboot memory map");
    }
    let mut st = [0u64; 16];
    let n = pci::storage_mem_bars(&mut st);
    for &x in &st[..n] {
        kprintln!("nic: mass-storage memory BAR {:#x} (max span {:#x}), NIC window {:#x}+0x100", x, 1u64 << x.trailing_zeros().min(32), base);
    }
    if !R::clear_of_storage(base, &st[..n]) {
        return Err("NIC register window could overlap a storage controller BAR");
    }
    map_uc_4k(base);
    Ok(base)
}

/// Subset of rtl8169_get_mac_version (r8169_main.c:2318-2325, first match wins): the 8101E/8102E/8103E

pub fn probe(prefer_mmio: bool) -> Result<Rtl, &'static str> {
    if let Some(b) = pci::find(0x10EC, 0x8136) {
        return probe_8101e(b);
    }
    if let Some(b) = pci::find(0x10EC, 0x8139) {
        return probe_8139cp(b, prefer_mmio);
    }
    Err("no 10ec:8136 or 10ec:8139 found (bridges walked)")
}

fn probe_8139cp(b: Bdf, mmio: bool) -> Result<Rtl, &'static str> {
    let bus = if mmio {
        set_pci_cmd(b, 0, PCI_CMD_BM);
        let m = mmio_window(b)?;
        set_pci_cmd(b, PCI_CMD_MEM, PCI_CMD_BM);
        Bus::Mmio(m)
    } else {
        set_pci_cmd(b, PCI_CMD_IO, PCI_CMD_BM);
        let bar0 = pci::read32(b, 0x10);
        if bar0 & 1 == 0 {
            return Err("rtl8139 BAR0 is not I/O");
        }
        Bus::Pio((bar0 & !0x3) as u16)
    };
    let mut mac = [0u8; 6];
    for (i, m) in mac.iter_mut().enumerate() {
        *m = unsafe { bus.r8(R::RTL8139CP.mac0 + i as u16) };
    }
    Ok(Rtl { bus, regs: &R::RTL8139CP, bdf: b, mac, rx_next: 0, rx_cfg: 0x0000_E70A, ver: 0 })
}

            for i in 0..NRX {
                let d = addr_of_mut!(RX_DESC.0[i]);
                let mut o = r.desc_own | BUFSZ as u32;

                    o |= r.ring_end;

            self.bus.w32(r.tx_ring, tx as u32);
            self.bus.w32(r.rx_ring + 4, (rx >> 32) as u32);
            self.bus.w32(r.rx_ring, rx as u32);

        }
        let t = time::now_ms();
        while unsafe { self.bus.r8(r.phystatus) } & R::LINK_STATUS == 0 && time::now_ms() - t < 5000 {}
        let st = unsafe { self.bus.r8(r.phystatus) };
        kprintln!("nic: PHYstatus {:#04x} link {}", st, if st & R::LINK_STATUS != 0 { "up" } else { "DOWN after 5 s" });
        if st & R::LINK_STATUS == 0 {
            // Review fix (GLM-5.2 C3): no link -> stop the chip and bus mastering instead of running SSH blind.
            unsafe { self.bus.w8(r.chip_cmd, 0) };
            set_pci_cmd(self.bdf, 0, PCI_CMD_BM);
            return Err("no link after 5 s (Rx/Tx and bus mastering turned off again)");
        }
        Ok(())
    }

    /// Non-blocking: if the next RX descriptor is ours, return (index, frame length without FCS).
    /// The caller must hand the descriptor back with `rx_release(index)`.
    pub fn rx_poll(&mut self) -> Option<(usize, usize)> {
        let r = self.regs;
        let i = self.rx_next;
        unsafe {
            let o = read_volatile(addr_of!(RX_DESC.0[i].opts1));
            if o & r.desc_own != 0 {
                return None;
            }
            fence(Ordering::SeqCst);
            self.rx_next = (i + 1) % NRX;
            if o & (r.rx_err | r.first_frag | r.last_frag) != (r.first_frag | r.last_frag) {
                rx_release(r, i); // errored or multi-descriptor frame: drop it
                return None;
            }
            Some((i, ((o & r.rx_len_mask) as usize).saturating_sub(4).min(BUFSZ)))
        }
    }
}

/// Only valid between `rx_poll` returning (i, len) and `rx_release(i)` (net::Rx enforces this).
pub(crate) fn rx_buf(i: usize, len: usize) -> &'static [u8] {
    assert!(i < NRX && len <= BUFSZ);
    unsafe { core::slice::from_raw_parts(addr_of!(RX_BUF.0[i]) as *const u8, len) }
}

pub fn rx_release(r: &Regs, i: usize) {
    let mut no = r.desc_own | BUFSZ as u32;
    if i == NRX - 1 {
        no |= r.ring_end;
    }
    fence(Ordering::SeqCst);
    unsafe { write_volatile(addr_of_mut!(RX_DESC.0[i].opts1), no) };
}

/// Set once a TX descriptor fails to come back (OWN still set after the deadline). From then on the
/// TX DMA buffer and descriptor are treated as owned by the NIC forever: nothing is written to them.
static mut TX_WEDGED: bool = false;
static mut TX_SCRATCH: [u8; BUFSZ] = [0; BUFSZ];

/// Let `fill` build a `len`-byte frame directly in the DMA buffer, then send it.
/// Never touches the buffer/descriptor unless the NIC has cleared OWN (Codex M2 MUST-FIX).
pub fn tx_with<R2>(bus: Bus, r: &Regs, len: usize, fill: impl FnOnce(&mut [u8]) -> R2) -> R2 {
    let len = len.min(BUFSZ);
    unsafe {
        let d = addr_of_mut!(TX_DESC.0[0]);
        if TX_WEDGED || read_volatile(addr_of!((*d).opts1)) & r.desc_own != 0 {
            TX_WEDGED = true;
            // Frame is built into scratch and dropped: the NIC may still be DMAing the real buffer.
            let scratch = &mut *core::ptr::addr_of_mut!(TX_SCRATCH);
            return fill(&mut scratch[..len]);
        }
        let buf = core::slice::from_raw_parts_mut(addr_of_mut!(TX_BUF.0[0]) as *mut u8, BUFSZ);
        let res = fill(&mut buf[..len]);
        let wire = len.max(60);
        for b in &mut buf[len..wire] {
            *b = 0;
        }
        write_volatile(
            d,
            Desc {
                opts1: r.desc_own | r.ring_end | r.first_frag | r.last_frag | wire as u32,
                opts2: 0,
                addr: buf.as_ptr() as u64,
            },
        );
        fence(Ordering::SeqCst);
        bus.w8(r.tx_poll, r.npq);
        let t0 = rdtsc();
        while read_volatile(addr_of!((*d).opts1)) & r.desc_own != 0 {
            if rdtsc() - t0 > 2_000_000_000 {
                TX_WEDGED = true;
                kprintln!("nic: TX descriptor not returned by NIC, TX disabled");
                break;
            }
            core::hint::spin_loop();
        }
        fence(Ordering::SeqCst);
        let st = bus.r16(r.intr_status);
        bus.w16(r.intr_status, st);
        res
    }
}
//! bmctl: authenticated UDP control port (5515) so an agent can inspect and drive the kernel over the LAN, the way
//! the DS client is driven over net2. Frame: [seq u64 LE][ciphertext][16-byte tag], ChaCha20-Poly1305 (RFC 8439),
//! nonce = b"C2K\0"|b"K2C\0" ++ seq, AAD = this boot's random 16-byte id. Master = SHA-256("nc110-bmctl-v1" ++
//! client seed from the key module): nothing secret is compiled in. Handshake (Codex MUST-FIX x2, nonce reuse):
//! `who` ++ a fresh random 16-byte client nonce Nc is the only clear-text request; the reply carries the boot id, a
//! NEW session number and check = SHA-256("nc110-bmctl-v1 who" ++ master ++ Nc ++ boot id ++ session)[..16].
//! Traffic key = SHA-256("nc110-bmctl-v1 session" ++ master ++ boot id ++ session ++ Nc); seq counts from 1. A
//! replayed or forged `who` reply cannot make two client runs share a key, because Nc is fresh per run, and the
//! kernel never reissues a session number in a boot. A session binds to the endpoint that asked `who`; pending and authenticated sessions live in separate tables.
//! A duplicate of the last accepted seq gets the cached reply bytes back (safe UDP retries, nothing re-executed).
//! The loop pets the watchdog; 10 minutes without a valid command -> 0xCF9 reset.
//! `peek` is read-only and allowlisted (see `peek`): RAM only inside the multiboot map and never the kernel image
//! (text, data, bss: stack, session table, keys), NIC registers only inside its 256-byte MMIO window as aligned
//! dwords, ports only the ACPI PM block (PMBASE..+0x80, which contains the TCO); everything else is refused.
use crate::net::Nic;
use crate::nic::{Bus, Rtl};
use crate::port::{inb, inl, inw};
use crate::{console, kprintln, rng, time, wdt};
use chacha20::cipher::{KeyIvInit, StreamCipher};
use chacha20::ChaCha20;
use poly1305::universal_hash::{KeyInit, UniversalHash};
use poly1305::Poly1305;
use sha2::{Digest, Sha256};
use smoltcp::iface::{Config, Interface, SocketSet, SocketStorage};
use smoltcp::socket::{dhcpv4, udp};
use smoltcp::time::Instant;
use smoltcp::wire::{EthernetAddress, HardwareAddress, IpAddress, IpCidr, IpEndpoint, Ipv4Address};

const PORT: u16 = 5515;
const IDLE_MS: i64 = 600_000;

static mut SESS: [Option<Sess>; 16] = [const { None }; 16]; // authenticated
static mut PEND: [Option<Sess>; 8] = [const { None }; 8]; // issued by `who`, no good frame yet
static mut SOCKS: [SocketStorage<'static>; 2] = [SocketStorage::EMPTY; 2];
static mut RX_META: [udp::PacketMetadata; 8] = [udp::PacketMetadata::EMPTY; 8];
static mut RX_BUF: [u8; 8192] = [0; 8192];
static mut TX_META: [udp::PacketMetadata; 8] = [udp::PacketMetadata::EMPTY; 8];
static mut TX_BUF: [u8; 8192] = [0; 8192];

fn now() -> Instant {
    Instant::from_millis(time::now_ms())
}

fn nonce(dir: &[u8; 4], seq: u64) -> [u8; 12] {
    let mut n = [0u8; 12];
    n[..4].copy_from_slice(dir);
    n[4..].copy_from_slice(&seq.to_le_bytes());
    n
}

/// RFC 8439 tag over aad and ct with the one-time key from keystream block 0.
fn tag(pk: &[u8; 32], aad: &[u8], ct: &[u8]) -> [u8; 16] {
    let mut m = Poly1305::new(pk.into());
    m.update_padded(aad);
    m.update_padded(ct);
    let mut lens = [0u8; 16];
    lens[..8].copy_from_slice(&(aad.len() as u64).to_le_bytes());
    lens[8..].copy_from_slice(&(ct.len() as u64).to_le_bytes());
    m.update_padded(&lens);
    m.finalize().into()
}

fn cipher(key: &[u8; 32], n: &[u8; 12]) -> (ChaCha20, [u8; 32]) {
    let mut c = ChaCha20::new(key.into(), n.into());
    let mut b0 = [0u8; 64];
    c.apply_keystream(&mut b0); // block 0 -> Poly1305 key; the stream continues at block 1
    let mut pk = [0u8; 32];
    pk.copy_from_slice(&b0[..32]);
    (c, pk)
}

pub fn seal(key: &[u8; 32], n: &[u8; 12], aad: &[u8], buf: &mut [u8]) -> [u8; 16] {
    let (mut c, pk) = cipher(key, n);
    c.apply_keystream(buf);
    tag(&pk, aad, buf)
}

pub fn open(key: &[u8; 32], n: &[u8; 12], aad: &[u8], buf: &mut [u8], t: &[u8]) -> bool {
    let (mut c, pk) = cipher(key, n);
    let want = tag(&pk, aad, buf);
    if t.len() != 16 {
        return false;
    }
    let diff = want.iter().zip(t).fold(0u8, |a, (x, y)| a | (x ^ y));
    if diff != 0 {
        return false;
    }
    c.apply_keystream(buf);
    true
}

pub fn session_key(master: &[u8; 32], boot_id: &[u8; 16], sess: u32, nc: &[u8; 16]) -> [u8; 32] {
    let mut h = Sha256::new();
    h.update(b"nc110-bmctl-v1 session");
    h.update(master);
    h.update(boot_id);
    h.update(sess.to_le_bytes());
    h.update(nc);
    h.finalize().into()
}

fn who_check(master: &[u8; 32], nc: &[u8; 16], boot_id: &[u8; 16], sess: u32) -> [u8; 32] {
    let mut h = Sha256::new();
    h.update(b"nc110-bmctl-v1 who");
    h.update(master);
    h.update(nc);
    h.update(boot_id);
    h.update(sess.to_le_bytes());
    h.finalize().into()
}

struct Sess {
    id: u32,
    key: [u8; 32],
    last_seq: u64,
    peer: Option<IpEndpoint>,
    last_t: i64,
    retx: u8,
    reply: [u8; 8 + 1300 + 16],
    reply_len: usize,
}

pub fn derive_key(seed: &[u8; 32]) -> [u8; 32] {
    let mut h = Sha256::new();
    h.update(b"nc110-bmctl-v1");
    h.update(seed);
    h.finalize().into()
}

fn hex(b: &[u8], out: &mut heapless::String<64>) {
    for x in b {
        let _ = core::fmt::write(out, format_args!("{:02x}", x));
    }
}

extern "C" {
    static __load_start: u8;
    static __bss_end: u8;
}

/// Allowlisted read. Returns Err(reason) for anything outside the policy; never touches refused addresses.
fn peek(w: &mut Writer, nic_mmio: Option<u64>, what: &str, a: &str, b: &str) -> Result<(), &'static str> {
    let addr = u64::from_str_radix(a.trim_start_matches("0x"), 16).map_err(|_| "addr: hex")?;
    let n: u64 = b.parse().map_err(|_| "len/width: decimal")?;
    match what {
        "mem" => {
            if n == 0 || n > 512 {
                return Err("len 1..512");
            }
            let end = addr.checked_add(n).ok_or("overflow")?;
            let (ks, ke) = (core::ptr::addr_of!(__load_start) as u64, core::ptr::addr_of!(__bss_end) as u64);
            if addr < ke && ks < end {
                return Err("refused: kernel image (code, stack, keys)");
            }
            if end > 0x1_0000_0000 || !crate::mb::ram_contains(addr, n) {
                return Err("refused: not inside one multiboot RAM range below 4 GiB");
            }
            for i in 0..n {
                let v = unsafe { core::ptr::read_volatile((addr + i) as *const u8) };
                let _ = core::fmt::write(w, format_args!("{:02x}", v));
            }
        }
        "nic" => {
            let base = nic_mmio.ok_or("refused: NIC not on MMIO")?;
            if addr % 4 != 0 || n % 4 != 0 || n == 0 || addr.checked_add(n).is_none_or(|e| e > 0x100) {
                return Err("nic: dword-aligned offset/len inside 0..0x100");
            }
            for o in (addr..addr + n).step_by(4) {
                let v = unsafe { core::ptr::read_volatile((base + o) as *const u32) };
                let _ = core::fmt::write(w, format_args!("{:08x} ", v));
            }
        }
        "io" => {
            let pm = wdt::pmbase() as u64;
            if pm == 0 || !matches!(n, 1 | 2 | 4) || addr % n != 0 || addr < pm || addr + n > pm + 0x80 {
                return Err("io: width 1|2|4, aligned, inside PMBASE..PMBASE+0x80 only");
            }
            let p = addr as u16;
            let v = unsafe { match n { 1 => inb(p) as u32, 2 => inw(p) as u32, _ => inl(p) } };
            let _ = core::fmt::write(w, format_args!("{:#x}", v));
        }
        _ => return Err("peek mem <hexaddr> <len> | peek nic <hexoff> <len> | peek io <hexport> <1|2|4>"),
    }
    Ok(())
}

pub fn run(nic: Rtl, seed: [u8; 32]) {
    let nic_mmio = match nic.bus {
        Bus::Mmio(b) => Some(b),
        Bus::Pio(_) => None,
    };
    if rng::init(false).is_err() {
        return;
    }
    let key = derive_key(&seed);
    let mut boot_id = [0u8; 16];
    let mut rs = [0u8; 8];
    if rng::fill(&mut boot_id).is_err() || rng::fill(&mut rs).is_err() {
        kprintln!("bmctl: entropy failed -> not starting");
        return;
    }
    let mut bid = heapless::String::<64>::new();
    hex(&boot_id, &mut bid);
    let mac = nic.mac;
    let mut dev = Nic(nic);
    let mut cfg = Config::new(HardwareAddress::Ethernet(EthernetAddress(mac)));
    cfg.random_seed = u64::from_le_bytes(rs);
    let mut iface = Interface::new(cfg, &mut dev, now());
    let mut sockets = SocketSet::new(unsafe { &mut *core::ptr::addr_of_mut!(SOCKS) }[..].as_mut());
    let dhcp_h = sockets.add(dhcpv4::Socket::new());
    let udp_h = sockets.add(udp::Socket::new(
        udp::PacketBuffer::new(unsafe { &mut *core::ptr::addr_of_mut!(RX_META) }[..].as_mut(), unsafe { &mut *core::ptr::addr_of_mut!(RX_BUF) }[..].as_mut()),
        udp::PacketBuffer::new(unsafe { &mut *core::ptr::addr_of_mut!(TX_META) }[..].as_mut(), unsafe { &mut *core::ptr::addr_of_mut!(TX_BUF) }[..].as_mut()),
    ));
    let t0 = time::now_ms();
    let mut ip = None;
    while ip.is_none() && time::now_ms() - t0 < 15_000 {
        wdt::pet();
        crate::net::rx_budget(Some(32));
        iface.poll(now(), &mut dev, &mut sockets);
        if let Some(dhcpv4::Event::Configured(c)) = sockets.get_mut::<dhcpv4::Socket>(dhcp_h).poll() {
            iface.update_ip_addrs(|a| {
                a.clear();
                let _ = a.push(IpCidr::Ipv4(c.address));
            });
            if let Some(gw) = c.router {
                let _ = iface.routes_mut().add_default_ipv4_route(gw);
            }
            ip = Some(c.address);
        }
    }
    let Some(ip) = ip else {
        kprintln!("bmctl: no DHCP lease in 15 s -> not starting");
        return;
    };
    sockets.get_mut::<udp::Socket>(udp_h).bind(PORT).expect("bind");
    kprintln!("bmctl: listening on {}:{} boot id {} (idle reset after {} s)", ip.address(), PORT, bid, IDLE_MS / 1000);
    crate::udplog::dump(5);

    let (mut last_ok, mut last_hello) = (time::now_ms(), 0i64);
    let (mut ok, mut bad, mut replay, mut retx) = (0u32, 0u32, 0u32, 0u32);
    let (mut next_sess, mut who_window, mut who_count) = (1u32, 0i64, 0u32);
    let sess = unsafe { &mut *core::ptr::addr_of_mut!(SESS) };
    let pend = unsafe { &mut *core::ptr::addr_of_mut!(PEND) };
    let mut reboot_at: Option<i64> = None;
    loop {
        wdt::pet();
        crate::net::rx_budget(Some(32)); // bounded ingress per poll so the deadline checks below always run
        iface.poll(now(), &mut dev, &mut sockets);
        let t = time::now_ms();
        if t - last_ok > IDLE_MS {
            kprintln!("bmctl: idle {} s -> reset", IDLE_MS / 1000);
            return;
        }
        if reboot_at.is_some_and(|r| t > r) {
            kprintln!("bmctl: reboot requested");
            wdt::hard_reset();
        }
        let sock = sockets.get_mut::<udp::Socket>(udp_h);
        if t - last_hello > 2000 {
            last_hello = t;
            let mut h = heapless::String::<128>::new();
            let _ = core::fmt::write(&mut h, format_args!("nc110 hello bmctl {}:{} boot {}", ip.address(), PORT, bid));
            let _ = sock.send_slice(h.as_bytes(), IpEndpoint::new(IpAddress::Ipv4(Ipv4Address::BROADCAST), 5514));
        }
        let mut buf = [0u8; 1500];
        let Ok((n, meta)) = sock.recv_slice(&mut buf) else { continue };
        let from = meta.endpoint;
        if n == 3 + 16 && &buf[..3] == b"who" {
            if t - who_window > 1000 {
                (who_window, who_count) = (t, 0);
            }
            who_count += 1;
            if who_count > 10 {
                continue; // global cap: at most 10 clear-text replies per second (reflection)
            }
            // Unauthenticated `who` only ever writes the pending table (round-robin), never an authenticated session.
            let slot = Some((next_sess as usize) % pend.len());
            let (Some(slot), Some(next)) = (slot, next_sess.checked_add(1)) else { continue };
            let id = next_sess;
            next_sess = next;
            let mut nc = [0u8; 16];
            nc.copy_from_slice(&buf[3..19]);
            pend[slot] = Some(Sess { id, key: session_key(&key, &boot_id, id, &nc), last_seq: 0, peer: Some(from), last_t: t, retx: 0, reply: [0; 8 + 1300 + 16], reply_len: 0 });
            let mut chk = heapless::String::<64>::new();
            hex(&who_check(&key, &nc, &boot_id, id)[..16], &mut chk);
            let mut h = heapless::String::<160>::new();
            let _ = core::fmt::write(&mut h, format_args!("nc110 boot {} session {} check {}", bid, id, chk));
            let _ = sock.send_slice(h.as_bytes(), from);
            continue;
        }
        // Frame: [session u32 LE][seq u64 LE][ct][tag]
        if n < 4 + 8 + 16 {
            bad += 1;
            continue;
        }
        let sid = u32::from_le_bytes(buf[..4].try_into().unwrap());
        // Authenticated session, or a pending one from the endpoint that asked `who` (promoted below once the frame
        // authenticates; only then may it evict the least recently used authenticated session).
        let in_sess = sess.iter().position(|x| x.as_ref().is_some_and(|x| x.id == sid && x.peer == Some(from)));
        let in_pend = pend.iter().position(|x| x.as_ref().is_some_and(|x| x.id == sid && x.peer == Some(from)));
        let se = match (in_sess, in_pend) {
            (Some(i), _) => sess[i].as_mut().unwrap(),
            (None, Some(i)) => pend[i].as_mut().unwrap(),
            _ => {
                bad += 1;
                continue;
            }
        };
        let seq = u64::from_le_bytes(buf[4..12].try_into().unwrap());
        let (body, tg) = buf[12..n].split_at_mut(n - 12 - 16);
        if !open(&se.key, &nonce(b"C2K\0", seq), &boot_id, body, tg) {
            bad += 1;
            continue;
        }
        if seq == se.last_seq && se.reply_len > 0 {
            if se.retx >= 5 {
                continue; // cap resends of one reply
            }
            se.retx += 1;
            retx += 1; // retry of the last request: same bytes again (to the bound peer), nothing re-executed
            let _ = sock.send_slice(&se.reply[..se.reply_len], from);
            continue;
        }
        if seq <= se.last_seq {
            replay += 1;
            continue;
        }
        se.last_seq = seq;
        se.last_t = t;
        se.retx = 0;
        last_ok = t;
        let se = if let (None, Some(i)) = (in_sess, in_pend) {
            let dst = (0..sess.len()).find(|&j| sess[j].is_none())
                .unwrap_or_else(|| (0..sess.len()).min_by_key(|&j| sess[j].as_ref().map_or(i64::MIN, |x| x.last_t)).unwrap());
            sess[dst] = pend[i].take();
            sess[dst].as_mut().unwrap()
        } else {
            se
        };
        ok += 1;
        let mut out = [0u8; 1300];
        let len = {
            let mut w = Writer { b: &mut out, n: 0 };
            let cmd = core::str::from_utf8(body).unwrap_or("");
            let mut it = cmd.split(' ');
            match it.next().unwrap_or("") {
                "ping" => {
                    let _ = core::fmt::write(&mut w, format_args!("pong uptime_ms={} ip={}", t, ip.address()));
                }
                "stats" => {
                    let _ = core::fmt::write(&mut w, format_args!("ok={} bad={} replay={} retx={} sessions_issued={} log_bytes={}", ok, bad, replay, retx, next_sess - 1, console::captured().len()));
                }
                "log" => {
                    let off: usize = it.next().and_then(|x| x.parse().ok()).unwrap_or(0);
                    let l = console::captured();
                    let st = off.min(l.len());
                    let s = &l[st..st.saturating_add(1200).min(l.len())];
                    w.put(s);
                }
                "peek" => {
                    let (what, x, y) = (it.next().unwrap_or(""), it.next().unwrap_or(""), it.next().unwrap_or(""));
                    if let Err(e) = peek(&mut w, nic_mmio, what, x, y) {
                        w.n = 0;
                        let _ = core::fmt::write(&mut w, format_args!("err {}", e));
                    }
                }
                "reboot" => {
                    let _ = core::fmt::write(&mut w, format_args!("rebooting in 500 ms"));
                    reboot_at = Some(t + 500);
                }
                _ => {
                    let _ = core::fmt::write(&mut w, format_args!("err unknown command (ping|stats|log <off>|peek ...|reboot)"));
                }
            }
            w.n
        };
        let pkt = &mut se.reply;
        pkt[..8].copy_from_slice(&seq.to_le_bytes());
        pkt[8..8 + len].copy_from_slice(&out[..len]);
        let tg = seal(&se.key, &nonce(b"K2C\0", seq), &boot_id, &mut pkt[8..8 + len]);
        pkt[8 + len..8 + len + 16].copy_from_slice(&tg);
        se.reply_len = 8 + len + 16;
        let _ = sock.send_slice(&se.reply[..se.reply_len], from);
    }
}

struct Writer<'a> {
    b: &'a mut [u8],
    n: usize,
}
impl Writer<'_> {
    fn put(&mut self, s: &[u8]) {
        let k = s.len().min(self.b.len() - self.n);
        self.b[self.n..self.n + k].copy_from_slice(&s[..k]);
        self.n += k;
    }
}
impl core::fmt::Write for Writer<'_> {
    fn write_str(&mut self, s: &str) -> core::fmt::Result {
        self.put(s.as_bytes());
        Ok(())
    }
}

- **MUST-FIX — Port-range overflow** ([bmctl.rs:175](~/nc110-baremetal/src/bmctl.rs:175)). `peek io ffffffffffffffff 1` wraps `addr+n` to zero in release, passes validation, then reads port `0xffff`. Widths 2/4 similarly expose `0xfffe`/`0xfffc`. This is a narrow escape, **not arbitrary port access**. Checked-overflow builds panic instead. Use `checked_add` and validate before narrowing to `u16`.

- **MUST-FIX — NM10 PM reads can stop execution.** Allowed offsets `PMBASE+0x14/15/16` request C2/C3/C4; execution resumes only on a break event. Thus an authenticated peek can disrupt or potentially hang this polling kernel. Exclude every access overlapping these registers, including wider reads; preferably allowlist documented register/width pairs. TCO status is write-one-to-clear, and reading TCO_RLD does **not** reload it—the concrete danger here is power-state entry. [Intel NM10 datasheet, §§13.8.3.6–8, 13.9](https://www.mouser.com/pdfdocs/82NM10datasheet.pdf).
